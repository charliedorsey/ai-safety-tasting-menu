    3/-!
    4# Quillon CSC3 F5a: the R-route engine (VK along a residue class, plus Cauchy with a class split)
    5
    6Inputs: `GoldB.shiftedSuzukiCore_unconditional` (Hunt B, GREEN). No other analytic input.
    7
    8* `suzuki_AP`: for `Q ≥ 1`, `ρ < Q`, `1 ≤ N`, `M ≤ 2N`, `|Y| ≥ 2Q`,
    9
   10      ‖Σ_{t∈(N,M]} e(Y/(ρ + Q t))‖ ≤ C·(N·exp(-c log³N / log²(|Y|/Q)) + N²·Q/|Y|).
   11
   12  This holds for either sign of `Y`.
   13* `cauchy_class_split`: for any finite `𝒰, 𝒱`, any coefficients `a, b` and phases
   14  `F(v,u) = e(φ v u)·G v (u % Q)` with `‖G‖ ≤ 1`,
   15
   16      ‖Σ_u b_u Σ_v a_v F(v,u)‖² ≤ (Σ_u ‖b_u‖²) · Σ_{v,v'} ‖a_v‖‖a_v'‖ · Σ_{ρ<Q} ‖Σ_{u∈I, u%Q=ρ} e(φ v u − φ v' u)‖,
   17
   18  where the extended `u`-range is any `I ⊇ 𝒰`. This is the R0003 "absolute" Cauchy step: the
   19  mask on `u` is dropped by positivity, and the class phase is removed exactly per residue.
   20
   21No axiom, no `sorry`.
   22-/

   29/-- `e(x) = exp(2πi x)`. -/
   30noncomputable def eR (x : ℝ) : ℂ := Complex.exp (2 * Real.pi * Complex.I * (x : ℂ))

   32lemma norm_eR (x : ℝ) : ‖eR x‖ = 1 := by
   33  unfold eR
   34  rw [show 2 * Real.pi * Complex.I * (x : ℂ) = ((2 * Real.pi * x : ℝ) : ℂ) * Complex.I by
   35    push_cast; ring]
   36  exact Complex.norm_exp_ofReal_mul_I _

   38lemma eR_eq (x : ℝ) : eR x = Complex.exp (((2 * Real.pi * x : ℝ) : ℂ) * Complex.I) := by

   51/-- **VK along a residue class.** -/
   52theorem suzuki_AP : ∃ c C : ℝ, 0 < c ∧ 0 ≤ C ∧
   53    ∀ (Q ρ N M : ℕ) (Y : ℝ), 1 ≤ Q → ρ < Q → 1 ≤ N → M ≤ 2 * N → 2 * (Q : ℝ) ≤ |Y| →
   54      ‖∑ t ∈ Ioc N M, eR (Y / ((ρ : ℝ) + Q * t))‖ ≤
   55        C * ((N : ℝ) * Real.exp (-(c * Real.log (N : ℝ) ^ 3 / Real.log (|Y| / Q) ^ 2)) +
   56          (N : ℝ) ^ 2 / (|Y| / Q)) := by
   57  obtain ⟨c, C, hc, hC, hS⟩ := GoldB.shiftedSuzukiCore_unconditional
   58  refine ⟨c, C, hc, hC, fun Q ρ N M Y hQ hρ hN hM hY => ?_⟩
   59  have hQR : (0 : ℝ) < Q := by exact_mod_cast hQ
   60  set ϑ : ℝ := (ρ : ℝ) / Q with hϑ
   61  have hϑ0 : 0 ≤ ϑ := by positivity
   62  have hϑ1 : ϑ ≤ 1 := by
   63    rw [hϑ, div_le_one hQR]; exact_mod_cast hρ.le
   64  have hY2 : 2 ≤ |Y| / Q := by rw [le_div_iff₀ hQR]; linarith
   65  have key := hS (|Y| / Q) hY2 N hN ϑ hϑ0 hϑ1 M hM
   66  -- termwise identity: Y/(ρ + Q t) = (Y/Q)/(t + ϑ)
   67  have hterm : ∀ t ∈ Ioc N M, eR (Y / ((ρ : ℝ) + Q * t)) =
   68      eR ((Y / Q) / ((t : ℝ) + ϑ)) := by
   69    intro t _
   70    congr 1
   71    rw [hϑ]; field_simp; ring
   72  rw [Finset.sum_congr rfl hterm]
   73  rcases le_or_gt 0 Y with hY0 | hY0
   74  · have habs : |Y| = Y := abs_of_nonneg hY0
   75    have : ∑ t ∈ Ioc N M, eR ((Y / Q) / ((t : ℝ) + ϑ)) =
   76        GoldB.Iface.shiftedModeSum 1 (|Y| / Q) ϑ 0 N M := by
   77      unfold GoldB.Iface.shiftedModeSum GoldB.Iface.shiftedExpMode
   78      refine Finset.sum_congr rfl fun t _ => ?_
   79      unfold eR; rw [habs]; push_cast; ring_nf
   80    rw [this]; exact key
   81  · have habs : |Y| = -Y := abs_of_neg hY0
   82    have : ∑ t ∈ Ioc N M, eR ((Y / Q) / ((t : ℝ) + ϑ)) =
   83        (starRingEnd ℂ) (GoldB.Iface.shiftedModeSum 1 (|Y| / Q) ϑ 0 N M) := by
   84      unfold GoldB.Iface.shiftedModeSum
   85      rw [map_sum]
   86      refine Finset.sum_congr rfl fun t _ => ?_
   87      have : GoldB.Iface.shiftedExpMode 1 (|Y| / Q) ϑ 0 t = eR (-((Y / Q) / ((t : ℝ) + ϑ))) := by
   88        unfold GoldB.Iface.shiftedExpMode eR; rw [habs]; push_cast; ring_nf
   89      rw [this, eR_neg, Complex.conj_conj]
   90    rw [this, Complex.norm_conj]; exact key

   92/-! ## Cauchy with the `u`-mask dropped and the class phase split exactly -/
   93
   94theorem cauchy_class_split {Q : ℕ} (hQ : 1 ≤ Q) (𝒰 I 𝒱 : Finset ℕ) (h𝒰 : 𝒰 ⊆ I)
   95    (a b : ℕ → ℂ) (φ : ℕ → ℕ → ℝ) (G : ℕ → ℕ → ℂ) (hG : ∀ v ρ, ‖G v ρ‖ ≤ 1) :
   96    ‖∑ u ∈ 𝒰, b u * ∑ v ∈ 𝒱, a v * (eR (φ v u) * G v (u % Q))‖ ^ 2 ≤
   97      (∑ u ∈ 𝒰, ‖b u‖ ^ 2) *
   98        ∑ v ∈ 𝒱, ∑ v' ∈ 𝒱, ‖a v‖ * ‖a v'‖ *
   99          ∑ ρ ∈ range Q, ‖∑ u ∈ I.filter (fun u => u % Q = ρ), eR (φ v u - φ v' u)‖ := by
  100  set F : ℕ → ℂ := fun u => ∑ v ∈ 𝒱, a v * (eR (φ v u) * G v (u % Q)) with hF
  101  -- Cauchy–Schwarz
  102  have hCS : ‖∑ u ∈ 𝒰, b u * F u‖ ^ 2 ≤ (∑ u ∈ 𝒰, ‖b u‖ ^ 2) * ∑ u ∈ 𝒰, ‖F u‖ ^ 2 := by
  103    have h1 : ‖∑ u ∈ 𝒰, b u * F u‖ ≤ ∑ u ∈ 𝒰, ‖b u‖ * ‖F u‖ :=
  104      (norm_sum_le _ _).trans (le_of_eq (Finset.sum_congr rfl fun u _ => norm_mul _ _))
  105    have h3 : (∑ u ∈ 𝒰, ‖b u‖ * ‖F u‖) ^ 2 ≤
  106        (∑ u ∈ 𝒰, ‖b u‖ ^ 2) * ∑ u ∈ 𝒰, ‖F u‖ ^ 2 :=
  107      Finset.sum_mul_sq_le_sq_mul_sq 𝒰 (fun u => ‖b u‖) (fun u => ‖F u‖)
  108    calc ‖∑ u ∈ 𝒰, b u * F u‖ ^ 2 ≤ (∑ u ∈ 𝒰, ‖b u‖ * ‖F u‖) ^ 2 := by
  109          gcongr
  110      _ ≤ _ := h3
  111  refine hCS.trans (mul_le_mul_of_nonneg_left ?_ (Finset.sum_nonneg fun u _ => by positivity))
  112  -- drop the mask: Σ_{𝒰} ‖F‖² ≤ Σ_I ‖F‖²
  113  have hdrop : ∑ u ∈ 𝒰, ‖F u‖ ^ 2 ≤ ∑ u ∈ I, ‖F u‖ ^ 2 :=
  114    Finset.sum_le_sum_of_subset_of_nonneg h𝒰 fun u _ _ => by positivity
  115  refine hdrop.trans ?_
  116  -- expand ‖F u‖² = Σ_{v,v'} a_v conj(a_v') e(φ v u - φ v' u) G v ρ conj(G v' ρ)
  117  have hexp : ∀ u, ((‖F u‖ ^ 2 : ℝ) : ℂ) = ∑ v ∈ 𝒱, ∑ v' ∈ 𝒱,
  118      a v * (starRingEnd ℂ) (a v') * (eR (φ v u - φ v' u) *
  119        (G v (u % Q) * (starRingEnd ℂ) (G v' (u % Q)))) := by
  120    intro u
  121    rw [Complex.ofReal_pow, ← Complex.mul_conj', hF]
  122    simp only [map_sum, map_mul]
  123    rw [Finset.sum_mul_sum]
  124    refine Finset.sum_congr rfl fun v _ => Finset.sum_congr rfl fun v' _ => ?_
  125    rw [eR_sub]; ring
  126  -- regroup by classes
  127  have hreal : ∑ u ∈ I, ‖F u‖ ^ 2 = ‖∑ u ∈ I, ((‖F u‖ ^ 2 : ℝ) : ℂ)‖ := by
  128    rw [← Complex.ofReal_sum, Complex.norm_real, Real.norm_eq_abs,
  129      abs_of_nonneg (Finset.sum_nonneg fun u _ => by positivity)]
  130  rw [hreal]
  131  simp_rw [hexp]
  132  rw [Finset.sum_comm]
  133  refine (norm_sum_le _ _).trans (Finset.sum_le_sum fun v _ => ?_)
  134  rw [Finset.sum_comm]
  135  refine (norm_sum_le _ _).trans (Finset.sum_le_sum fun v' _ => ?_)
  136  -- split I by residue classes
  137  have hsplit : ∑ u ∈ I, a v * (starRingEnd ℂ) (a v') * (eR (φ v u - φ v' u) *
  138        (G v (u % Q) * (starRingEnd ℂ) (G v' (u % Q)))) =
  139      a v * (starRingEnd ℂ) (a v') * ∑ ρ ∈ range Q, (G v ρ * (starRingEnd ℂ) (G v' ρ)) *
  140        ∑ u ∈ I.filter (fun u => u % Q = ρ), eR (φ v u - φ v' u) := by
  141    rw [← Finset.mul_sum]
  142    congr 1
  143    rw [← Finset.sum_fiberwise_of_maps_to (g := fun u => u % Q) (t := range Q)
  144      (fun u _ => Finset.mem_range.2 (Nat.mod_lt _ (by omega)))]
  145    refine Finset.sum_congr rfl fun ρ _ => ?_
  146    rw [Finset.mul_sum]
  147    refine Finset.sum_congr rfl fun u hu => ?_
  148    rw [(Finset.mem_filter.1 hu).2]; ring
  149  rw [hsplit, norm_mul, norm_mul, Complex.norm_conj]
  150  gcongr
  151  refine (norm_sum_le _ _).trans (Finset.sum_le_sum fun ρ _ => ?_)
  152  rw [norm_mul, norm_mul, Complex.norm_conj]
  153  have h1 := hG v ρ
  154  have h2 := hG v' ρ
  155  calc ‖G v ρ‖ * ‖G v' ρ‖ * ‖∑ u ∈ I.filter (fun u => u % Q = ρ), eR (φ v u - φ v' u)‖
  156      ≤ 1 * 1 * ‖∑ u ∈ I.filter (fun u => u % Q = ρ), eR (φ v u - φ v' u)‖ := by gcongr
  157    _ = _ := by ring

  161lemma classFilter_eq_image (Q ρ U₀ U₁ : ℕ) (hQ : 1 ≤ Q) (hρ : ρ < Q) (hρU : ρ ≤ U₀) :
  162    (Ioc U₀ U₁).filter (fun u => u % Q = ρ) =
  163      (Ioc ((U₀ - ρ) / Q) ((U₁ - ρ) / Q)).image (fun t => ρ + Q * t) := by
  164  have hQ0 : 0 < Q := by omega
  165  ext u
  166  simp only [Finset.mem_filter, Finset.mem_Ioc, Finset.mem_image]
  167  constructor
  168  · rintro ⟨⟨h1, h2⟩, h3⟩
  169    have hdm : Q * (u / Q) + ρ = u := by
  170      have := Nat.div_add_mod u Q; rw [h3] at this; exact this
  171    have hq : u / Q * Q = u - ρ := by rw [Nat.mul_comm]; omega
  172    refine ⟨u / Q, ⟨?_, ?_⟩, by omega⟩
  173    · rw [Nat.div_lt_iff_lt_mul hQ0]; omega
  174    · rw [Nat.le_div_iff_mul_le hQ0]; omega
  175  · rintro ⟨t, ⟨h1, h2⟩, rfl⟩
  176    have ht : t * Q = Q * t := Nat.mul_comm _ _
  177    have h1' : U₀ - ρ < t * Q := (Nat.div_lt_iff_lt_mul hQ0).1 h1
  178    have h2' : t * Q ≤ U₁ - ρ := (Nat.le_div_iff_mul_le hQ0).1 h2
  179    refine ⟨⟨by omega, by omega⟩, ?_⟩
  180    rw [Nat.add_mul_mod_self_left, Nat.mod_eq_of_lt hρ]

  182/-- One class, one off-diagonal pair: VK along the class. -/
  183theorem class_offdiag :
  184    ∃ c C : ℝ, 0 < c ∧ 0 ≤ C ∧
  185      ∀ (Q ρ U₀ U₁ : ℕ) (Y : ℝ), 1 ≤ Q → ρ < Q → 2 * Q ≤ U₀ → U₁ + 2 * Q ≤ 2 * U₀ →
  186        2 * (Q : ℝ) ≤ |Y| →
  187        ‖∑ u ∈ (Ioc U₀ U₁).filter (fun u => u % Q = ρ), eR (Y / u)‖ ≤
  188          C * ((((U₀ - ρ) / Q : ℕ) : ℝ) * Real.exp (-(c * Real.log (((U₀ - ρ) / Q : ℕ) : ℝ) ^ 3 /
  189              Real.log (|Y| / Q) ^ 2)) + (((U₀ - ρ) / Q : ℕ) : ℝ) ^ 2 / (|Y| / Q)) := by
  190  obtain ⟨c, C, hc, hC, hS⟩ := suzuki_AP
  191  refine ⟨c, C, hc, hC, fun Q ρ U₀ U₁ Y hQ hρ hU₀ hU₁ hY => ?_⟩
  192  rw [classFilter_eq_image Q ρ U₀ U₁ hQ hρ (by omega)]
  193  have hinj : Set.InjOn (fun t => ρ + Q * t) (Ioc ((U₀ - ρ) / Q) ((U₁ - ρ) / Q) : Set ℕ) := by
  194    intro x _ y _ h
  195    exact Nat.eq_of_mul_eq_mul_left (show 0 < Q by omega) (Nat.add_left_cancel h)
  196  rw [Finset.sum_image hinj]
  197  have hN : 1 ≤ (U₀ - ρ) / Q := by
  198    rw [Nat.le_div_iff_mul_le (by omega)]; omega
  199  have hM : (U₁ - ρ) / Q ≤ 2 * ((U₀ - ρ) / Q) := by
  200    set N := (U₀ - ρ) / Q
  201    have h1 : Q * N + (U₀ - ρ) % Q = U₀ - ρ := Nat.div_add_mod (U₀ - ρ) Q
  202    have h2 := Nat.mod_lt (U₀ - ρ) (show 0 < Q by omega)
  203    have h3 : (U₁ - ρ) / Q < 2 * N + 1 := by
  204      rw [Nat.div_lt_iff_lt_mul (by omega), show (2 * N + 1) * Q = 2 * (Q * N) + Q by ring]
  205      omega
  206    omega
  207  have := hS Q ρ ((U₀ - ρ) / Q) ((U₁ - ρ) / Q) Y hQ hρ hN hM hY
  208  simp only [Nat.cast_add, Nat.cast_mul]
  209  exact this
