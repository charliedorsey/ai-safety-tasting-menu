    6/-!
    7# Quillon CSC3: the cotangent form of the saw DFT coefficients, and monotonicity
    8
    9* `phi_strictAnti`: `φ(y) = y·cos y/sin y` is strictly decreasing on `(0, π/2]`.
   10* `sawCoeff_cot`: for `0 < k < m`, `a_m(k) = (i/(2m))·cot(πk/m)`, i.e. `(i/2k)·φ(πk/m)/π`.
   11
   12For a fixed frequency `j`, the R-route amplitude `a_{vu}(j)·conj(a_{v'u}(j))` is therefore a product of
   13two positive increasing functions of `u` (when `j ≤ vu/2`, `j ≤ v'u/2`). Its variation telescopes.
   14
   15No axiom, no `sorry`.
   16-/

   27    have := (hasDerivAt_id' y).mul (Real.hasDerivAt_cos y)

   33    rw [show (Real.cos y - y * Real.sin y) * Real.sin y - y * Real.cos y * Real.cos y =
   34      Real.cos y * Real.sin y - y by linear_combination (-y) * this]

   57theorem sawCoeff_cot (m k : ℕ) (hm : 2 ≤ m) (hk : k ∈ Finset.Ioo 0 m) :
   58    GoldC.QuillonSawDFT.sawCoeff m k =
   59      Complex.I * ((Real.cos (π * k / m) / Real.sin (π * k / m) : ℝ) : ℂ) / (2 * m) := by
   60  rw [GoldC.QuillonSawDFT.sawCoeff_closed m k hm hk]

   64  have hθ0 : 0 < θ := by rw [hθ]; have : (0:ℝ) < k := by exact_mod_cast hk1.1
   65                         positivity
   66  have hθπ : θ < π := by
   67    rw [hθ, div_lt_iff₀ hmR]
   68    have : (k : ℝ) < m := by exact_mod_cast hk1.2
   69    nlinarith [Real.pi_pos]
   70  have hsθ : Real.sin θ ≠ 0 := (Real.sin_pos_of_pos_of_lt_pi hθ0 hθπ).ne'
   71  -- ζ^{-k} = exp(-2iθ) = (cos θ − i sin θ)²
   72  have hz : GoldC.QuillonSawDFT.zeta m ^ (-(k : ℤ)) =
   73      (Complex.cos (θ : ℂ) - Complex.sin (θ : ℂ) * Complex.I) ^ 2 := by
   74    unfold GoldC.QuillonSawDFT.zeta
   75    rw [← Complex.exp_int_mul]
   76    have e1 : ((-(k : ℤ) : ℤ) : ℂ) * (2 * (π : ℂ) * Complex.I / (m : ℂ)) =
   77        2 * ((-(θ : ℂ)) * Complex.I) := by
   78      rw [hθ]; push_cast; field_simp
   79    rw [e1, show (2 : ℂ) * (-(θ : ℂ) * Complex.I) = (-(θ : ℂ) * Complex.I) + (-(θ : ℂ) * Complex.I) by ring,
   80      Complex.exp_add, Complex.exp_mul_I, Complex.cos_neg, Complex.sin_neg]
   81    ring
   82  rw [hz]
   83  have hmC : (m : ℂ) ≠ 0 := by exact_mod_cast (show m ≠ 0 by omega)
   84  have hsC : Complex.sin (θ : ℂ) ≠ 0 := by
   85    rw [← Complex.ofReal_sin]; exact_mod_cast hsθ

   96    (eq_inv_of_mul_eq_one_left hprod).symm
   97  have hR : (((Real.cos (π * k / m) / Real.sin (π * k / m) : ℝ)) : ℂ) = c / sn := by

  146/-- **Amplitude variation along an increasing progression.** If `x_t` is increasing with
  147`2j ≤ v·x_t` and `2j ≤ v'·x_t`, then the real amplitude `amp j (v x_t) · amp j (v' x_t)` is
  148increasing, nonnegative and `≤ 1`. Its Abel factor is `≤ 2`. -/
  149theorem amp_abel_factor (j v v' : ℕ) (hj : 1 ≤ j) (x : ℕ → ℕ) (N M : ℕ) (hNM : N + 1 ≤ M)
  150    (hx : ∀ t, N + 1 ≤ t → t < M → x t ≤ x (t + 1))
  151    (hpos : ∀ t, N + 1 ≤ t → t ≤ M → 2 * j ≤ v * x t ∧ 2 * j ≤ v' * x t) :
  152    ‖((amp j (v * x M) * amp j (v' * x M) : ℝ) : ℂ)‖ +
  153      ∑ t ∈ Finset.Ico (N + 1) M, ‖((amp j (v * x (t + 1)) * amp j (v' * x (t + 1)) : ℝ) : ℂ) -
  154        ((amp j (v * x t) * amp j (v' * x t) : ℝ) : ℂ)‖ ≤ 2 := by
  155  set f : ℕ → ℝ := fun t => amp j (v * x t) * amp j (v' * x t) with hf
  156  have hbounds : ∀ t, N + 1 ≤ t → t ≤ M → 0 ≤ f t ∧ f t ≤ 1 := by
  157    intro t h1 h2
  158    obtain ⟨ha, hb⟩ := hpos t h1 h2
  159    have hin : ∀ m : ℕ, 2 * j ≤ m → 0 < π * j / m ∧ π * j / m ≤ π / 2 := by
  160      intro m hm
  161      have hmR : (0 : ℝ) < m := by exact_mod_cast (show 0 < m by omega)
  162      have hjR : (0 : ℝ) < j := by exact_mod_cast hj
  163      refine ⟨by positivity, ?_⟩
  164      rw [div_le_div_iff₀ hmR (by norm_num)]
  165      have : (2 * j : ℝ) ≤ m := by exact_mod_cast hm
  166      nlinarith [Real.pi_pos]
  167    have e1 := hin _ ha; have e2 := hin _ hb
  168    have p1 := phi_nonneg e1.1 e1.2; have p2 := phi_nonneg e2.1 e2.2
  169    have q1 := phi_le_one e1.1 e1.2; have q2 := phi_le_one e2.1 e2.2
  170    simp only [hf, amp]
  171    exact ⟨mul_nonneg p1 p2, by nlinarith⟩
  172  have hmono : ∀ t, N + 1 ≤ t → t < M → f t ≤ f (t + 1) := by
  173    intro t h1 h2
  174    obtain ⟨ha, hb⟩ := hpos t h1 h2.le
  175    obtain ⟨ha', hb'⟩ := hpos (t + 1) (by omega) (by omega)
  176    have hxt := hx t h1 h2
  177    have m1 := amp_mono hj ha (Nat.mul_le_mul_left v hxt)
  178    have m2 := amp_mono hj hb (Nat.mul_le_mul_left v' hxt)
  179    have n1 := (hbounds t h1 h2.le)
  180    -- both factors nonnegative
  181    have hin : ∀ m : ℕ, 2 * j ≤ m → 0 ≤ amp j m := by
  182      intro m hm
  183      have hmR : (0 : ℝ) < m := by exact_mod_cast (show 0 < m by omega)
  184      have hjR : (0 : ℝ) < j := by exact_mod_cast hj
  185      refine phi_nonneg (by positivity) ?_
  186      rw [div_le_div_iff₀ hmR (by norm_num)]
  187      have : (2 * j : ℝ) ≤ m := by exact_mod_cast hm
  188      nlinarith [Real.pi_pos]
  189    simp only [hf]
  190    exact mul_le_mul m1 m2 (hin _ hb) (le_trans (hin _ ha) m1)
  191  have hvar := GoldC.QuillonAbel.variation_monotone f N M hNM hmono
  192  have hM := hbounds M (by omega) le_rfl
  193  have hN := hbounds (N + 1) le_rfl hNM
  194  rw [Complex.norm_real, Real.norm_eq_abs, abs_of_nonneg hM.1]
  195  have : ∑ t ∈ Finset.Ico (N + 1) M, ‖((f (t + 1) : ℝ) : ℂ) - ((f t : ℝ) : ℂ)‖ = f M - f (N + 1) :=
  196    hvar
  197  simp only [hf] at this ⊢
  198  rw [this]
  199  linarith [hM.2, hN.1]

  301The coefficients depend on `m` only through `ampC(k,m)`, which is monotone and in `[0,1]`. -/

  304      ∑ k ∈ Finset.Icc 1 J, (Complex.I * ((ampC k m : ℝ) : ℂ) / (2 * π * k)) *
  305        (GoldC.QuillonSawDFT.zeta m ^ ((k : ℤ) * c) - GoldC.QuillonSawDFT.zeta m ^ (-((k : ℤ) * c))) := by
  306  rw [GoldC.QuillonSawDFT.prof_expansion m c hm hc]

  308  set ζ := GoldC.QuillonSawDFT.zeta m
  309  have hζ := GoldC.QuillonSawDFT.zeta_prim m (by omega)
  310  have hζ0 := GoldC.QuillonSawDFT.zeta_ne_zero m
  311  -- a 0 = 0
  312  have ha0 : a 0 = 0 := by
