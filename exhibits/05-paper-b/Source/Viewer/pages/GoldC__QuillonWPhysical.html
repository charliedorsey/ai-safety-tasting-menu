    4# Quillon CSC3 F8.7: the **physical W′ theorem** (integrated, end-to-end for the W′ cell)
    5
    6For every `d` (decoration), the W′ part of the actual CORE pair sum
    7
    8    WPart_d = Σ_{B,C ⊆ T_d} 1[B,C ≠ ∅, B∩C = ∅, Mk₀(B,C), B ∩ J_W(C) ≠ ∅] · ω(B)ω(C) · foldRow(D,H,∏B,∏C)
    9
   10satisfies `|WPart_d| ≤ Λ^{−80}` for `Λ = log h ≥ Λ₀(q, δ)`, uniformly in `a`, `d`, `Jlow`, `Jhigh`.
   11* `T_d = ordinaryFiniteSupport q H d (H²+1)` and `H = h/∏d`.
   12* `Mk₀` is the exact source pair mask from `pairTerm`: `Z < rs`, `4rs ≤ H²`,
   13  `Jlow ≤ |B|+|C| < Jhigh`, `¬longEither δ`, `centralGate h`.
   14* `J_W(C) = (T∖C) ∩ [X, ⌊s^{u₀}⌋]`, with `X = ⌈Λ^{200}⌉`.
   15
   16All suppliers meet in this one theorem:
   17* the source copies;
   18* `pairSum`-shape double sum, `double_sum_fixC`;
   19* the bridge `foldRow = rowAbs`;
   20* `W_cell` (R0010: `W_context`, `W_bilinear_full`, `W_cell_coupled`, Ramaré);
   21* the explicit majorant `WTerm_le`;
   22* the eventual analysis `W_term_ev`;
   23* the outer mass `Σ_C ω(C) ≤ e^{10}5^{10}Λ^{10}`.
   24
   25No axiom, no `sorry`.
   26-/

   36/-- The active prime universe of decoration `d`. -/
   37noncomputable def Tsrc (q h : ℕ) (d : Finset ℕ) : Finset ℕ :=
   38  ordinaryFiniteSupport q (h / primeSetProduct d) d ((h / primeSetProduct d) ^ 2 + 1)
   39
   40/-- The exact CORE pair mask (the `pairTerm` condition after `B ≠ ∅ ∧ C ≠ ∅ ∧ Disjoint B C`). -/
   41def Mk0 (Jlow Jhigh h H : ℕ) (δ : ℝ) (B C : Finset ℕ) : Prop :=
   42  originalZ h < primeSetProduct B * primeSetProduct C ∧
   43    4 * (primeSetProduct B * primeSetProduct C) ≤ H ^ 2 ∧
   44    Jlow ≤ B.card + C.card ∧ B.card + C.card < Jhigh ∧
   45    ¬ longEither δ (primeSetProduct B) (primeSetProduct C) ∧
   46    centralGate h H (primeSetProduct B) (primeSetProduct C)
   47
   48/-- The W′ cell: `B` has a prime in `[X, ⌊s^{u₀}⌋]`. -/
   49def WMask (h : ℕ) (δ : ℝ) (T B C : Finset ℕ) : Prop :=
   50  (B ∩ (T \ C).filter (fun p => Xp h ≤ p ∧ p ≤ Yp δ (∏ p ∈ C, p))).Nonempty
   51
   52/-- The W′ part of the CORE pair sum at decoration `d`. -/
   53noncomputable def WPart (Jlow Jhigh q a h : ℕ) (δ : ℝ) (d : Finset ℕ) : ℝ :=
   54  ∑ B ∈ (Tsrc q h d).powerset, ∑ C ∈ (Tsrc q h d).powerset,
   55    if B ≠ ∅ ∧ C ≠ ∅ ∧ Disjoint B C ∧
   56        (Mk0 Jlow Jhigh h (h / primeSetProduct d) δ B C ∧ WMask h δ (Tsrc q h d) B C) then
   57      omegaWeight B * omegaWeight C *
   58        foldRow q a (primeSetProduct d) (h / primeSetProduct d) (primeSetProduct B) (primeSetProduct C)
   59    else 0

   61lemma Tsrc_facts (q h : ℕ) (d : Finset ℕ) :
   62    ∀ p ∈ Tsrc q h d, p.Prime ∧ 5 ≤ p ∧ ¬ p ∣ q ∧ p.Coprime (h / primeSetProduct d) ∧
   63      p ≤ (h / primeSetProduct d) ^ 2 := by
   64  intro p hp
   65  simp only [Tsrc, ordinaryFiniteSupport, mem_filter, mem_range] at hp
   66  obtain ⟨hlt, hact, hH, -⟩ := hp
   67  obtain ⟨hpp, h5, hq⟩ := hact
   68  exact ⟨hpp, h5, hq, (Nat.Prime.coprime_iff_not_dvd hpp).2 hH, by omega⟩

   70/-- The outer mass `Σ_{C ⊆ T} ω(C) ≤ e^{10}5^{10}Λ^{10}`. -/
   71lemma outer_mass (q h : ℕ) (d : Finset ℕ) (hh : 1 ≤ Real.log h) :
   72    ∑ C ∈ (Tsrc q h d).powerset, ωR C ≤ Real.exp 10 * (2 * 2 + 1) ^ 10 * (Real.log h) ^ (10 : ℝ) := by
   73  rw [sum_ωR_powerset]
   74  have hT := Tsrc_facts q h d
   75  have hh0 : (0 : ℝ) < h := by
   76    by_contra hc; push_neg at hc
   77    have : (h : ℝ) = 0 := le_antisymm hc (by positivity)
   78    rw [this, Real.log_zero] at hh; linarith
   79  refine GoldC.QuillonLam.prodU_le_pow _ (fun p hp => ⟨(hT p hp).1, (hT p hp).2.1⟩) (h ^ 2)
   80    (fun p hp => le_trans (hT p hp).2.2.2.2 (Nat.pow_le_pow_left (Nat.div_le_self _ _) 2)) 2 _
   81    (by norm_num) hh ?_
   82  push_cast; rw [Real.log_pow]; push_cast; linarith

   84/-- **Physical W′ theorem.** -/
   85theorem W_physical (q : ℕ) (hq : 0 < q) (δ : ℝ) (hδ0 : 0 < δ) (hδ1 : δ < 1 / 3) :
   86    ∃ Λ₀ : ℝ, 2 ≤ Λ₀ ∧ ∀ (Jlow Jhigh a h : ℕ) (d : Finset ℕ), Λ₀ ≤ Real.log h → 2 ≤ h →
   87      |WPart Jlow Jhigh q a h δ d| ≤ (Real.log h) ^ (-(80 : ℝ)) := by
   88  obtain ⟨Λ₁, hΛ₁, hW⟩ := GoldC.QuillonWFinal.W_final q hq δ hδ0 hδ1
   89  set K : ℝ := Real.exp 10 * (2 * 2 + 1) ^ 10 * 4 with hK
   90  obtain ⟨Λ₂, hΛ₂, hKΛ⟩ := GoldC.QuillonLam.const_le_rpow K 1 one_pos
   91  refine ⟨max Λ₁ Λ₂, le_trans hΛ₁ (le_max_left _ _), ?_⟩
   92  intro Jlow Jhigh a h d hΛ hh
   93  have hΛa : Λ₁ ≤ Real.log h := le_trans (le_max_left _ _) hΛ
   94  have hΛb : Λ₂ ≤ Real.log h := le_trans (le_max_right _ _) hΛ
   95  set Λ := Real.log h with hΛdef
   96  have hΛ2 : 2 ≤ Λ := le_trans hΛ₁ hΛa
   97  have hΛ0 : 0 < Λ := by linarith
   98  set H := h / primeSetProduct d with hH
   99  set T := Tsrc q h d with hT
  100  have hTf := Tsrc_facts q h d
  101  have hHh : H ≤ h := Nat.div_le_self _ _
  102  unfold WPart
  103  rw [GoldC.QuillonCells.double_sum_fixC]
  104  -- per-`C` bound
  105  have hinner : ∀ C ∈ T.powerset.filter (fun C => C ≠ ∅),
  106      |∑ B ∈ (T \ C).powerset.filter (fun B => B ≠ ∅),
  107        (if Mk0 Jlow Jhigh h H δ B C ∧ WMask h δ T B C then
  108          omegaWeight B * omegaWeight C *
  109            foldRow q a (primeSetProduct d) H (primeSetProduct B) (primeSetProduct C) else 0)| ≤
  110        ωR C * (4 * Λ ^ (-(95 : ℝ))) := by
  111    intro C hC
  112    have hCT : C ⊆ T := mem_powerset.1 (mem_filter.1 hC).1
  113    have hω0 : 0 ≤ ωR C := ωR_nonneg C fun p hp => (hTf p (hCT hp)).2.1
  114    have key := hW a h H (primeSetProduct d) T C
  115      (fun k => Jlow ≤ k + C.card ∧ k + C.card < Jhigh) hΛa hh hHh hTf hCT
  116    have e : ∑ B ∈ (T \ C).powerset.filter (fun B => B ≠ ∅),
  117        (if Mk0 Jlow Jhigh h H δ B C ∧ WMask h δ T B C then
  118          omegaWeight B * omegaWeight C *
  119            foldRow q a (primeSetProduct d) H (primeSetProduct B) (primeSetProduct C) else 0) =
  120        ωR C * ∑ B ∈ (T \ C).powerset.filter (fun B => B ≠ ∅),
  121          (if ((Jlow ≤ B.card + C.card ∧ B.card + C.card < Jhigh) ∧
  122              Geo0 h H δ (∏ p ∈ B, p) (∏ p ∈ C, p)) ∧
  123              (B ∩ (T \ C).filter (fun p => Xp h ≤ p ∧ p ≤ Yp δ (∏ p ∈ C, p))).Nonempty
  124            then ωR B * foldRow q a (primeSetProduct d) H (∏ p ∈ B, p) (∏ p ∈ C, p) else 0) := by
  125      rw [mul_sum]
  126      refine sum_congr rfl fun B _ => ?_
  127      have hiff : (Mk0 Jlow Jhigh h H δ B C ∧ WMask h δ T B C) ↔
  128          (((Jlow ≤ B.card + C.card ∧ B.card + C.card < Jhigh) ∧
  129              Geo0 h H δ (∏ p ∈ B, p) (∏ p ∈ C, p)) ∧
  130              (B ∩ (T \ C).filter (fun p => Xp h ≤ p ∧ p ≤ Yp δ (∏ p ∈ C, p))).Nonempty) := by
  131        constructor
  132        · rintro ⟨⟨h1, h2, h3, h4, h5, h6⟩, hw⟩; exact ⟨⟨⟨h3, h4⟩, h1, h2, h5, h6⟩, hw⟩
  133        · rintro ⟨⟨⟨h3, h4⟩, h1, h2, h5, h6⟩, hw⟩; exact ⟨⟨h1, h2, h3, h4, h5, h6⟩, hw⟩
  134      split_ifs with h1 h2 h2
  135      · show ωR B * ωR C * foldRow q a (primeSetProduct d) H (∏ p ∈ B, p) (∏ p ∈ C, p) =
  136          ωR C * (ωR B * foldRow q a (primeSetProduct d) H (∏ p ∈ B, p) (∏ p ∈ C, p))
  137        ring
  138      · exact absurd (hiff.1 h1) h2
  139      · exact absurd (hiff.2 h2) h1
  140      · rw [mul_zero]
  141    rw [e, abs_mul, abs_of_nonneg hω0]
  142    exact mul_le_mul_of_nonneg_left key hω0
  143  calc _ ≤ ∑ C ∈ T.powerset.filter (fun C => C ≠ ∅),
  144        |∑ B ∈ (T \ C).powerset.filter (fun B => B ≠ ∅),
  145          (if Mk0 Jlow Jhigh h H δ B C ∧ WMask h δ T B C then
  146            omegaWeight B * omegaWeight C *
  147              foldRow q a (primeSetProduct d) H (primeSetProduct B) (primeSetProduct C) else 0)| :=
  148        abs_sum_le_sum_abs _ _
  149    _ ≤ ∑ C ∈ T.powerset.filter (fun C => C ≠ ∅), ωR C * (4 * Λ ^ (-(95 : ℝ))) := sum_le_sum hinner
  150    _ ≤ ∑ C ∈ T.powerset, ωR C * (4 * Λ ^ (-(95 : ℝ))) := by
  151        apply sum_le_sum_of_subset_of_nonneg (filter_subset _ _)
  152        intro C hC _
  153        have hCT : C ⊆ T := mem_powerset.1 hC
  154        have := ωR_nonneg C fun p hp => (hTf p (hCT hp)).2.1
  155        positivity
  156    _ = (∑ C ∈ T.powerset, ωR C) * (4 * Λ ^ (-(95 : ℝ))) := by rw [sum_mul]
  157    _ ≤ (Real.exp 10 * (2 * 2 + 1) ^ 10 * Λ ^ (10 : ℝ)) * (4 * Λ ^ (-(95 : ℝ))) :=
  158        mul_le_mul_of_nonneg_right (outer_mass q h d (by linarith)) (by positivity)
  159    _ = K * Λ ^ (-(85 : ℝ)) := by
  160        rw [hK, show Λ ^ (-(85 : ℝ)) = Λ ^ (10 : ℝ) * Λ ^ (-(95 : ℝ)) by
  161          rw [← Real.rpow_add hΛ0]; norm_num]
  162        ring
  163    _ ≤ Λ ^ (1 : ℝ) * Λ ^ (-(85 : ℝ)) := mul_le_mul_of_nonneg_right (hKΛ Λ hΛb) (by positivity)
  164    _ = Λ ^ (-(84 : ℝ)) := by rw [← Real.rpow_add hΛ0]; norm_num
  165    _ ≤ Λ ^ (-(80 : ℝ)) := Real.rpow_le_rpow_of_exponent_le (by linarith) (by norm_num)
