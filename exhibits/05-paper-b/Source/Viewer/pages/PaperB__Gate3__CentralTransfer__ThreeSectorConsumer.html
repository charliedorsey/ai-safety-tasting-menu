   21/-- The original source parents selected by the middle cardinality window.
   22The filters are applied to actual quarterParents, before oriented summation. -/
   23noncomputable def regionPairSumMiddle
   24    (Jlow Jhigh q h : ℕ)
   25    (gate : ℕ → ℕ → ℕ → Prop)
   26    (d : Finset ℕ) (G : ℕ → ℕ → ℕ → ℝ) : ℝ :=
   27  ∑ e ∈ (quarterParents q h (h / primeSetProduct d) d
   28    ((h / primeSetProduct d) ^ 2 + 1)).filter
   29      (fun e => Jlow ≤ e.card ∧ e.card < Jhigh),
   30    ∑ B ∈ ((e.powerset.erase e).erase ∅),
   31      if gate (h / primeSetProduct d)
   32          (primeSetProduct B) (primeSetProduct (e \ B)) then
   33        omegaWeight e * G (h / primeSetProduct d)
   34          (primeSetProduct B) (primeSetProduct (e \ B))
   35      else 0

   37/-- Decorated, source-exact middle-cardinality sector. -/
   38noncomputable def regionSumMiddle
   39    (Jlow Jhigh q h : ℕ)
   40    (gate : ℕ → ℕ → ℕ → Prop)
   41    (F : ℕ → ℕ → ℕ → ℕ → ℝ) : ℝ :=
   42  ∑ d ∈ regionDecorations q h,
   43    decorationWeight h d *
   44      regionPairSumMiddle Jlow Jhigh q h gate d (F (primeSetProduct d))

   46/-- Source-exact low-to-middle identity; no source support is enlarged. -/
   47theorem regionPairSumCard_low_middle
   48    (Jlow Jhigh q h : ℕ) (hord : Jlow ≤ Jhigh)
   49    (gate : ℕ → ℕ → ℕ → Prop)
   50    (d : Finset ℕ) (G : ℕ → ℕ → ℕ → ℝ) :
   51    regionPairSumCard Jhigh false q h gate d G =
   52      regionPairSumCard Jlow false q h gate d G +
   53      regionPairSumMiddle Jlow Jhigh q h gate d G := by
   54  classical
   55  unfold regionPairSumCard regionPairSumMiddle
   56  simp only [Bool.false_eq_true, ↓reduceIte, Finset.sum_filter]
   57  rw [← Finset.sum_add_distrib]
   58  apply Finset.sum_congr rfl
   59  intro e he
   60  by_cases hlo : e.card < Jlow
   61  · have hhi : e.card < Jhigh := lt_of_lt_of_le hlo hord
   62    simp [hlo, hhi, Nat.not_le_of_gt hlo]
   63  · by_cases hhi : e.card < Jhigh
   64    · have hge : Jlow ≤ e.card := Nat.le_of_not_gt hlo
   65      simp [hlo, hhi, hge]
   66    · simp [hlo, hhi]

   72    regionSumCard Jhigh false q h gate F =

   74      regionSumMiddle Jlow Jhigh q h gate F := by

   79  rw [← mul_add]

  106/-- Supplier-quantifier contract: fixed positive squarefree q, every
  107admissible anchor a, original parity condition, and eventually in h. -/
  108def SourceUniformSublog (S : ℕ → ℕ → ℕ → ℝ) : Prop :=
  109  ∀ q : ℕ, 0 < q → Squarefree q →
  110    ∀ eps : ℝ, 0 < eps →
  111      ∃ H0 : ℕ, ∀ a h : ℕ,
  112        q.Coprime a → 0 < h → (¬ 2 ∣ q → 2 ∣ h) →
  113        H0 ≤ h →
  114        |S q a h| ≤ eps * Real.log (h : ℝ)

  126/-- Conditional consumer. LOW and HIGH may use positive estimates;
  127MIDDLE must be a source-weighted signed o(log h) result. The conclusion
  128is for the folded physical GoldC source, NOT YET the literal R0056 terminal.
  129-/
  130theorem foldedCentral_sublog_of_three
  131    (Jlow Jhigh : ℕ → ℕ) (hord : ∀ h, Jlow h ≤ Jhigh h)
  132    (hLow : SourceUniformSublog (lowSector Jlow))
  133    (hMiddle : SourceUniformSublog (middleSector Jlow Jhigh))
  134    (hHigh : SourceUniformSublog (highSector Jhigh)) :
  135    SourceUniformSublog
  136      (fun q a h => regionSum q h (centralGate h) (foldRow q a)) := by
  137  intro q hq hsq eps heps
  138  let delta : ℝ := eps / 3
  139  have hdelta : 0 < delta := by dsimp [delta]; positivity
  140  obtain ⟨Hlow, hlow⟩ := hLow q hq hsq delta hdelta
  141  obtain ⟨Hmid, hmid⟩ := hMiddle q hq hsq delta hdelta
  142  obtain ⟨Hhigh, hhigh⟩ := hHigh q hq hsq delta hdelta
  143  refine ⟨max (max Hlow Hmid) Hhigh, ?_⟩
  144  intro a h hqa hh hpar hH
  145  have hHlow : Hlow ≤ h :=
  146    le_trans (le_max_left Hlow Hmid)
  147      (le_trans (le_max_left (max Hlow Hmid) Hhigh) hH)
  148  have hHmid : Hmid ≤ h :=
  149    le_trans (le_max_right Hlow Hmid)
  150      (le_trans (le_max_left (max Hlow Hmid) Hhigh) hH)
  151  have hHhigh : Hhigh ≤ h :=
  152    le_trans (le_max_right (max Hlow Hmid) Hhigh) hH
  153  have hl := hlow a h hqa hh hpar hHlow
  154  have hm := hmid a h hqa hh hpar hHmid
  155  have hu := hhigh a h hqa hh hpar hHhigh
  156  change
  157    |regionSumCard (Jlow h) false q h (centralGate h) (foldRow q a)| ≤
  158      delta * Real.log (h : ℝ) at hl
  159  change
  160    |regionSumMiddle (Jlow h) (Jhigh h) q h (centralGate h)
  161      (foldRow q a)| ≤ delta * Real.log (h : ℝ) at hm
  162  change
  163    |regionSumCard (Jhigh h) true q h (centralGate h) (foldRow q a)| ≤
  164      delta * Real.log (h : ℝ) at hu
  165  change |regionSum q h (centralGate h) (foldRow q a)| ≤
  166    eps * Real.log (h : ℝ)
  167  rw [foldedCentral_three_sectors (Jlow h) (Jhigh h) q a h (hord h)]
  168  calc
  169    _ ≤
  170      |regionSumCard (Jlow h) false q h (centralGate h) (foldRow q a)| +
  171      |regionSumMiddle (Jlow h) (Jhigh h) q h
  172         (centralGate h) (foldRow q a)| +
  173      |regionSumCard (Jhigh h) true q h (centralGate h) (foldRow q a)| := by
  174        have h1 := abs_add_le
  175          (regionSumCard (Jlow h) false q h (centralGate h) (foldRow q a) +
  176           regionSumMiddle (Jlow h) (Jhigh h) q h
  177             (centralGate h) (foldRow q a))
  178          (regionSumCard (Jhigh h) true q h
  179            (centralGate h) (foldRow q a))
  180        have h2 := abs_add_le
  181          (regionSumCard (Jlow h) false q h (centralGate h) (foldRow q a))
  182          (regionSumMiddle (Jlow h) (Jhigh h) q h
  183            (centralGate h) (foldRow q a))
  184        linarith
  185    _ ≤ delta * Real.log (h : ℝ) +
  186        delta * Real.log (h : ℝ) +
  187        delta * Real.log (h : ℝ) := by
  188          gcongr
  189    _ = eps * Real.log (h : ℝ) := by dsimp [delta]; ring
