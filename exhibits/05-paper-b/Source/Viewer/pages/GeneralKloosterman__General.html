    3/-!
    4# General modulus: CRT assembly and the Weil–Estermann bound
    5
    6* `kgcd_crt` — the gcd loss is multiplicative across the toolkit's Bézout/CRT splitting
    7  (the local coefficients are unit multiples of the reductions).
    8* **`norm_kSum_le`** — for every modulus `N ≥ 1` and all `a, b ∈ ZMod N`:
    9
   10    `‖S(a,b;N)‖ ≤ τ(N) · √(gcd(a,b,N) · N)`,
   11
   12  `τ(N) = #N.divisors`. Proof: induction over coprime prime-power factorizations
   13  (`Nat.recOnPosPrimePosCoprime`), using the local theorem `norm_kSum_primePow_le` at each prime
   14  power and the toolkit's exact twisted-multiplicativity identity `norm_kloostermanUnitSum_crt`.
   15* `norm_kloostermanUnitSum_le` — the same in the toolkit's notation.
   16* `norm_kSum_le_of_isUnit_left/right` — a unit coefficient: `‖S‖ ≤ τ(N) √N`.
   17-/

   24def vg (m : ℕ) (z : ZMod m) : ℕ := Nat.gcd z.val m

   26theorem gcd_gcd_eq (x y m : ℕ) :
   27    Nat.gcd (Nat.gcd x y) m = Nat.gcd (Nat.gcd x m) (Nat.gcd y m) := by
   28  apply Nat.dvd_antisymm
   29  · exact Nat.dvd_gcd
   30      (Nat.dvd_gcd ((Nat.gcd_dvd_left _ _).trans (Nat.gcd_dvd_left _ _)) (Nat.gcd_dvd_right _ _))
   31      (Nat.dvd_gcd ((Nat.gcd_dvd_left _ _).trans (Nat.gcd_dvd_right _ _)) (Nat.gcd_dvd_right _ _))
   32  · exact Nat.dvd_gcd
   33      (Nat.dvd_gcd ((Nat.gcd_dvd_left _ _).trans (Nat.gcd_dvd_left _ _))
   34        ((Nat.gcd_dvd_right _ _).trans (Nat.gcd_dvd_left _ _)))
   35      ((Nat.gcd_dvd_left _ _).trans (Nat.gcd_dvd_right _ _))

   60/-- The gcd loss across the CRT splitting. -/
   61theorem kgcd_crt {m n : ℕ} [NeZero m] [NeZero n] [NeZero (m * n)] (hcop : m.Coprime n)
   62    (a b : ZMod (m * n)) :
   63    kgcd (m * n) a b =
   64      kgcd m (((crtLeftCoeff m n : ℤ) : ZMod m) * (ZMod.chineseRemainder hcop a).1)
   65          (((crtLeftCoeff m n : ℤ) : ZMod m) * (ZMod.chineseRemainder hcop b).1) *
   66        kgcd n (((crtRightCoeff m n : ℤ) : ZMod n) * (ZMod.chineseRemainder hcop a).2)
   67          (((crtRightCoeff m n : ℤ) : ZMod n) * (ZMod.chineseRemainder hcop b).2) := by
   68  rw [kgcd, Nat.Coprime.gcd_mul _ hcop, kgcd_eq_vg, kgcd_eq_vg,
   69    vg_unit_mul _ _ (crtLeftCoeff_isUnit hcop), vg_unit_mul _ _ (crtLeftCoeff_isUnit hcop),
   70    vg_unit_mul _ _ (crtRightCoeff_isUnit hcop), vg_unit_mul _ _ (crtRightCoeff_isUnit hcop),
   71    crt_fst, crt_fst, crt_snd, crt_snd, vg_castHom, vg_castHom, vg_castHom, vg_castHom,
   72    gcd_gcd_eq a.val b.val m, gcd_gcd_eq a.val b.val n]

   77theorem sqrt_mul_assemble (t1 t2 g1 g2 m n : ℕ) :
   78    (t1 : ℝ) * Real.sqrt (g1 * m) * ((t2 : ℝ) * Real.sqrt (g2 * n)) =
   79      ((t1 * t2 : ℕ) : ℝ) * Real.sqrt (((g1 * g2 : ℕ) : ℝ) * ((m * n : ℕ) : ℝ)) := by
   80  rw [Nat.cast_mul, Nat.cast_mul, Nat.cast_mul]
   81  have h : Real.sqrt ((g1 : ℝ) * g2 * (m * n)) = Real.sqrt (g1 * m) * Real.sqrt (g2 * n) := by
   82    rw [← Real.sqrt_mul (by positivity)]; congr 1; ring
   83  rw [h]; ring

   85theorem norm_kSum_le_aux : ∀ N : ℕ, ∀ (_ : NeZero N) (a b : ZMod N),
   86    ‖kSum N a b‖ ≤ (#N.divisors : ℝ) * Real.sqrt (kgcd N a b * N) := by
   87  refine Nat.recOnPosPrimePosCoprime ?_ ?_ ?_ ?_
   88  · intro p k hp hk _ a b
   89    have : Fact p.Prime := ⟨hp⟩
   90    have := norm_kSum_primePow_le k (p ^ k) rfl hk a b
   91    rw [card_divisors_prime_pow hp]
   92    push_cast at this ⊢
   93    exact this
   94  · intro h; exact absurd rfl h.out
   95  · intro _ a b
   96    refine (norm_kSum_le_card a b).trans ?_
   97    simp [kgcd]
   98  · intro m n hm hn hcop ihm ihn _ a b
   99    have : NeZero m := ⟨by omega⟩
  100    have : NeZero n := ⟨by omega⟩
  101    have h1 := ihm inferInstance (((crtLeftCoeff m n : ℤ) : ZMod m) * (ZMod.chineseRemainder hcop a).1)
  102      (((crtLeftCoeff m n : ℤ) : ZMod m) * (ZMod.chineseRemainder hcop b).1)
  103    have h2 := ihn inferInstance (((crtRightCoeff m n : ℤ) : ZMod n) * (ZMod.chineseRemainder hcop a).2)
  104      (((crtRightCoeff m n : ℤ) : ZMod n) * (ZMod.chineseRemainder hcop b).2)
  105    rw [kSum_eq_kloostermanUnitSum] at h1 h2
  106    rw [kSum_eq_kloostermanUnitSum, norm_kloostermanUnitSum_crt hcop, kgcd_crt hcop,
  107      Nat.Coprime.card_divisors_mul hcop]
  108    exact (mul_le_mul h1 h2 (norm_nonneg _) (by positivity)).trans
  109      (le_of_eq (sqrt_mul_assemble _ _ _ _ _ _))

  111/-- **General-modulus Weil–Estermann bound.** For every `N ≥ 1` and all `a, b ∈ ZMod N`,
  112`‖S(a,b;N)‖ ≤ τ(N) · √(gcd(a,b,N) · N)`. -/
  113theorem norm_kSum_le (N : ℕ) [NeZero N] (a b : ZMod N) :
  114    ‖kSum N a b‖ ≤ (#N.divisors : ℝ) * Real.sqrt (kgcd N a b * N) :=
  115  norm_kSum_le_aux N inferInstance a b
  116
  117/-- The same bound in the toolkit's notation `kloostermanUnitSum`. -/
  118theorem norm_kloostermanUnitSum_le (N : ℕ) [NeZero N] (a b : ZMod N) :
  119    ‖kloostermanUnitSum N a b‖ ≤
  120      (#N.divisors : ℝ) * Real.sqrt (Nat.gcd (Nat.gcd a.val b.val) N * N) :=
  121  norm_kSum_le N a b

  123theorem kgcd_eq_one_of_isUnit_left {N : ℕ} [NeZero N] (a b : ZMod N) (ha : IsUnit a) :
  124    kgcd N a b = 1 := by
  125  obtain ⟨U, rfl⟩ := ha
  126  have hc := ZMod.val_coe_unit_coprime U
  127  unfold kgcd
  128  exact Nat.eq_one_of_dvd_one (hc ▸ Nat.dvd_gcd ((Nat.gcd_dvd_left _ _).trans
  129    (Nat.gcd_dvd_left _ _)) (Nat.gcd_dvd_right _ _))

  131theorem kgcd_comm {N : ℕ} (a b : ZMod N) : kgcd N a b = kgcd N b a := by
  132  unfold kgcd; rw [Nat.gcd_comm a.val]

  134/-- A unit coefficient: `‖S(a,b;N)‖ ≤ τ(N) √N`. -/
  135theorem norm_kSum_le_of_isUnit_left {N : ℕ} [NeZero N] (a b : ZMod N) (ha : IsUnit a) :
  136    ‖kSum N a b‖ ≤ (#N.divisors : ℝ) * Real.sqrt N := by
  137  have := norm_kSum_le N a b
  138  rwa [kgcd_eq_one_of_isUnit_left a b ha, Nat.cast_one, one_mul] at this

  141    ‖kSum N a b‖ ≤ (#N.divisors : ℝ) * Real.sqrt N := by
  142  have := norm_kSum_le N a b
