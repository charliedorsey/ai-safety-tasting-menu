41/-- `e(x) = exp(2πix)`. -/
42noncomputable def eAdd (x : ℝ) : ℂ := Complex.exp (2 * Real.pi * Complex.I * (x : ℂ))

49/-- `Σ_{0 ≤ i < n} e(f(a+i))`: the exponential sum over the `n` consecutive integers
50`a, a+1, …, a+n-1`.  (Every set of integers in an interval is such a run.) -/
51noncomputable def expSum (f : ℝ → ℝ) (a : ℤ) (n : ℕ) : ℂ :=
52  ∑ i ∈ range n, eAdd (f ((a : ℝ) + i))

59/-- Tao's hypothesis (3): `|f^{(j)}(x)| = exp(O(j²)) T/N^j` for **all** `j ≥ 1`, with the
60`O`-constant `A₀`, two-sided. -/
61def DerivHyp (A₀ N T : ℝ) (f : ℝ → ℝ) (I : Set ℝ) : Prop :=
62  ∀ j : ℕ, 1 ≤ j → ∀ x ∈ I,
63    Real.exp (-(A₀ * (j : ℝ) ^ 2)) * (T / N ^ j) ≤ |iteratedDeriv j f x| ∧
64    |iteratedDeriv j f x| ≤ Real.exp (A₀ * (j : ℝ) ^ 2) * (T / N ^ j)
65
66/-- **The order-uniform Vinogradov exponential-sum estimate** (Tao, *254A Notes 5*, Thm 2(ii)).
67
68For `2 ≤ N ≤ B₀ T`, any run of at most `N + 1` consecutive integers (an interval of length
69`≤ N`), `f` smooth on an open set containing the interval and obeying hypothesis (3) with
70constant `A₀`, and any `k ≥ 1` with `T ≤ N^k`:
71`|Σ e(f(n))| ≤ C · N · N^{-c/k²}`,
72where `c > 0` and `C` depend only on `(A₀, B₀)`.
73
74This is a `Prop`, not an axiom.  Everything downstream takes it as an explicit hypothesis. -/
75def UniformVinogradovEstimate : Prop :=
76  ∀ A₀ B₀ : ℝ, 0 < A₀ → 0 < B₀ →
77    ∃ c C : ℝ, 0 < c ∧ 0 ≤ C ∧
78      ∀ (N T : ℝ) (k : ℕ) (f : ℝ → ℝ) (a : ℤ) (n : ℕ) (U : Set ℝ),
79        2 ≤ N → N ≤ B₀ * T → 1 ≤ k → T ≤ N ^ k →
80        (n : ℝ) - 1 ≤ N →
81        IsOpen U → runIcc a n ⊆ U → ContDiffOn ℝ ∞ f U →
82        DerivHyp A₀ N T f (runIcc a n) →
83        ‖expSum f a n‖ ≤ C * N * N ^ (-(c / (k : ℝ) ^ 2))

226/-- **Smallest Type I analytic input.** For every `A > 0`, for all large `T`:
227for every scale `L ≥ 2` with `L ≤ T` and `log L ≥ (log T)^σ`, every shift `θ`, and every run of
228consecutive integers `a, …, a+n-1` with all `x+θ ∈ [L, 2L]`,
229`|Σ e(TL/(x+θ))| ≤ L (log T)^{-A}`. -/
230def ReciprocalPhaseVK (σ : ℝ) : Prop :=
231  ∀ A : ℝ, 0 < A → ∃ T₀ : ℝ, ∀ (L T θ : ℝ) (a : ℤ) (n : ℕ),
232    T₀ ≤ T → 2 ≤ L → L ≤ T → Real.log T ^ σ ≤ Real.log L →
233    (∀ x ∈ runIcc a n, L ≤ x + θ ∧ x + θ ≤ 2 * L) →
234    ‖expSum (recipPhase (T * L) θ) a n‖ ≤ L * Real.log T ^ (-A)

316    calc c / 4 * τ ^ (3 * σ - 2) ≤ c / 4 * (ℓ ^ 3 / τ ^ 2) :=

337  have hℓpos : 0 < Real.log L := Real.log_pos (by linarith)
338  have hρ1 : 1 ≤ Real.log T / Real.log L := by
339    rw [le_div_iff₀ hℓpos, one_mul]; exact Real.log_le_log hLpos hLT
340  have hk1 : 1 ≤ k := by

342    exact_mod_cast this
343  have hTLk : T ≤ L ^ k := by
344    have h1 : Real.log T ≤ k * Real.log L := by
345      have := Nat.le_ceil (Real.log T / Real.log L)
346      rw [div_le_iff₀ hℓpos] at this; linarith
347    rw [← Real.exp_log hTpos, ← Real.rpow_natCast, Real.rpow_def_of_pos hLpos]
348    apply Real.exp_le_exp.2; linarith
349  -- window ⇒ run length ≤ L
350  have hlen : (n : ℝ) - 1 ≤ L := by
351    rcases Nat.eq_zero_or_pos n with rfl | hn
352    · simp; linarith
353    have ha : (a : ℝ) ∈ runIcc a n := by
354      refine ⟨le_refl _, ?_⟩
355      have : (1 : ℝ) ≤ n := by exact_mod_cast hn
356      linarith
357    have hb : (a : ℝ) + n - 1 ∈ runIcc a n := by
358      refine ⟨?_, le_refl _⟩
359      have : (1 : ℝ) ≤ n := by exact_mod_cast hn
360      linarith
361    have h1 := (hwin _ ha).1
362    have h2 := (hwin _ hb).2
363    linarith
364  have hsub : runIcc a n ⊆ Set.Ioi (-θ) := by
365    intro x hx
366    have := (hwin x hx).1
367    show -θ < x
368    linarith
369  have hder : DerivHyp 2 L T (recipPhase (T * L) θ) (runIcc a n) := by
370    have h := recipPhase_derivHyp (X := T * L) (θ := θ) (by positivity) hLpos (runIcc a n) hwin
371    rwa [mul_div_assoc, div_self hLpos.ne', mul_one] at h
372  have hmain := hbound L T k (recipPhase (T * L) θ) a n (Set.Ioi (-θ)) hL
373    (by rw [one_mul]; exact hLT) hk1 hTLk hlen isOpen_Ioi hsub (recipPhase_contDiffOn _ _) hder
374  have hsave := saving_le hc hL hLT hσL
375  have hy := hy₀ (Real.log T) hlogT

378        mul_le_mul_of_nonneg_left hsave (by positivity)

380    _ ≤ L * Real.log T ^ (-A) := mul_le_mul_of_nonneg_left hy hLpos.le

384/-- Abel summation: nonnegative nonincreasing weights cost only the first weight. -/
385theorem abel_antitone {z : ℕ → ℂ} {w : ℕ → ℝ} (n : ℕ) (hw0 : ∀ i, 0 ≤ w i)
386    (hwm : ∀ i, w (i + 1) ≤ w i) {B : ℝ}
387    (hB : ∀ m ≤ n, ‖∑ i ∈ range m, z i‖ ≤ B) :
388    ‖∑ i ∈ range n, (w i : ℂ) * z i‖ ≤ w 0 * B := by
389  have hB0 : 0 ≤ B := le_trans (norm_nonneg _) (hB 0 (Nat.zero_le _))
390  -- induction on n with the stronger statement involving the last weight
391  -- S_n = Σ_{i<n} w_i z_i = w_{n-1} Z_n + Σ_{i<n-1} (w_i - w_{i+1}) Z_{i+1}
392  set Z : ℕ → ℂ := fun m => ∑ i ∈ range m, z i with hZ
393  have hrep : ∀ m, ∑ i ∈ range m, (w i : ℂ) * z i =
394      (w m : ℂ) * Z m + ∑ i ∈ range m, ((w i - w (i + 1) : ℝ) : ℂ) * Z (i + 1) := by
395    intro m
396    induction m with
397    | zero => simp [hZ]
398    | succ m ih =>
399      rw [sum_range_succ, ih, sum_range_succ]
400      simp only [hZ, sum_range_succ]
401      push_cast
402      ring
403  have hmono : ∀ m, w m ≤ w 0 := by
404    intro m; induction m with
405    | zero => exact le_refl _
406    | succ m ih => exact (hwm m).trans ih
407  have htel : ∀ m, ∑ i ∈ range m, (w i - w (i + 1)) = w 0 - w m := by
408    intro m; induction m with
409    | zero => simp
410    | succ m ih => rw [sum_range_succ, ih]; ring
411  rw [hrep n]
412  calc ‖(w n : ℂ) * Z n + ∑ i ∈ range n, ((w i - w (i + 1) : ℝ) : ℂ) * Z (i + 1)‖
413      ≤ ‖(w n : ℂ) * Z n‖ + ‖∑ i ∈ range n, ((w i - w (i + 1) : ℝ) : ℂ) * Z (i + 1)‖ :=
414        norm_add_le _ _
415    _ ≤ w n * B + ∑ i ∈ range n, (w i - w (i + 1)) * B := by
416        gcongr
417        · rw [norm_mul, Complex.norm_real, Real.norm_of_nonneg (hw0 n)]
418          exact mul_le_mul_of_nonneg_left (hB n le_rfl) (hw0 n)
419        · refine (norm_sum_le _ _).trans (sum_le_sum fun i hi => ?_)
420          rw [norm_mul, Complex.norm_real, Real.norm_of_nonneg (sub_nonneg.2 (hwm i))]
421          exact mul_le_mul_of_nonneg_left (hB (i + 1) (mem_range.1 hi))
422            (sub_nonneg.2 (hwm i))
423    _ = w 0 * B := by rw [← sum_mul, htel]; ring

425/-- **Type I class sum, harmonic weights (conditional on `ReciprocalPhaseVK σ` only).**
426
427Source shape (C2 §4.4, supplier §4): `V = Σ_j e(X₀/(j+θ)) / t_j` with `t_j = Q (j+θ)`, over any
428run of consecutive `j` with `j + θ ∈ [L, 2L]`, where `X₀ = T L`.  Then for every `A > 0` and all
429large `T`, in the VK range `L ≤ T`, `log L ≥ (log T)^σ`:  `|V| ≤ (log T)^{-A} / Q`.
430(Trivially `|V| ≤ (n/L)/Q ≤ 1/Q`; the gain is the full `(log T)^{-A}`, for every sub-run.) -/
431theorem typeI_class_sum_le {σ : ℝ} (hVK : ReciprocalPhaseVK σ) (A : ℝ) (hA : 0 < A) :
432    ∃ T₀ : ℝ, ∀ (L T θ Q : ℝ) (a : ℤ) (n : ℕ),
433      T₀ ≤ T → 2 ≤ L → L ≤ T → Real.log T ^ σ ≤ Real.log L → 0 < Q →
434      (∀ x ∈ runIcc a n, L ≤ x + θ ∧ x + θ ≤ 2 * L) →
435      ‖∑ i ∈ range n, ((1 / (Q * ((a : ℝ) + i + θ)) : ℝ) : ℂ) *
436          eAdd (recipPhase (T * L) θ ((a : ℝ) + i))‖ ≤ Real.log T ^ (-A) / Q := by
437  obtain ⟨T₀, hT₀⟩ := hVK A hA
438  refine ⟨T₀, fun L T θ Q a n hT hL hLT hσL hQ hwin => ?_⟩
439  have hLpos : 0 < L := by linarith
440  -- every prefix run lies in the window
441  have hprefix : ∀ m ≤ n, ∀ x ∈ runIcc a m, L ≤ x + θ ∧ x + θ ≤ 2 * L := by
442    intro m hm x hx
443    apply hwin
444    refine ⟨hx.1, hx.2.trans ?_⟩
445    have : (m : ℝ) ≤ n := by exact_mod_cast hm
446    linarith
447  -- the weights as a function of i (extended by the boundary value past n, still antitone)
448  set w : ℕ → ℝ := fun i => 1 / (Q * ((a : ℝ) + i + θ)) with hw
449  -- positivity of a + i + θ for i < n, and a + θ ≥ L
450  by_cases hn : n = 0
451  · subst hn; simp
452    exact div_nonneg (Real.rpow_nonneg (Real.log_nonneg (by linarith)) _) hQ.le
453  have haθ : L ≤ (a : ℝ) + θ := by
454    have ha : (a : ℝ) ∈ runIcc a n := by
455      refine ⟨le_refl _, ?_⟩
456      have : (1 : ℝ) ≤ n := by exact_mod_cast Nat.one_le_iff_ne_zero.2 hn
457      linarith
458    exact (hwin _ ha).1
459  have hpos : ∀ i : ℕ, 0 < (a : ℝ) + i + θ := by
460    intro i; have : (0 : ℝ) ≤ i := Nat.cast_nonneg i; linarith
461  have hw0 : ∀ i, 0 ≤ w i := fun i => by
462    simp only [hw]; exact div_nonneg zero_le_one (mul_pos hQ (hpos i)).le
463  have hwm : ∀ i, w (i + 1) ≤ w i := fun i => by
464    simp only [hw]
465    apply one_div_le_one_div_of_le (mul_pos hQ (hpos i))
466    apply mul_le_mul_of_nonneg_left _ hQ.le
467    push_cast; linarith
468  have hB : ∀ m ≤ n, ‖∑ i ∈ range m, eAdd (recipPhase (T * L) θ ((a : ℝ) + i))‖ ≤
469      L * Real.log T ^ (-A) := by
470    intro m hm
471    exact hT₀ L T θ a m hT hL hLT hσL (hprefix m hm)
472  have habel := abel_antitone (z := fun i => eAdd (recipPhase (T * L) θ ((a : ℝ) + i)))
473    (w := w) n hw0 hwm hB
474  have hw0le : w 0 ≤ 1 / (Q * L) := by
475    simp only [hw]
476    apply one_div_le_one_div_of_le (mul_pos hQ hLpos)
477    apply mul_le_mul_of_nonneg_left _ hQ.le
478    simpa using haθ
479  calc ‖∑ i ∈ range n, ((1 / (Q * ((a : ℝ) + i + θ)) : ℝ) : ℂ) *
480          eAdd (recipPhase (T * L) θ ((a : ℝ) + i))‖
481      = ‖∑ i ∈ range n, (w i : ℂ) * eAdd (recipPhase (T * L) θ ((a : ℝ) + i))‖ := rfl
482    _ ≤ w 0 * (L * Real.log T ^ (-A)) := habel
483    _ ≤ 1 / (Q * L) * (L * Real.log T ^ (-A)) := by
484        apply mul_le_mul_of_nonneg_right hw0le
485        exact mul_nonneg hLpos.le (Real.rpow_nonneg (Real.log_nonneg (by linarith)) _)
486    _ = Real.log T ^ (-A) / Q := by field_simp
