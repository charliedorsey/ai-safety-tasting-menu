    8  longInverseLeft δ r s ∨ longInverseRight δ r s

   11    (δ : ℝ) (q a h : ℕ) : ℝ :=
   12  regionSumMiddleWhere (Jlow h) (Jhigh h) q h

   20theorem foldedMiddle_long_core (Jlow Jhigh q a h : ℕ) (δ : ℝ) :
   21    regionSumMiddle Jlow Jhigh q h (centralGate h) (foldRow q a) =
   22      regionSumMiddleWhere Jlow Jhigh q h (centralGate h)
   23        (longEither δ) (foldRow q a) +
   24      regionSumMiddleWhere Jlow Jhigh q h (centralGate h)
   25        (fun r s => ¬ longEither δ r s) (foldRow q a) :=
   26  regionSumMiddleWhere_add_compl Jlow Jhigh q h

   29theorem middleSublog_of_long_core
   30    (Jlow Jhigh : ℕ → ℕ) (δ : ℝ)
   31    (hLong : SourceUniformSublog (signedLongMiddle Jlow Jhigh δ))
   32    (hCore : SourceUniformSublog (signedCoreMiddle Jlow Jhigh δ)) :
   33    SourceUniformSublog (middleSector Jlow Jhigh) := by
   34  intro q hq hsq eps heps
   35  have he : 0 < eps / 2 := by positivity
   36  obtain ⟨Hl, hl⟩ := hLong q hq hsq (eps / 2) he
   37  obtain ⟨Hc, hc⟩ := hCore q hq hsq (eps / 2) he
   38  refine ⟨max Hl Hc, ?_⟩
   39  intro a h ha hh hp hH
   40  have h1 := hl a h ha hh hp ((le_max_left _ _).trans hH)
   41  have h2 := hc a h ha hh hp ((le_max_right _ _).trans hH)
   42  change |regionSumMiddle (Jlow h) (Jhigh h) q h
   43    (centralGate h) (foldRow q a)| ≤ eps * Real.log (h : ℝ)
   44  rw [foldedMiddle_long_core]
   45  change |signedLongMiddle Jlow Jhigh δ q a h +
   46    signedCoreMiddle Jlow Jhigh δ q a h| ≤ eps * Real.log (h : ℝ)
   47  calc
   48    _ ≤ |signedLongMiddle Jlow Jhigh δ q a h| +
   49        |signedCoreMiddle Jlow Jhigh δ q a h| := abs_add_le _ _
   50    _ ≤ (eps / 2) * Real.log (h : ℝ) +
   51        (eps / 2) * Real.log (h : ℝ) := add_le_add h1 h2
   52    _ = eps * Real.log (h : ℝ) := by ring
