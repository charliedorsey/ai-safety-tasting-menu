7# `GoldC.QuarterFold` — Hunt A: quarter-middle regions, the generalized fold, reflection oddness

9Transcription-level (GoldC namespace, independent of the production tree).

11* `regionSum q h gate F`: the literal quarter-middle source shape — **all** decorations
12  `d ⊆ decoratedHeightPrimeSet q h`, the quarter mask `Z(h) < ∏e`, `4∏e ≤ H_d²`, both ordered
13  orientations `∅ ≠ B ⊊ e`, weights `V_d ω(e)`, with an arbitrary pair gate.

479/-- Two `M`-periodic solutions of the same step equation that agree at `0` agree everywhere. -/
480lemma step_unique (M n : ℕ) (hM : 2 ≤ M) (hn : n.Coprime M) (f₁ f₂ g : ℕ → ℝ)
481    (h₁ : ∀ x y, x ≡ y [MOD M] → f₁ x = f₁ y) (h₂ : ∀ x y, x ≡ y [MOD M] → f₂ x = f₂ y)
482    (s₁ : ∀ y, f₁ (y + n) - f₁ y = g y) (s₂ : ∀ y, f₂ (y + n) - f₂ y = g y) (h0 : f₁ 0 = f₂ 0)
483    (y : ℕ) : f₁ y = f₂ y := by
484  have hk : ∀ k, f₁ (k * n) = f₂ (k * n) := by
485    intro k
486    induction k with
487    | zero => simpa using h0
488    | succ k ih =>
489      have e1 := s₁ (k * n)
490      have e2 := s₂ (k * n)
491      rw [show (k + 1) * n = k * n + n by ring]
492      linarith
493  have hy : y ≡ ((y * invMod n M) % M) * n [MOD M] := by
494    have hnu : n * invMod n M ≡ 1 [MOD M] := mul_invMod_modEq hM hn
495    calc y = y * 1 := (mul_one y).symm
496      _ ≡ y * (n * invMod n M) [MOD M] := Nat.ModEq.mul_left _ hnu.symm
497      _ = (y * invMod n M) * n := by ring
498      _ ≡ ((y * invMod n M) % M) * n [MOD M] := Nat.ModEq.mul_right _ (Nat.mod_modEq _ _).symm
499  rw [h₁ _ _ hy, h₂ _ _ hy, hk]

505lemma pot_reflect (g : ℕ → ℝ) (M ν : ℕ) (hM : 2 ≤ M) (hν : ν.Coprime M) (hνM : ν ≤ M)
506    (hper : ∀ y, g (y + M) = g y)
507    (hcyc : ∀ n, n.Coprime M → ∑ t ∈ Finset.range M, g (t * n) = 0) (y : ℕ) :

1052theorem quarterMiddleSublogT_of_regions (hC : RegionSublog centralGate) (hL : RegionSublog leftGate)
1053    (hR : RegionSublog rightGate) (hU : RegionSublog upperGate) : QuarterMiddleSublogT := by
