    4/-!
    5# Quillon CSC3 F5d: the R route with a `u`-dependent amplitude (exact-DFT coefficients)
    6
    7* `cauchy_class_split_amp`: as `cauchy_class_split`, but the inner coefficient is `a v · c v u`.
    8  After Cauchy, each class sum carries the amplitude `A_{vv'}(u) = c v u · conj(c v' u)`.
    9* `class_amp_bound`: VK along `ρ + Q t` with a bounded-variation amplitude (Abel summation):
   10
   11      ‖Σ_{t∈(N,M]} A t · e(Y/(ρ+Qt))‖ ≤ (‖A M‖ + Σ‖ΔA‖) · C·(N e^{−c log³N/log²(|Y|/Q)} + N²/(|Y|/Q)).
   12
   13  This removes the need for any Fourier truncation in R. The coefficients
   14  `a_{vu}(k) = (i/2vu)cot(πk/vu)` are handled exactly.
   15
   16No axiom, no `sorry`.
   17-/

   26theorem cauchy_class_split_amp {Q : ℕ} (hQ : 1 ≤ Q) (𝒰 I 𝒱 : Finset ℕ) (h𝒰 : 𝒰 ⊆ I)
   27    (a b : ℕ → ℂ) (c : ℕ → ℕ → ℂ) (φ : ℕ → ℕ → ℝ) (G : ℕ → ℕ → ℂ)
   28    (hG : ∀ v ρ, ‖G v ρ‖ ≤ 1) :
   29    ‖∑ u ∈ 𝒰, b u * ∑ v ∈ 𝒱, a v * (c v u * (eR (φ v u) * G v (u % Q)))‖ ^ 2 ≤
   30      (∑ u ∈ 𝒰, ‖b u‖ ^ 2) *
   31        ∑ v ∈ 𝒱, ∑ v' ∈ 𝒱, ‖a v‖ * ‖a v'‖ *
   32          ∑ ρ ∈ range Q, ‖∑ u ∈ I.filter (fun u => u % Q = ρ),
   33            (c v u * (starRingEnd ℂ) (c v' u)) * eR (φ v u - φ v' u)‖ := by
   34  set F : ℕ → ℂ := fun u => ∑ v ∈ 𝒱, a v * (c v u * (eR (φ v u) * G v (u % Q))) with hF
   35  have hCS : ‖∑ u ∈ 𝒰, b u * F u‖ ^ 2 ≤ (∑ u ∈ 𝒰, ‖b u‖ ^ 2) * ∑ u ∈ 𝒰, ‖F u‖ ^ 2 := by
   36    calc ‖∑ u ∈ 𝒰, b u * F u‖ ^ 2 ≤ (∑ u ∈ 𝒰, ‖b u‖ * ‖F u‖) ^ 2 := by
   37          refine pow_le_pow_left₀ (norm_nonneg _) ?_ 2
   38          exact (norm_sum_le _ _).trans (le_of_eq (Finset.sum_congr rfl fun u _ => norm_mul _ _))
   39      _ ≤ _ := Finset.sum_mul_sq_le_sq_mul_sq _ _ _
   40  refine hCS.trans (mul_le_mul_of_nonneg_left ?_ (Finset.sum_nonneg fun u _ => by positivity))
   41  have hdrop : ∑ u ∈ 𝒰, ‖F u‖ ^ 2 ≤ ∑ u ∈ I, ‖F u‖ ^ 2 :=
   42    Finset.sum_le_sum_of_subset_of_nonneg h𝒰 fun u _ _ => by positivity
   43  refine hdrop.trans ?_
   44  have hexp : ∀ u, ((‖F u‖ ^ 2 : ℝ) : ℂ) = ∑ v ∈ 𝒱, ∑ v' ∈ 𝒱,
   45      a v * (starRingEnd ℂ) (a v') * ((c v u * (starRingEnd ℂ) (c v' u)) *
   46        eR (φ v u - φ v' u) * (G v (u % Q) * (starRingEnd ℂ) (G v' (u % Q)))) := by
   47    intro u
   48    rw [Complex.ofReal_pow, ← Complex.mul_conj']
   49    simp only [hF, map_sum, map_mul]
   50    rw [Finset.sum_mul_sum]
   51    refine Finset.sum_congr rfl fun v _ => Finset.sum_congr rfl fun v' _ => ?_
   52    rw [eR_sub]; ring
   53  have hreal : ∑ u ∈ I, ‖F u‖ ^ 2 = ‖∑ u ∈ I, ((‖F u‖ ^ 2 : ℝ) : ℂ)‖ := by
   54    rw [← Complex.ofReal_sum, Complex.norm_real, Real.norm_eq_abs,
   55      abs_of_nonneg (Finset.sum_nonneg fun u _ => by positivity)]
   56  rw [hreal]
   57  simp_rw [hexp]
   58  rw [Finset.sum_comm]
   59  refine (norm_sum_le _ _).trans (Finset.sum_le_sum fun v _ => ?_)
   60  rw [Finset.sum_comm]
   61  refine (norm_sum_le _ _).trans (Finset.sum_le_sum fun v' _ => ?_)
   62  have hsplit : ∑ u ∈ I, a v * (starRingEnd ℂ) (a v') * ((c v u * (starRingEnd ℂ) (c v' u)) *
   63        eR (φ v u - φ v' u) * (G v (u % Q) * (starRingEnd ℂ) (G v' (u % Q)))) =
   64      a v * (starRingEnd ℂ) (a v') * ∑ ρ ∈ range Q, (G v ρ * (starRingEnd ℂ) (G v' ρ)) *
   65        ∑ u ∈ I.filter (fun u => u % Q = ρ),
   66          (c v u * (starRingEnd ℂ) (c v' u)) * eR (φ v u - φ v' u) := by
   67    rw [← Finset.mul_sum]
   68    congr 1
   69    rw [← Finset.sum_fiberwise_of_maps_to (g := fun u => u % Q) (t := range Q)
   70      (fun u _ => Finset.mem_range.2 (Nat.mod_lt _ (by omega)))]
   71    refine Finset.sum_congr rfl fun ρ _ => ?_
   72    rw [Finset.mul_sum]
   73    refine Finset.sum_congr rfl fun u hu => ?_
   74    rw [(Finset.mem_filter.1 hu).2]; ring
   75  rw [hsplit, norm_mul, norm_mul, Complex.norm_conj]
   76  gcongr
   77  refine (norm_sum_le _ _).trans (Finset.sum_le_sum fun ρ _ => ?_)
   78  rw [norm_mul, norm_mul, Complex.norm_conj]
   79  have h1 := hG v ρ
   80  have h2 := hG v' ρ
   81  calc ‖G v ρ‖ * ‖G v' ρ‖ * ‖∑ u ∈ I.filter (fun u => u % Q = ρ),
   82        (c v u * (starRingEnd ℂ) (c v' u)) * eR (φ v u - φ v' u)‖
   83      ≤ 1 * 1 * ‖∑ u ∈ I.filter (fun u => u % Q = ρ),
   84        (c v u * (starRingEnd ℂ) (c v' u)) * eR (φ v u - φ v' u)‖ := by gcongr
   85    _ = _ := by ring

   88theorem class_amp_bound :
   89    ∃ c C : ℝ, 0 < c ∧ 0 ≤ C ∧
   90      ∀ (Q ρ N M : ℕ) (Y : ℝ) (A : ℕ → ℂ), 1 ≤ Q → ρ < Q → 1 ≤ N → N ≤ M → M ≤ 2 * N →
   91        2 * (Q : ℝ) ≤ |Y| →
   92        ‖∑ t ∈ Ioc N M, A t * eR (Y / ((ρ : ℝ) + Q * t))‖ ≤
   93          (‖A M‖ + ∑ n ∈ Ico (N + 1) M, ‖A (n + 1) - A n‖) *
   94            (C * ((N : ℝ) * Real.exp (-(c * Real.log (N : ℝ) ^ 3 / Real.log (|Y| / Q) ^ 2)) +
   95              (N : ℝ) ^ 2 / (|Y| / Q))) := by
   96  obtain ⟨c, C, hc, hC, hS⟩ := suzuki_AP
   97  refine ⟨c, C, hc, hC, fun Q ρ N M Y A hQ hρ hN hNM hM hY => ?_⟩
   98  refine abel_bound A (fun t => eR (Y / ((ρ : ℝ) + Q * t))) N M hNM _ fun j hj1 hj2 => ?_
   99  unfold psum
  100  exact hS Q ρ N j Y hQ hρ hN (by omega) hY
