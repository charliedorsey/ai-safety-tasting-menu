   32/-- Verbatim copy of `GoldC.decoratedHeightPrimeSet`. -/
   33def decoratedHeightPrimeSet (q h : ℕ) : Finset ℕ :=
   34  h.primeFactors.filter (fun p => 5 ≤ p ∧ ¬ p ∣ q)

   36/-- Verbatim copy of `GoldC.decorationWeight`. -/
   37noncomputable def decorationWeight (h : ℕ) (s : Finset ℕ) : ℝ :=
   38  ∏ p ∈ s, if p ∣ h then 1 / ((p : ℝ) - 2) else 1 / ((p : ℝ) - 3)

   40lemma mem_P {q h p : ℕ} (hp : p ∈ decoratedHeightPrimeSet q h) :
   41    p.Prime ∧ p ∣ h ∧ 5 ≤ p ∧ h ≠ 0 := by
   42  unfold decoratedHeightPrimeSet at hp
   43  rw [Finset.mem_filter, Nat.mem_primeFactors] at hp
   44  exact ⟨hp.1.1, hp.1.2.1, hp.2.1, hp.1.2.2⟩

   46/-- **Exact identity**: the decoration mass is an Euler product. -/
   47theorem decoration_mass_eq_prod (q h : ℕ) :
   48    ∑ d ∈ (decoratedHeightPrimeSet q h).powerset, decorationWeight h d =
   49      ∏ p ∈ decoratedHeightPrimeSet q h, (1 + 1 / ((p : ℝ) - 2)) := by
   50  rw [Finset.prod_one_add]
   51  refine Finset.sum_congr rfl fun d hd => ?_
   52  unfold decorationWeight
   53  refine Finset.prod_congr rfl fun p hp => ?_
   54  have := (mem_P (Finset.mem_powerset.1 hd hp)).2.1
   55  simp [this]

   66/-- Primes in the dyadic block `[2^k, 2^(k+1))`. -/
   67def block (k : ℕ) : Finset ℕ := (Finset.Ico (2 ^ k) (2 ^ (k + 1))).filter Nat.Prime
   68
   69lemma block_card_mul_le (k : ℕ) : k * (block k).card ≤ 2 ^ (k + 2) := by
   70  -- (2^k)^card ≤ ∏ block ≤ primorial (2^(k+1)) ≤ 4^(2^(k+1)) = 2^(2^(k+2))
   71  have h1 : (2 ^ k) ^ (block k).card ≤ ∏ p ∈ block k, p :=
   72    Finset.pow_card_le_prod _ _ _ fun p hp => (Finset.mem_Ico.1 (Finset.mem_filter.1 hp).1).1
   73  have hsub : block k ⊆ (Finset.range (2 ^ (k + 1) + 1)).filter Nat.Prime := by
   74    intro p hp
   75    rw [block, Finset.mem_filter, Finset.mem_Ico] at hp
   76    rw [Finset.mem_filter, Finset.mem_range]
   77    exact ⟨by omega, hp.2⟩
   78  have h2 : ∏ p ∈ block k, p ≤ primorial (2 ^ (k + 1)) := by
   79    unfold primorial
   80    apply Nat.le_of_dvd
   81    · exact Finset.prod_pos fun p hp => (Finset.mem_filter.1 hp).2.pos
   82    · exact Finset.prod_dvd_prod_of_subset _ _ _ hsub
   83  have h3 := primorial_le_four_pow (2 ^ (k + 1))
   84  have h4 : (4 : ℕ) ^ (2 ^ (k + 1)) = 2 ^ (2 ^ (k + 2)) := by
   85    rw [show (4 : ℕ) = 2 ^ 2 by norm_num, ← pow_mul]; congr 1; ring
   86  have h5 : 2 ^ (k * (block k).card) ≤ 2 ^ (2 ^ (k + 2)) := by
   87    rw [pow_mul]; calc _ ≤ _ := h1
   88      _ ≤ _ := h2
   89      _ ≤ _ := h3
   90      _ = _ := h4
   91  exact (Nat.pow_le_pow_iff_right (by norm_num)).1 h5

   93lemma block_sum_le (k : ℕ) (hk : 1 ≤ k) :
   94    ∑ p ∈ block k, (1 / (p : ℝ)) ≤ 4 / k := by
   95  have hk' : (0 : ℝ) < k := by exact_mod_cast hk
   96  have hpos : (0 : ℝ) < 2 ^ k := by positivity
   97  calc ∑ p ∈ block k, (1 / (p : ℝ)) ≤ ∑ _p ∈ block k, (1 / (2 : ℝ) ^ k) := by
   98        refine Finset.sum_le_sum fun p hp => ?_
   99        have : 2 ^ k ≤ p := (Finset.mem_Ico.1 (Finset.mem_filter.1 hp).1).1
  100        have : (2 : ℝ) ^ k ≤ p := by exact_mod_cast this
  101        exact one_div_le_one_div_of_le hpos this
  102    _ = (block k).card / (2 : ℝ) ^ k := by rw [Finset.sum_const, nsmul_eq_mul]; ring
  103    _ ≤ 4 / k := by
  104        rw [div_le_div_iff₀ hpos hk']
  105        have h := block_card_mul_le k
  106        have h' : (k : ℝ) * (block k).card ≤ 2 ^ (k + 2) := by exact_mod_cast h
  107        rw [pow_add] at h'; nlinarith

  132/-- **Dyadic Mertens-type bound**: `Σ_{p < 2^(K+1)} 1/p ≤ 4·H_K`. -/
  133theorem sum_inv_primes_below_two_pow (K : ℕ) :
  134    ∑ p ∈ primesBelowPow (K + 1), (1 / (p : ℝ)) ≤ 4 * (harmonic K : ℝ) := by
  135  induction K with
  136  | zero =>
  137    have : primesBelowPow 1 = ∅ := by
  138      ext p; simp only [primesBelowPow, Finset.mem_filter, Finset.mem_range,
  139        Finset.notMem_empty, iff_false, not_and]
  140      intro hp hpr; have := hpr.two_le; norm_num at hp; omega
  141    simp [this]
  142  | succ K ih =>
  143    rw [primesBelowPow_succ, Finset.sum_union (disjoint_block _), harmonic_succ]
  144    have hb := block_sum_le (K + 1) (by omega)
  145    push_cast at hb ⊢
  146    have : (4 : ℝ) / (K + 1) = 4 * (K + 1 : ℝ)⁻¹ := by ring
  147    linarith

  157/-- Large decorated primes: at most `log h / log (Y+1)` of them exceed `Y`. -/
  158lemma large_part_le (q h Y : ℕ) (hh : 0 < h) :
  159    ∑ p ∈ (decoratedHeightPrimeSet q h).filter (fun p => Y < p), (1 / (p : ℝ)) ≤
  160      Real.log h / (((Y : ℝ) + 1) * Real.log ((Y : ℝ) + 1)) ∨ Y = 0 := by
  161  rcases Nat.eq_zero_or_pos Y with hY | hY
  162  · exact Or.inr hY
  163  left
  164  set S := (decoratedHeightPrimeSet q h).filter (fun p => Y < p)
  165  have hY0 : (0:ℝ) < Y := by exact_mod_cast hY
  166  have hY1 : (1 : ℝ) < (Y : ℝ) + 1 := by linarith
  167  have hlogY : 0 < Real.log ((Y : ℝ) + 1) := Real.log_pos hY1
  168  -- (Y+1)^card ≤ ∏ S ≤ h
  169  have hprod : ((Y + 1) ^ S.card : ℕ) ≤ h := by
  170    have h1 : (Y + 1) ^ S.card ≤ ∏ p ∈ S, p :=
  171      Finset.pow_card_le_prod _ _ _ fun p hp => (Finset.mem_filter.1 hp).2
  172    have h2 : ∏ p ∈ S, p ∣ h := by
  173      calc ∏ p ∈ S, p ∣ ∏ p ∈ h.primeFactors, p := by
  174            refine Finset.prod_dvd_prod_of_subset S h.primeFactors id fun p hp => ?_
  175            have := mem_P (Finset.mem_filter.1 hp).1
  176            exact Nat.mem_primeFactors.2 ⟨this.1, this.2.1, hh.ne'⟩
  177        _ ∣ h := Nat.prod_primeFactors_dvd h
  178    exact h1.trans (Nat.le_of_dvd hh h2)
  179  have hcard : (S.card : ℝ) * Real.log ((Y : ℝ) + 1) ≤ Real.log h := by
  180    have : (((Y + 1) ^ S.card : ℕ) : ℝ) ≤ h := by exact_mod_cast hprod
  181    have hpos : (0 : ℝ) < (((Y + 1) ^ S.card : ℕ) : ℝ) := by positivity
  182    have := Real.log_le_log hpos this
  183    push_cast at this
  184    rwa [Real.log_pow] at this
  185  calc ∑ p ∈ S, (1 / (p : ℝ)) ≤ ∑ _p ∈ S, (1 / ((Y : ℝ) + 1)) := by
  186        refine Finset.sum_le_sum fun p hp => ?_
  187        have : Y + 1 ≤ p := (Finset.mem_filter.1 hp).2
  188        have : (Y : ℝ) + 1 ≤ p := by exact_mod_cast this
  189        exact one_div_le_one_div_of_le (by linarith) this
  190    _ = S.card / ((Y : ℝ) + 1) := by rw [Finset.sum_const, nsmul_eq_mul]; ring
  191    _ ≤ Real.log h / (((Y : ℝ) + 1) * Real.log ((Y : ℝ) + 1)) := by
  192        rw [div_le_div_iff₀ (by linarith) (by positivity)]
  193        nlinarith

  208/-- **Lemma G (decoration Gronwall, crude scale)**: for `16 ≤ h`,
  209`Σ_{d ⊆ P} decorationWeight h d ≤ exp (10 + 8 · log (Nat.log 2 ⌊log h⌋₊))`. -/
  210theorem decoration_mass_le_exp_loglog (q h : ℕ) (hh : 16 ≤ h) :
  211    ∑ d ∈ (decoratedHeightPrimeSet q h).powerset, decorationWeight h d ≤
  212      Real.exp (10 + 8 * Real.log (Nat.log 2 ⌊Real.log h⌋₊)) := by
  213  set P := decoratedHeightPrimeSet q h
  214  set Y := ⌊Real.log h⌋₊
  215  have hh0 : 0 < h := by omega
  216  have hlogh : Real.log 16 ≤ Real.log h :=
  217    Real.log_le_log (by norm_num) (by exact_mod_cast hh)
  218  have hlog16 : (2.7 : ℝ) < Real.log 16 := by
  219    rw [show (16 : ℝ) = 2 ^ 4 by norm_num, Real.log_pow]
  220    have := Real.log_two_gt_d9; push_cast; linarith
  221  have hY2 : 2 ≤ Y := by
  222    apply Nat.le_floor; push_cast; linarith
  223  have hYlt : Real.log h < (Y : ℝ) + 1 := Nat.lt_floor_add_one _
  224  -- large part ≤ 1

  226    rcases large_part_le q h Y hh0 with h1 | h1
  227    · refine h1.trans ?_
  228      have hY3 : (3 : ℝ) ≤ (Y : ℝ) + 1 := by
  229        have : (2 : ℝ) ≤ Y := by exact_mod_cast hY2
  230        linarith
  231      have hl3 : 1 < Real.log ((Y : ℝ) + 1) := by
  232        have : Real.log 3 ≤ Real.log ((Y : ℝ) + 1) := Real.log_le_log (by norm_num) hY3
  233        have h3 : (1 : ℝ) < Real.log 3 := by

  235          rw [Real.lt_log_iff_exp_lt (by norm_num)]; linarith
  236        linarith

  238      nlinarith

  240  have hsmall := small_part_le q h Y
  241  have hharm := harmonic_le_one_add_log (Nat.log 2 Y)
  242  -- Σ 1/(p-2) ≤ 2 Σ 1/p ≤ 2 (4(1+log m) + 1)
  243  have hsplit : ∑ p ∈ P, (1 / (p : ℝ)) =
  244      ∑ p ∈ P.filter (fun p => ¬ Y < p), (1 / (p : ℝ)) +
  245      ∑ p ∈ P.filter (fun p => Y < p), (1 / (p : ℝ)) := by

  251      _ ≤ 10 + 8 * Real.log (Nat.log 2 Y) := by
  252          rw [hsplit]; nlinarith

  254  refine (prod_one_add_le_exp_sum P _ fun p hp => ?_).trans (Real.exp_le_exp.2 hsum)
  255  have : (5 : ℝ) ≤ p := by exact_mod_cast (mem_P hp).2.2.1

  298/-! ## Sign-safe exact decomposition (replaces any "sandwich" of a signed sum)
  299
  300If a coupled mask is written exactly as `G = A + E` with `E ≥ 0`, then for any signed
  301summand `F` with `|F| ≤ C` and nonnegative weights `w`, the error made by replacing `G`
  302by `A` is at most `C · Σ w E`. No upper/lower sandwich of a signed sum is used. -/
  303theorem signed_mask_replace {ι : Type*} (S : Finset ι) (w F G A E : ι → ℝ) (C : ℝ)
  304    (hw : ∀ i ∈ S, 0 ≤ w i) (hF : ∀ i ∈ S, |F i| ≤ C)
  305    (hdec : ∀ i ∈ S, G i = A i + E i) (hE : ∀ i ∈ S, 0 ≤ E i) :
  306    |∑ i ∈ S, w i * F i * G i - ∑ i ∈ S, w i * F i * A i| ≤ C * ∑ i ∈ S, w i * E i := by
  307  rw [← Finset.sum_sub_distrib, Finset.mul_sum]
  308  refine (Finset.abs_sum_le_sum_abs _ _).trans (Finset.sum_le_sum fun i hi => ?_)
  309  rw [hdec i hi, show w i * F i * (A i + E i) - w i * F i * A i = F i * (w i * E i) by ring,
  310    abs_mul, abs_of_nonneg (mul_nonneg (hw i hi) (hE i hi))]
  311  exact mul_le_mul_of_nonneg_right (hF i hi) (mul_nonneg (hw i hi) (hE i hi))

  313/-! ## Corrected W-diagonal gcd count (checkpoint §4)
  314
  315The naive count `max_y Σ_{y'≡y (g)} α_{y'} ≤ C/Y₀` loses a fatal `Y^{1/2}/Y₀`. The symmetric
  316double count below is the correct one: `Σ_{y<y', g | y'-y} 1/(y y') ≤ (1+log Y)²/g`. The
  317full off-diagonal (`y ≠ y'`) is twice this, by symmetry. -/
  318theorem gcd_pair_sum_le (g Y : ℕ) (hg : 1 ≤ g) :
  319    ∑ y ∈ Finset.Icc 1 Y, ∑ y' ∈ (Finset.Icc 1 Y).filter (fun y' => y < y' ∧ g ∣ y' - y),
  320      (1 / ((y : ℝ) * y')) ≤ (1 + Real.log Y) ^ 2 / g := by
  321  have hgR : (0 : ℝ) < g := by exact_mod_cast hg
  322  set H : ℝ := ∑ k ∈ Finset.Icc 1 Y, (1 / (k : ℝ)) with hH
  323  have hHle : H ≤ 1 + Real.log Y := by
  324    have := harmonic_le_one_add_log Y
  325    rw [harmonic_eq_sum_Icc] at this; push_cast at this
  326    simpa [hH, one_div] using this
  327  have hH0 : 0 ≤ H := Finset.sum_nonneg fun k _ => by positivity
  328  have inner : ∀ y ∈ Finset.Icc 1 Y,
  329      ∑ y' ∈ (Finset.Icc 1 Y).filter (fun y' => y < y' ∧ g ∣ y' - y), (1 / ((y : ℝ) * y'))
  330        ≤ (1 / (y : ℝ)) * (1 / (g : ℝ)) * H := by
  331    intro y hy
  332    have hy1 : 1 ≤ y := (Finset.mem_Icc.1 hy).1
  333    have hyR : (0 : ℝ) < y := by exact_mod_cast hy1
  334    have hinj : Set.InjOn (fun k => y + k * g) (Finset.Icc 1 Y : Set ℕ) := by
  335      intro a _ b _ hab
