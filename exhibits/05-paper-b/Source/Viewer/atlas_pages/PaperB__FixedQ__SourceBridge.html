17/-- Source-facing anchored level-one summand.  The admissibility indicator and
18the active low-prime factors are written through the literal occupied-residue
19local ratios; the high-prime tail uses the same occupied-residue ratios. -/
20noncomputable def manuscriptAnchoredSummand
21    (q a h y : ℕ) : ℝ :=
22  occupiedFixedQLowWeight q a h y *
23    (∏' p : ℕ, occupiedActiveHighLocalRatio q h y p)

25/-- Exact source-facing level-one anchored kernel, before the boundary
26subtraction appearing in the final transfer theorem. -/
27noncomputable def manuscriptA1Kernel
28    (q a h : ℕ) : ℝ :=
29  (∑ y ∈ Finset.Ioo 0 h, manuscriptAnchoredSummand q a h y) -
