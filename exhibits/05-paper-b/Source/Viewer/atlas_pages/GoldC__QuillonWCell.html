291theorem W_cell {s : ℕ} [NeZero s] (hs2 : 2 ≤ s) (P : Finset ℕ) (hP : ∀ p ∈ P, p.Prime)
292    (hsP : s = ∏ p ∈ P, p) (c₀ : ZMod s) (𝒴 𝒩 : Finset ℕ) (Y N M : ℕ) (hNM : N ≤ M)
293    (h𝒩 : 𝒩 ⊆ Ioc N M) (h𝒴 : 𝒴 ⊆ Icc 1 Y) (hunit : ∀ y ∈ 𝒴, IsUnit (y : ZMod s))
294    (α β : ℕ → ℂ) (A : ℝ) (hα : ∀ y ∈ 𝒴, ‖α y‖ ≤ A / y) :
295    ‖∑ n ∈ 𝒩, β n * ∑ y ∈ 𝒴, α y * GoldC.QuillonW.g c₀ y n‖ ^ 2 ≤
296      (∑ n ∈ 𝒩, ‖β n‖ ^ 2) * (((M - N : ℕ) : ℝ) * ∑ y ∈ 𝒴, ‖α y‖ ^ 2 +
297        A ^ 2 * (((M - N : ℕ) : ℝ) * (2 * (1 + Real.log Y) ^ 2 / s) +
298          (3 : ℝ) ^ ((P.card : ℝ) / 4) * (s : ℝ) ^ ((3 : ℝ) / 4) *
299            (∏ p ∈ P.filter (fun p => (ZMod.cast c₀ : ZMod p) = 0), (p : ℝ)) ^ ((1 : ℝ) / 4) *
300            (((M - N : ℕ) : ℝ) / s + 1 + Real.log s) *
301            (2 * (1 + Real.log Y) ^ 2 * ∏ p ∈ P, (1 + (p : ℝ) ^ (-(3 : ℝ) / 4))))) := by
302  classical
303  refine (GoldC.QuillonW.W_cauchy hs2 c₀ 𝒴 𝒩 hNM h𝒩 hunit α β (KB P)
304    (fun c b => norm_K_le_KB P hP s hsP b c)).trans ?_
305  apply mul_le_mul_of_nonneg_left _ (sum_nonneg fun n _ => by positivity)
306  set L : ℝ := ((M - N : ℕ) : ℝ) with hL
307  set G : ℝ := (∏ p ∈ P.filter (fun p => (ZMod.cast c₀ : ZMod p) = 0), (p : ℝ)) ^ ((1 : ℝ) / 4)
308    with hG
309  set Kc : ℝ := (3 : ℝ) ^ ((P.card : ℝ) / 4) * (s : ℝ) ^ ((3 : ℝ) / 4) * G *
310    (L / s + 1 + Real.log s) with hKc
311  have hL0 : 0 ≤ L := by positivity
312  have hsR : (1 : ℝ) ≤ s := by exact_mod_cast (show 1 ≤ s by omega)
313  have hlogs : 0 ≤ Real.log s := Real.log_nonneg hsR
314  have hG0 : 0 ≤ G := by positivity
315  have hKc0 : 0 ≤ Kc := by positivity
316  have hA : ∀ y ∈ 𝒴, 0 ≤ A / y := fun y hy => (norm_nonneg _).trans (hα y hy)
317  set T1 : ℕ → ℕ → ℝ := fun y y' => if y = y' then L * ‖α y‖ ^ 2 else 0 with hT1
318  set T2 : ℕ → ℕ → ℝ := fun y y' =>
319    if y ≠ y' ∧ ((s : ℕ) : ℤ) ∣ (y : ℤ) - y' then 1 / ((y : ℝ) * y') else 0 with hT2
320  set T3 : ℕ → ℕ → ℝ := fun y y' => if y ≠ y' then (1 / ((y : ℝ) * y')) *
321    (∏ p ∈ P.filter (fun p : ℕ => (p : ℤ) ∣ (y : ℤ) - y'), (p : ℝ)) ^ ((1 : ℝ) / 4) else 0
322    with hT3
323  have hT2n : ∀ y y', 0 ≤ T2 y y' := fun y y' => by simp only [hT2]; split_ifs <;> positivity
324  have hT3n : ∀ y y', 0 ≤ T3 y y' := fun y y' => by simp only [hT3]; split_ifs <;> positivity
325  have hT1n : ∀ y y', 0 ≤ T1 y y' := fun y y' => by simp only [hT1]; split_ifs <;> positivity
326  -- pointwise
327  have hpt : ∀ y ∈ 𝒴, ∀ y' ∈ 𝒴, ‖α y‖ * ‖α y'‖ *
328      (if (y : ZMod s) = y' then L else
329        KB P (c₀ * ((y : ZMod s)⁻¹ - (y' : ZMod s)⁻¹)) * (L / s + 1 + Real.log s)) ≤
330      T1 y y' + A ^ 2 * L * T2 y y' + A ^ 2 * Kc * T3 y y' := by
331    intro y hy y' hy'
332    have hy1 : 1 ≤ y := (mem_Icc.1 (h𝒴 hy)).1
333    have hy1' : 1 ≤ y' := (mem_Icc.1 (h𝒴 hy')).1
334    have hprod : ‖α y‖ * ‖α y'‖ ≤ A ^ 2 * (1 / ((y : ℝ) * y')) := by
335      calc ‖α y‖ * ‖α y'‖ ≤ (A / y) * (A / y') :=
336            mul_le_mul (hα y hy) (hα y' hy') (norm_nonneg _) (hA y hy)
337        _ = _ := by ring
338    have h2 := hT2n y y'; have h3 := hT3n y y'; have h1 := hT1n y y'
339    have hA2 : 0 ≤ A ^ 2 := sq_nonneg A
340    by_cases hyy : y = y'
341    · subst hyy
342      rw [if_pos rfl]
343      have hT1v : T1 y y = L * ‖α y‖ ^ 2 := if_pos rfl
344      have : ‖α y‖ * ‖α y‖ * L = L * ‖α y‖ ^ 2 := by ring
345      rw [this, hT1v]
346      have := mul_nonneg (mul_nonneg hA2 hL0) h2
347      have := mul_nonneg (mul_nonneg hA2 hKc0) h3
348      linarith
349    · have hT1z : T1 y y' = 0 := by simp only [hT1, if_neg hyy]
350      rw [hT1z, zero_add]
351      split_ifs with hcong
352      · -- congruent distinct pair
353        have hdvd : ((s : ℕ) : ℤ) ∣ (y : ℤ) - y' := by
354          have : ((y' : ℤ) : ZMod s) = ((y : ℤ) : ZMod s) := by push_cast; exact hcong.symm
355          exact (ZMod.intCast_eq_intCast_iff_dvd_sub _ _ _).1 this
356        have hT2v : T2 y y' = 1 / ((y : ℝ) * y') := if_pos (And.intro hyy hdvd)
357        rw [hT2v]
358        have := mul_nonneg (mul_nonneg hA2 hKc0) h3
359        calc ‖α y‖ * ‖α y'‖ * L ≤ A ^ 2 * (1 / ((y : ℝ) * y')) * L :=
360              mul_le_mul_of_nonneg_right hprod hL0
361          _ = A ^ 2 * L * (1 / ((y : ℝ) * y')) := by ring
362          _ ≤ _ := by linarith
363      · have hT3v : T3 y y' = (1 / ((y : ℝ) * y')) *
364            (∏ p ∈ P.filter (fun p : ℕ => (p : ℤ) ∣ (y : ℤ) - y'), (p : ℝ)) ^ ((1 : ℝ) / 4) := by
365          simp only [hT3, if_neg (not_not.2 hyy), if_pos hyy]
366        rw [hT3v]
367        have hKB := KB_offdiag P hP hsP c₀ y y' (hunit y hy) (hunit y' hy')
368        have hLf : 0 ≤ L / s + 1 + Real.log s := by positivity
369        have hKB0 : 0 ≤ KB P (c₀ * ((y : ZMod s)⁻¹ - (y' : ZMod s)⁻¹)) := by
370          unfold KB; exact Real.rpow_nonneg (KB_base_nonneg P _) _
371        have := mul_nonneg (mul_nonneg hA2 hL0) h2
372        calc ‖α y‖ * ‖α y'‖ * (KB P (c₀ * ((y : ZMod s)⁻¹ - (y' : ZMod s)⁻¹)) *
373              (L / s + 1 + Real.log s))
374            ≤ A ^ 2 * (1 / ((y : ℝ) * y')) * ((3 : ℝ) ^ ((P.card : ℝ) / 4) *
375                (s : ℝ) ^ ((3 : ℝ) / 4) * G *
376                (∏ p ∈ P.filter (fun p : ℕ => (p : ℤ) ∣ (y : ℤ) - y'), (p : ℝ)) ^ ((1 : ℝ) / 4) *
377                (L / s + 1 + Real.log s)) :=
378              mul_le_mul hprod (mul_le_mul_of_nonneg_right hKB hLf) (by positivity) (by positivity)
379          _ = A ^ 2 * Kc * ((1 / ((y : ℝ) * y')) *
380              (∏ p ∈ P.filter (fun p : ℕ => (p : ℤ) ∣ (y : ℤ) - y'), (p : ℝ)) ^ ((1 : ℝ) / 4)) := by
381              rw [hKc]; ring
382          _ ≤ _ := by linarith
383  -- sum the pointwise bound
384  calc _ ≤ ∑ y ∈ 𝒴, ∑ y' ∈ 𝒴, (T1 y y' + A ^ 2 * L * T2 y y' + A ^ 2 * Kc * T3 y y') :=
385        sum_le_sum fun y hy => sum_le_sum fun y' hy' => hpt y hy y' hy'
386    _ = ∑ y ∈ 𝒴, ∑ y' ∈ 𝒴, T1 y y' + A ^ 2 * L * ∑ y ∈ 𝒴, ∑ y' ∈ 𝒴, T2 y y' +
387          A ^ 2 * Kc * ∑ y ∈ 𝒴, ∑ y' ∈ 𝒴, T3 y y' := by
388        simp only [sum_add_distrib, mul_sum]
389    _ ≤ L * ∑ y ∈ 𝒴, ‖α y‖ ^ 2 + A ^ 2 * L * (2 * (1 + Real.log Y) ^ 2 / s) +
390          A ^ 2 * Kc * (2 * (1 + Real.log Y) ^ 2 * ∏ p ∈ P, (1 + (p : ℝ) ^ (-(3 : ℝ) / 4))) := by
391        have e1 : ∑ y ∈ 𝒴, ∑ y' ∈ 𝒴, T1 y y' = L * ∑ y ∈ 𝒴, ‖α y‖ ^ 2 := by
392          rw [mul_sum]; refine sum_congr rfl fun y hy => ?_
393          simp only [hT1]; rw [sum_ite_eq 𝒴 y (fun _ => L * ‖α y‖ ^ 2), if_pos hy]
394        have e2 : ∑ y ∈ 𝒴, ∑ y' ∈ 𝒴, T2 y y' ≤ 2 * (1 + Real.log Y) ^ 2 / s := by
395          calc _ ≤ ∑ y ∈ Icc 1 Y, ∑ y' ∈ Icc 1 Y, T2 y y' := by
396                refine (sum_le_sum fun y _ => sum_le_sum_of_subset_of_nonneg h𝒴
397                  fun y' _ _ => hT2n y y').trans ?_
398                exact sum_le_sum_of_subset_of_nonneg h𝒴 fun y _ _ => sum_nonneg fun y' _ => hT2n y y'
399            _ = ∑ y ∈ Icc 1 Y, ∑ y' ∈ (Icc 1 Y).filter (fun y' => y ≠ y' ∧
400                  ((s : ℕ) : ℤ) ∣ (y : ℤ) - y'), (1 / ((y : ℝ) * y')) := by
401                refine sum_congr rfl fun y _ => ?_
402                rw [sum_filter]
403            _ ≤ _ := gcd_pair_sum_sym s Y (by omega)
404        have e3 : ∑ y ∈ 𝒴, ∑ y' ∈ 𝒴, T3 y y' ≤
405            2 * (1 + Real.log Y) ^ 2 * ∏ p ∈ P, (1 + (p : ℝ) ^ (-(3 : ℝ) / 4)) := by
406          calc _ ≤ ∑ y ∈ Icc 1 Y, ∑ y' ∈ Icc 1 Y, T3 y y' := by
407                refine (sum_le_sum fun y _ => sum_le_sum_of_subset_of_nonneg h𝒴
408                  fun y' _ _ => hT3n y y').trans ?_
409                exact sum_le_sum_of_subset_of_nonneg h𝒴 fun y _ _ => sum_nonneg fun y' _ => hT3n y y'
410            _ = ∑ y ∈ Icc 1 Y, ∑ y' ∈ (Icc 1 Y).filter (fun y' => y ≠ y'),
411                  (1 / ((y : ℝ) * y')) *
412                    (∏ p ∈ P.filter (fun p : ℕ => (p : ℤ) ∣ (y : ℤ) - y'), (p : ℝ)) ^
413                      ((1 : ℝ) / 4) := by
414                refine sum_congr rfl fun y _ => ?_
415                rw [sum_filter]
416            _ ≤ _ := gcd_moment P hP Y
417        rw [e1]
418        have hAL : 0 ≤ A ^ 2 * L := by positivity
419        have hAK : 0 ≤ A ^ 2 * Kc := by positivity
420        have := mul_le_mul_of_nonneg_left e2 hAL
421        have := mul_le_mul_of_nonneg_left e3 hAK
422        linarith
423    _ = _ := by rw [hKc]; ring
