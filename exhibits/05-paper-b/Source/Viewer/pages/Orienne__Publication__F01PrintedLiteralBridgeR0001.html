  108theorem printed_pair_local_positive_high (p h : ℕ)
  109    (hp : 3 ≤ p) (hh : 0 < h) :
  110    0 < printedLocal p ({0,h} : Finset ℕ) := by

  114      (Finset.image (fun x : ℕ => x % p) ({0,h} : Finset ℕ)).card ≤
  115          ({0,h} : Finset ℕ).card := Finset.card_image_le
  116      _ = 2 := card_pair_zero_of_ne_zero (Nat.ne_of_gt hh)
  117  have hnum : 0 < (1 : ℝ) -
  118      (occupiedResidues p ({0,h} : Finset ℕ)).card / (p : ℝ) := by
  119    have hpR : (2 : ℝ) < (p : ℝ) := by exact_mod_cast (show 2 < p by omega)
  120    have hnR : ((occupiedResidues p ({0,h} : Finset ℕ)).card : ℝ) ≤ 2 := by
  121      exact_mod_cast hcard
  122    have hpR0 : (0 : ℝ) < (p : ℝ) := by positivity
  123    apply sub_pos.mpr
  124    apply (div_lt_one hpR0).mpr

  203/-- All nonexceptional triple factors agree with the pair factor times
  204the native occupied anchored local ratio. The missing low primes are finite. -/
  205theorem printed_triple_local_quotient (p h y : ℕ)
  206    (hy0 : 0 < y) (hyh : y < h) (hp : 5 ≤ p) :
  207    printedLocal p ({0,y,h} : Finset ℕ) =
  208      printedLocal p ({0,h} : Finset ℕ) *
  209        occupiedAnchoredLocalFactorRatio p h y := by
  210  have hp0 : (p : ℝ) ≠ 0 := by exact_mod_cast (show p ≠ 0 by omega)
  211  have hp1 : (1 : ℝ) - 1 / (p : ℝ) ≠ 0 := by
  212    have : (1 : ℝ) < (p : ℝ) := by exact_mod_cast (show 1 < p by omega)
  213    exact ne_of_gt (sub_pos.mpr ((div_lt_one (by positivity)).mpr this))
  214  have hpair : ({0,h} : Finset ℕ).card = 2 :=
  215    card_pair_zero_of_ne_zero (Nat.ne_of_gt (by omega))
  216  have htriple : ({0,y,h} : Finset ℕ).card = 3 := by
  217    have h01 : (0 : ℕ) ≠ y := Nat.ne_of_lt hy0
  218    have h0h : (0 : ℕ) ≠ h := Nat.ne_of_lt (by omega)
  219    have hyh' : y ≠ h := Nat.ne_of_lt hyh
  220    simp [h01, h0h, hyh']
  221  have hden : (1 : ℝ) -
  222      (occupiedResidues p ({0,h} : Finset ℕ)).card / (p : ℝ) ≠ 0 := by

  228    have hpr : (2 : ℝ) < (p : ℝ) := by exact_mod_cast (show 2 < p by omega)
  229    exact ne_of_gt (sub_pos.mpr ((div_lt_one (by positivity)).mpr (by linarith)))
  230  unfold printedLocal occupiedAnchoredLocalFactorRatio
  231  rw [hpair, htriple, ← occupied_pair_card, ← occupied_triple_card]
  232  exact normalized_triple_factor _ _ _ hp1 hden
