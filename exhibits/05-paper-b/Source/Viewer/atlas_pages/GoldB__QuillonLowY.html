9* `quillon_shifted_lowY : ShiftedSuzukiLowY`. If `N ≥ 4096` and
10  `log Y < (3/2)(log N + log 2)`, then `Y < (2N)^{3/2} ≤ N²/2`. The reciprocal phase
11  `Y/(m+ϑ)` over `N < m ≤ U ≤ 2N` is then in the Kusmin–Landau regime. Quillon's proved
12  `GoldC.VK.recipSum_small_T` (window `[L,2L]`, `L = N+1`, `T = Y/L ≤ L/2`) gives
13  `‖S‖ ≤ 8L²/Y ≤ 16 N²/Y`. The `N²/Y` term of the target dominates, and no mean value theorem is
14  used.
15* `shiftedSuzukiCore_unconditional : ShiftedSuzukiCore`. This joins the low-`Y` branch with the
16  archived `GoldB.shiftedSuzukiCore_of_lowY`.
17-/

25/-- The `ℕ`-indexed clip sum as a GoldC run sum starting at `N+1`. -/
26lemma shiftedModeSum_eq_expSum (Y ϑ : ℝ) (N U : ℕ) :
27    shiftedModeSum 1 Y ϑ 0 N U =
28      GoldC.VK.expSum (GoldC.VK.recipPhase Y ϑ) ((N + 1 : ℕ) : ℤ) (U - N) := by
29  unfold shiftedModeSum GoldC.VK.expSum
30  rw [← Finset.Ico_add_one_add_one_eq_Ioc, Finset.sum_Ico_eq_sum_range,
31    show U + 1 - (N + 1) = U - N by omega]
32  apply sum_congr rfl; intro i _
33  unfold shiftedExpMode GoldC.VK.eAdd GoldC.VK.recipPhase
34  congr 1; push_cast; ring

37theorem quillon_shifted_lowY : ShiftedSuzukiLowY := by
38  refine ⟨1, 16, one_pos, by norm_num, ?_⟩
39  intro Y hY N hN ϑ hϑ0 hϑ1 U hU hlow
40  have hNr : (4096 : ℝ) ≤ N := by exact_mod_cast hN
41  have hN0 : (0 : ℝ) < N := by linarith
42  have hY0 : 0 < Y := by linarith
43  have hYlt : Y < (N : ℝ) ^ 2 / 2 := by
44    have h32 : Real.log 32 ≤ Real.log N := Real.log_le_log (by norm_num) (by linarith)
45    have e32 : Real.log 32 = 5 * Real.log 2 := by
46      rw [show (32 : ℝ) = 2 ^ 5 by norm_num, Real.log_pow]; norm_num
47    have eN : Real.log ((N : ℝ) ^ 2 / 2) = 2 * Real.log N - Real.log 2 := by
48      rw [Real.log_div (by positivity) (by norm_num), Real.log_pow]; push_cast; ring
49    have : Real.log Y < Real.log ((N : ℝ) ^ 2 / 2) := by rw [eN]; linarith
50    exact (Real.log_lt_log_iff hY0 (by positivity)).mp this
51  have hR0 : 0 ≤ 16 * ((N : ℝ) * Real.exp (-(1 * Real.log (N : ℝ) ^ 3 / Real.log Y ^ 2)) +
52      (N : ℝ) ^ 2 / Y) := by positivity
53  rcases le_or_gt U N with hUN | hUN
54  · have : shiftedModeSum 1 Y ϑ 0 N U = 0 := by
55      unfold shiftedModeSum; rw [Finset.Ioc_eq_empty (by omega)]; simp
56    rw [this, norm_zero]; exact hR0
57  set L : ℝ := (N : ℝ) + 1 with hLdef
58  have hL : 0 < L := by positivity
59  set T : ℝ := Y / L with hTdef
60  have hT : 0 < T := by positivity
61  have hTL : T ≤ L / 2 := by
62    simp only [T]; rw [div_le_iff₀ hL]; nlinarith
63  have hwin : ∀ x ∈ GoldC.VK.runIcc ((N + 1 : ℕ) : ℤ) (U - N), L ≤ x + ϑ ∧ x + ϑ ≤ 2 * L := by
64    intro x hx
65    simp only [GoldC.VK.runIcc, Set.mem_Icc] at hx
66    have hUN' : ((U - N : ℕ) : ℝ) = (U : ℝ) - N := by rw [Nat.cast_sub hUN.le]
67    rw [hUN'] at hx
68    push_cast at hx
69    have hUr : (U : ℝ) ≤ 2 * N := by exact_mod_cast hU
70    constructor <;> simp only [L] <;> linarith [hx.1, hx.2]
71  have hS := GoldC.VK.recipSum_small_T (θ := ϑ) ((N + 1 : ℕ) : ℤ) (U - N) hL hT hTL hwin
72  have hTL' : T * L = Y := by simp only [T]; field_simp
73  rw [hTL'] at hS
74  rw [shiftedModeSum_eq_expSum]
75  refine hS.trans ?_
76  have e : 8 * L / T = 8 * L ^ 2 / Y := by simp only [T]; field_simp
77  rw [e]
78  have hLsq : 8 * L ^ 2 ≤ 16 * (N : ℝ) ^ 2 := by simp only [L]; nlinarith
79  have h1 : 8 * L ^ 2 / Y ≤ 16 * ((N : ℝ) ^ 2 / Y) := by
80    rw [mul_div_assoc']; exact div_le_div_of_nonneg_right hLsq hY0.le

82    positivity
