21/-- Exact admissible interior count in the translated variable `x = a + t`.
22
23The frozen paper writes
24`I_q(a;h) = {0 < t < h : gcd(a+t,q)=1}`.
25Translation by `a` is a bijection from those `t` to the filtered interval
26`a < x < a+h`, so this is the same finite cardinality without introducing
27residue-class quotient types in M2.
28-/
29def admissibleInteriorCount (q a h : ℕ) : ℕ :=
30  ((Finset.Ioo a (a + h)).filter fun x => q.Coprime x).card
31
32/-- A positive canonical representative of the gap class `b-a (mod q)`.
33For residue zero we choose `q` itself rather than `0`, because the paper's
34height representatives are positive.
35-/
36def positiveGapRepresentative (q a b : ℕ) : ℕ :=
37  let r := (b % q + q - a % q) % q
38  if r = 0 then q else r
39
40/-- The canonical representative is positive whenever the modulus is positive.
41-/
42theorem positiveGapRepresentative_pos
43    {q a b : ℕ} (hq : 0 < q) :
44    0 < positiveGapRepresentative q a b := by
45  dsimp [positiveGapRepresentative]
46  split_ifs with hr
47  · exact hq
48  · exact Nat.pos_of_ne_zero hr

50/-- Every complete block of `q` consecutive translated positions contributes
51exactly `phi(q)` coprime residues.
52-/
53theorem completePeriodCount (q start : ℕ) :
54    ((Finset.Ico start (start + q)).filter fun x => q.Coprime x).card =
55      Nat.totient q := by
56  simpa using Nat.filter_coprime_Ico_eq_totient q start

58/-- Increasing a positive height by one full modulus adds exactly one complete
59period and hence `phi(q)` admissible interior positions.
60-/
61theorem admissibleInteriorCount_add_period
62    (q a h : ℕ) (hh : 0 < h) :
63    admissibleInteriorCount q a (h + q) =
64      admissibleInteriorCount q a h + Nat.totient q := by
65  unfold admissibleInteriorCount
66  have hUnion :
67      Finset.Ioo a (a + (h + q)) =
68        Finset.Ioo a (a + h) ∪ Finset.Ico (a + h) ((a + h) + q) := by
69    ext x
70    simp
71    omega
72  have hDisjoint :
73      Disjoint
74        ((Finset.Ioo a (a + h)).filter fun x => q.Coprime x)
75        ((Finset.Ico (a + h) ((a + h) + q)).filter fun x => q.Coprime x) := by
76    rw [Finset.disjoint_left]
77    intro x hx₁ hx₂
78    simp only [Finset.mem_filter, Finset.mem_Ioo, Finset.mem_Ico] at hx₁ hx₂
79    omega
80  rw [hUnion, Finset.filter_union, Finset.card_union_of_disjoint hDisjoint]
81  rw [completePeriodCount q (a + h)]

83/-- Iteration of the complete-period increment.
84-/
85theorem admissibleInteriorCount_progression
86    (q a h0 n : ℕ) (hh0 : 0 < h0) :
87    admissibleInteriorCount q a (h0 + n * q) =
88      admissibleInteriorCount q a h0 + n * Nat.totient q := by
89  induction n with
90  | zero => simp
91  | succ n ih =>
92      calc
93        admissibleInteriorCount q a (h0 + Nat.succ n * q) =
94            admissibleInteriorCount q a ((h0 + n * q) + q) := by
95              rw [Nat.succ_mul, ← Nat.add_assoc]
96        _ = admissibleInteriorCount q a (h0 + n * q) + Nat.totient q := by
97              exact admissibleInteriorCount_add_period q a (h0 + n * q) (by omega)
98        _ = (admissibleInteriorCount q a h0 + n * Nat.totient q) +
99              Nat.totient q := by rw [ih]
100        _ = admissibleInteriorCount q a h0 + Nat.succ n * Nat.totient q := by
101              simpa [Nat.succ_mul, Nat.add_assoc]

103/-- Frozen-paper periodic defect `epsilon_q(a,b)`, using the canonical positive
104representative of the compatible gap class.
105-/
106def periodicError (q a b : ℕ) : ℚ :=
107  let h0 := positiveGapRepresentative q a b
108  (admissibleInteriorCount q a h0 : ℚ) -
109    (Nat.totient q : ℚ) / q * h0

111/-- Frozen paper `lem:Nq` / `eq:Nq-periodic`.
112
113For every positive modulus and fixed residue data `(a,b)`, every positive
114height on the canonical compatible progression has the exact form
115
116`N_q(a;h) = phi(q)/q * h + epsilon_q(a,b)`.
117
118No squarefreeness or analytic hypothesis is needed for this counting identity.
119-/
120theorem exactPeriodicCount
121    {q a b : ℕ} (hq : 0 < q) (n : ℕ) :
122    (admissibleInteriorCount q a
123        (positiveGapRepresentative q a b + n * q) : ℚ) =
124      (Nat.totient q : ℚ) / q *
125          (positiveGapRepresentative q a b + n * q) +
126        periodicError q a b := by
127  let h0 := positiveGapRepresentative q a b
128  have hh0 : 0 < h0 := by
129    dsimp [h0]
130    exact positiveGapRepresentative_pos hq
131  rw [show positiveGapRepresentative q a b = h0 by rfl]
132  rw [admissibleInteriorCount_progression q a h0 n hh0]
133  unfold periodicError
134  rw [show positiveGapRepresentative q a b = h0 by rfl]
135  push_cast
136  have hq0 : (q : ℚ) ≠ 0 :=
137    Nat.cast_ne_zero.mpr (Nat.ne_of_gt hq)
138  field_simp [hq0]
139  ring
