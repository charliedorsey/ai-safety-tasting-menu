570/-- The LONG-left region-C part at decoration `d`: `s^{1+δ} < r` and `B` misses `[X, ⌊s^{u₀}⌋]`. -/
571noncomputable def CLPart (Jlow Jhigh q a h : ℕ) (δ : ℝ) (d : Finset ℕ) : ℝ :=
572  ∑ B ∈ (Tsrc q h d).powerset, ∑ C ∈ (Tsrc q h d).powerset,
573    if B ≠ ∅ ∧ C ≠ ∅ ∧ Disjoint B C ∧
574        (MkL Jlow Jhigh h (h / primeSetProduct d) δ B C ∧
575          (longInverseLeft δ (primeSetProduct B) (primeSetProduct C) ∧ ¬ WMask h δ (Tsrc q h d) B C)) then
576      omegaWeight B * omegaWeight C *
577        foldRow q a (primeSetProduct d) (h / primeSetProduct d) (primeSetProduct B) (primeSetProduct C)
578    else 0
