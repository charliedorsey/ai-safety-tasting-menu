  540theorem RL_physical (q : ℕ) (hq : 0 < q) (δ : ℝ) (hδ0 : 0 < δ) (hδ1 : δ < 1 / 3) :
  541    ∃ Λ₀ : ℝ, 2 ≤ Λ₀ ∧ ∀ (Jlow Jhigh a h : ℕ) (d : Finset ℕ), Λ₀ ≤ Real.log h → 2 ≤ h →
  542      |RLPart Jlow Jhigh q a h δ d| ≤ (Real.log h) ^ (-(75 : ℝ)) := by
  543  obtain ⟨Λ₁, hΛ₁, hRf⟩ := R_final_L q hq δ hδ0 hδ1
  544  set E : ℝ := Real.exp 10 * (2 * 2 + 1) ^ 10 with hE
  545  obtain ⟨Λ₂, hΛ₂, hKΛ⟩ := const_le_rpow (4 * E) 1 one_pos
  546  obtain ⟨Λ₃, hΛ₃, h3ε'⟩ := εp3_le_ev δ hδ0
  547  refine ⟨max Λ₁ (max Λ₂ Λ₃), le_trans hΛ₁ (le_max_left _ _), ?_⟩
  548  intro Jlow Jhigh a h d hΛ hh
  549  have hΛa : Λ₁ ≤ Real.log h := le_trans (le_max_left _ _) hΛ
  550  have hΛb : Λ₂ ≤ Real.log h := le_trans (le_trans (le_max_left _ _) (le_max_right _ _)) hΛ
  551  have h3ε : 3 * εp h ≤ δ := h3ε' h (le_trans (le_trans (le_max_right _ _) (le_max_right _ _)) hΛ)
  552  set Λ := Real.log h with hΛdef
  553  have hΛ2 : 2 ≤ Λ := le_trans hΛ₁ hΛa
  554  have hΛ0 : 0 < Λ := by linarith
  555  have hK := hKΛ Λ hΛb
  556  rw [Real.rpow_one] at hK
  557  set H := h / primeSetProduct d with hH
  558  set T := Tsrc q h d with hT
  559  have hTf := Tsrc_facts q h d
  560  have hHh : H ≤ h := Nat.div_le_self _ _
  561  unfold RLPart
  562  rw [GoldC.QuillonCells.double_sum_fixB]
  563  have hinner : ∀ B ∈ T.powerset.filter (fun B => B ≠ ∅),
  564      |∑ C ∈ (T \ B).powerset.filter (fun C => C ≠ ∅),
  565        (if MkL Jlow Jhigh h H δ B C ∧
  566            (¬ longInverseLeft δ (primeSetProduct B) (primeSetProduct C) ∧ (C ∩ JR h T B).Nonempty) then
  567          omegaWeight B * omegaWeight C *
  568            foldRow q a (primeSetProduct d) H (primeSetProduct B) (primeSetProduct C) else 0)| ≤
  569        ωR B * (4 * Λ ^ (-(90 : ℝ))) := by
  570    intro B hB
  571    have hBT : B ⊆ T := mem_powerset.1 (mem_filter.1 hB).1
  572    have hω0 : 0 ≤ ωR B := ωR_nonneg B fun p hp => (hTf p (hBT hp)).2.1
  573    have e : ∑ C ∈ (T \ B).powerset.filter (fun C => C ≠ ∅),
  574        (if MkL Jlow Jhigh h H δ B C ∧
  575            (¬ longInverseLeft δ (primeSetProduct B) (primeSetProduct C) ∧ (C ∩ JR h T B).Nonempty) then
  576          omegaWeight B * omegaWeight C *
  577            foldRow q a (primeSetProduct d) H (primeSetProduct B) (primeSetProduct C) else 0) =
  578        ωR B * ∑ C ∈ (T \ B).powerset.filter (fun C => C ≠ ∅),
  579          (if ((Jlow ≤ B.card + C.card ∧ B.card + C.card < Jhigh) ∧ GeoRL h H δ B (∏ p ∈ C, p)) ∧
  580              (C ∩ (T \ B).filter (fun p => Xp h < p ∧ p ≤ 2 ^ (Kvp h (∏ p ∈ B, p)) * Xp h)).Nonempty
  581            then ωR C * foldRow q a (primeSetProduct d) H (∏ p ∈ B, p) (∏ p ∈ C, p) else 0) := by
  582      rw [mul_sum]
  583      refine sum_congr rfl fun C hC => ?_
  584      have hCB : C ⊆ T \ B := mem_powerset.1 (mem_filter.1 hC).1
  585      have hiff : (MkL Jlow Jhigh h H δ B C ∧
  586            (¬ longInverseLeft δ (primeSetProduct B) (primeSetProduct C) ∧ (C ∩ JR h T B).Nonempty)) ↔
  587          (((Jlow ≤ B.card + C.card ∧ B.card + C.card < Jhigh) ∧ GeoRL h H δ B (∏ p ∈ C, p)) ∧
  588              (C ∩ (T \ B).filter (fun p => Xp h < p ∧ p ≤ 2 ^ (Kvp h (∏ p ∈ B, p)) * Xp h)).Nonempty) := by
  589        unfold MkL GeoRL GeoLR JR
  590        constructor
  591        · rintro ⟨⟨h1, h2, h3, h4, h5, h6⟩, hnL, hj⟩
  592          have hR : longInverseRight δ (primeSetProduct B) (primeSetProduct C) := h5.resolve_left hnL
  593          exact ⟨⟨⟨h3, h4⟩, ⟨h1, h2, hR, h6⟩, LR_eps h3ε hR⟩, hj⟩
  594        · rintro ⟨⟨⟨h3, h4⟩, ⟨h1, h2, hR, h6⟩, -⟩, hj⟩
  595          exact ⟨⟨h1, h2, h3, h4, Or.inr hR, h6⟩, fun hL => not_both δ hδ0.le _ _ ⟨hL, hR⟩, hj⟩
  596      split_ifs with h1 h2 h2
  597      · show ωR B * ωR C * foldRow q a (primeSetProduct d) H (∏ p ∈ B, p) (∏ p ∈ C, p) =
  598          ωR B * (ωR C * foldRow q a (primeSetProduct d) H (∏ p ∈ B, p) (∏ p ∈ C, p))
  599        ring
  600      · exact absurd (hiff.1 h1) h2
  601      · exact absurd (hiff.2 h2) h1
  602      · rw [mul_zero]
  603    rw [e, abs_mul, abs_of_nonneg hω0]
  604    exact mul_le_mul_of_nonneg_left (hRf a h H (primeSetProduct d) T B
  605      (fun k => Jlow ≤ B.card + k ∧ B.card + k < Jhigh) hΛa hh hHh hTf hBT) hω0
  606  calc _ ≤ ∑ B ∈ T.powerset.filter (fun B => B ≠ ∅),
  607        |∑ C ∈ (T \ B).powerset.filter (fun C => C ≠ ∅),
  608          (if MkL Jlow Jhigh h H δ B C ∧
  609              (¬ longInverseLeft δ (primeSetProduct B) (primeSetProduct C) ∧ (C ∩ JR h T B).Nonempty) then
  610            omegaWeight B * omegaWeight C *
  611              foldRow q a (primeSetProduct d) H (primeSetProduct B) (primeSetProduct C) else 0)| :=
  612        abs_sum_le_sum_abs _ _

  614    _ ≤ ∑ B ∈ T.powerset, ωR B * (4 * Λ ^ (-(90 : ℝ))) := by

  616        intro B hB _
  617        have hBT : B ⊆ T := mem_powerset.1 hB
  618        have := ωR_nonneg B fun p hp => (hTf p (hBT hp)).2.1
  619        positivity

  622        mul_le_mul_of_nonneg_right (outer_mass q h d (by linarith)) (by positivity)
  623    _ = (4 * E) * Λ ^ (-(80 : ℝ)) := by
  624        rw [show Λ ^ (-(80 : ℝ)) = Λ ^ (10 : ℝ) * Λ ^ (-(90 : ℝ)) by rw [rp hΛ0]; norm_num]; ring
  625    _ ≤ Λ * Λ ^ (-(80 : ℝ)) := mul_le_mul_of_nonneg_right hK (by positivity)

  627        rw [show Λ * Λ ^ (-(80 : ℝ)) = Λ ^ (1 : ℝ) * Λ ^ (-(80 : ℝ)) by rw [Real.rpow_one], rp hΛ0]
  628        norm_num
  629    _ ≤ Λ ^ (-(75 : ℝ)) := Real.rpow_le_rpow_of_exponent_le (by linarith) (by norm_num)
