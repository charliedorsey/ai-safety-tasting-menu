20/-- Original balanced Central source = signed folded row plus
21    the already-paid inverse and density errors, with the literal
22    quarter-parent cutoff and all source decorations unchanged. -/
23theorem activeCentral_fold_plus_paid_errors
24    (q a h : ℕ) (hq : 0 < q) :
25    activeCentral q a h =
26      regionSum q h (centralGate h) (foldRow q a) +
27        (regionSum q h (centralGate h) (invRow q a) +

34  rw [← hfold]

38    the difference from the signed folded source is bounded by
39    the already-known decaying baseRate. -/
40theorem activeCentral_fold_error_abs_le
41    (q a h : ℕ) (hq : 0 < q) (hh : 0 < h) :
42    |activeCentral q a h -
43      regionSum q h (centralGate h) (foldRow q a)| ≤
44      (6 * (q : ℝ) + 2) * baseRate h := by
45  have heq : activeCentral q a h -
46      regionSum q h (centralGate h) (foldRow q a) =
47      regionSum q h (centralGate h) (invRow q a) +
48      regionSum q h (centralGate h) (densRow q a) := by
49    rw [activeCentral_fold_plus_paid_errors q a h hq]

62/-- The quantitative source comparison gives the original full
63    signed GoldC Central rate from only the physical folded rate.
64    No assumption of signed sawtooth equidistribution is introduced
65    beyond hFold's actual weighted source estimate. -/
66theorem signedActiveCentralSublog_of_folded
67    (hFold : SourceUniformSublog
68      (fun q a h => regionSum q h (centralGate h) (foldRow q a))) :
69    SignedActiveCentralSublog := by
70  intro q hq hsq eps heps
71  let C : ℝ := 6 * (q : ℝ) + 2
72  have hC : 0 < C := by dsimp [C]; positivity
73  have hdelta : 0 < eps / (2 * C) := by positivity
74  obtain ⟨Hfold, hfold⟩ := hFold q hq hsq (eps / 2) (by positivity)
75  obtain ⟨Herr, herr⟩ := baseRate_small (eps / (2 * C)) hdelta
76  refine ⟨max Hfold Herr, ?_⟩
77  intro a h hqa hh hpar hlarge
78  have hFoldBound : |regionSum q h (centralGate h) (foldRow q a)| ≤
79      (eps / 2) * Real.log (h : ℝ) :=
80    hfold a h hqa hh hpar (le_trans (le_max_left _ _) hlarge)
81  obtain ⟨-, hLog, hRate⟩ :=
82    herr h (le_trans (le_max_right _ _) hlarge)
83  have hErr := activeCentral_fold_error_abs_le q a h hq hh
84  have hErrBound :
85      |activeCentral q a h -
86        regionSum q h (centralGate h) (foldRow q a)| ≤
87          (eps / 2) * Real.log (h : ℝ) := by
88    calc
89      _ ≤ C * baseRate h := by simpa only [C] using hErr
90      _ ≤ C * (eps / (2 * C)) :=
91        mul_le_mul_of_nonneg_left hRate hC.le
92      _ = eps / 2 := by
93        field_simp
94      _ ≤ (eps / 2) * Real.log (h : ℝ) := by
95        have he : (eps / 2) * 1 ≤ (eps / 2) * Real.log (h : ℝ) :=
96          mul_le_mul_of_nonneg_left hLog (by positivity)
97        simpa only [mul_one] using he
98  calc
99    |activeCentral q a h| =
100      |(activeCentral q a h -
101         regionSum q h (centralGate h) (foldRow q a)) +
102         regionSum q h (centralGate h) (foldRow q a)| := by
103           simp only [sub_add_cancel]
104    _ ≤ |activeCentral q a h -
105           regionSum q h (centralGate h) (foldRow q a)| +
106         |regionSum q h (centralGate h) (foldRow q a)| :=
107      abs_add_le _ _
108    _ ≤ eps * Real.log (h : ℝ) := by linarith
