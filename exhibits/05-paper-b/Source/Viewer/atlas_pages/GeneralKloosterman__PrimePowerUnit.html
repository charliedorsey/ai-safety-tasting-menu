42/-- Square roots modulo `p ^ n` split into two classes modulo `p ^ (n - e)`. -/
43theorem sq_eq_sq_split {n e : ℕ} (hsep : Sep p e) (y y0 : ZMod (p ^ n)) (hy0 : IsUnit y0)
44    (hsq : y ^ 2 = y0 ^ 2) :
45    p ^ (n - e) ∣ (y - y0).val ∨ p ^ (n - e) ∣ (y + y0).val := by
46  have hAB : (y - y0) * (y + y0) = 0 := by linear_combination hsq
47  have hdvd : p ^ n ∣ (y - y0).val * (y + y0).val := by
48    rw [← ZMod.natCast_eq_zero_iff]; push_cast
49    rw [ZMod.natCast_zmod_val, ZMod.natCast_zmod_val]; exact hAB
50  obtain ⟨k1, k2, h1, h2, hk⟩ := dvd_mul.1 hdvd
51  have hk1 : k1 ∣ p ^ n := ⟨k2, hk⟩
52  have hk2 : k2 ∣ p ^ n := ⟨k1, by rw [hk, mul_comm]⟩
53  obtain ⟨i, hi, rfl⟩ := (Nat.dvd_prime_pow hp.out).1 hk1
54  obtain ⟨l, hl, rfl⟩ := (Nat.dvd_prime_pow hp.out).1 hk2
55  have hil : n = i + l := by
56    rw [← pow_add] at hk; exact Nat.pow_right_injective hp.out.two_le hk
57  by_cases hA : n - e ≤ i
58  · exact Or.inl ((pow_dvd_pow p hA).trans h1)
59  by_cases hB : n - e ≤ l
60  · exact Or.inr ((pow_dvd_pow p hB).trans h2)
61  exfalso
62  have hei : e + 1 ≤ n := by omega
63  have hA' : p ^ (e + 1) ∣ (y - y0).val := (pow_dvd_pow p (by omega)).trans h1
64  have hB' : p ^ (e + 1) ∣ (y + y0).val := (pow_dvd_pow p (by omega)).trans h2
65  rw [← red_eq_zero_iff hei] at hA' hB'
66  have hy : IsUnit y := by
67    have : IsUnit (y ^ 2) := hsq ▸ hy0.pow 2
68    exact (isUnit_pow_iff two_ne_zero).1 this
69  obtain ⟨u, rfl⟩ := hy
70  apply hsep (red (e + 1) hei (u : ZMod (p ^ n))) (red (e + 1) hei ((u⁻¹ : (ZMod (p ^ n))ˣ) : ZMod (p ^ n)))
71    (by rw [← map_mul, Units.mul_inv, map_one])
72  calc 2 * red (e + 1) hei (u : ZMod (p ^ n))
73      = red (e + 1) hei ((u : ZMod (p ^ n)) - y0) + red (e + 1) hei ((u : ZMod (p ^ n)) + y0) := by
74        rw [← map_add, show (u : ZMod (p ^ n)) - y0 + ((u : ZMod (p ^ n)) + y0) =
75          2 * (u : ZMod (p ^ n)) by ring, map_mul, map_ofNat]
76    _ = 0 := by rw [hA', hB', add_zero]

92  have hn' : m - e ≤ m := by omega
93  let T1 : Finset (ZMod (p ^ k)) := {x | red (m - e) hn x = red (m - e) hn x0}
94  let T2 : Finset (ZMod (p ^ k)) := {x | red (m - e) hn x = red (m - e) hn (-x0)}
95  have hsub : T ⊆ T1 ∪ T2 := by
96    intro x hx
97    have hsq : red m hm x ^ 2 = red m hm x0 ^ 2 := by rw [hT x hx, hT x0 hx0]

99      have : IsUnit (red m hm x0 ^ 2) := by rw [hT x0 hx0]; exact hc
100      exact (isUnit_pow_iff two_ne_zero).1 this

102    · rw [← map_sub, red_dvd_val_iff hm hn', ← red_eq_zero_iff hn, map_sub, sub_eq_zero] at h
103      exact mem_union_left _ (mem_filter.2 ⟨mem_univ _, h⟩)
104    · rw [← map_add, red_dvd_val_iff hm hn', ← red_eq_zero_iff hn, map_add,
105        add_eq_zero_iff_eq_neg, ← map_neg] at h
106      exact mem_union_right _ (mem_filter.2 ⟨mem_univ _, h⟩)

109  have hc2 : #T2 = p ^ (k - (m - e)) := by
