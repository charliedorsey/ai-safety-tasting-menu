321  have e2 : (((z * H : ℤ) : ℝ) / r) / s = ((z * H : ℤ) : ℝ) / ((r : ℝ) * s) := by

326lemma class_phase_R (H r s M b : ℕ) (hM : 2 ≤ M) (hr : 1 ≤ r) (hs : 2 ≤ s) (hrs : r.Coprime s)
327    (hMs : M.Coprime s) (z : ℤ) :
328    eR (((z * (classN H r s M b : ℕ) : ℤ) : ℝ) / ((M * s : ℕ) : ℝ)) =
329      eR ((((z * H : ℤ) : ℝ) / ((M * r : ℕ) : ℝ)) / s) *
330        eR (-(((z * H : ℤ) : ℝ) * invM s (M * r) / ((M * r : ℕ) : ℝ))) *
331        eR ((z : ℝ) * (((b * invM r M) % M * invM s M : ℕ) : ℝ) / M) := by
332  have hMR : (M : ℝ) ≠ 0 := by
333    have : (0 : ℝ) < M := by exact_mod_cast (show 0 < M by omega)
334    exact this.ne'
335  have hsR : (s : ℝ) ≠ 0 := by
336    have : (0 : ℝ) < s := by exact_mod_cast (show 0 < s by omega)
337    exact this.ne'
338  have e : (((z * (classN H r s M b : ℕ) : ℤ) : ℝ) / ((M * s : ℕ) : ℝ)) =
339      (z : ℝ) * (((H * invM r s) % s * invM M s : ℕ) : ℝ) / s +
340        (z : ℝ) * (((b * invM r M) % M * invM s M : ℕ) : ℝ) / M := by
341    rw [Int.cast_mul, Int.cast_natCast]
342    unfold classN; push_cast; field_simp
343  rw [e, eR_add]
344  congr 1
345  have hk : (H * invM r s) % s * invM M s ≡ H * invM (M * r) s [MOD s] :=
346    GoldC.QuillonCopyC.phase_kloosterman hs hrs hMs
347  rw [eR_modEq s (by omega) _ _ hk z]
348  have hMr : 2 ≤ M * r := by nlinarith
349  have hMrs : (M * r).Coprime s := Nat.coprime_mul_iff_left.2 ⟨hMs, hrs⟩
350  have h := GoldC.QuillonCopyRec.eR_reciprocity hMr hs hMrs (z * H)
351  have e1 : (z : ℝ) * ((H * invM (M * r) s : ℕ) : ℝ) / s =
352      ((z * H : ℤ) : ℝ) * (invR (M * r) s : ℝ) / s := by
353    push_cast; unfold GoldC.QuillonCopyRec.invMod GoldC.QuillonCopyC.invMod; ring
354  have hMrR : ((M * r : ℕ) : ℝ) ≠ 0 := by positivity
355  have e2 : (((z * H : ℤ) : ℝ) / ((M * r : ℕ) : ℝ)) / s = ((z * H : ℤ) : ℝ) / (((M * r : ℕ) : ℝ) * s) := by
356    field_simp
357  rw [e1, h, e2]
358  rfl

397theorem rowAbs_eq_rowR (M H : ℕ) (μ : ℝ) (Γ : ℕ → ℕ → ℝ) (r s : ℕ) (hM : 2 ≤ M) (hr : 2 ≤ r)
398    (hs : 2 ≤ s) (hrs : r.Coprime s) (hsM : s.Coprime M) (hHs : H.Coprime s)
399    (J : ℕ) (hJ : M * s ≤ 2 * J) :
400    ((rowAbs M H μ Γ r s : ℝ) : ℂ) =
401      rowR (Icc 1 J) (fun k => ((-(2 * μ) : ℝ) : ℂ) * βk k) (fun k => k) (fun _ => 1)
402          (fun k => (((k : ℤ) * H : ℤ) : ℝ) / r) (fun k σ => ΓmR r H k σ) (M * r) s +
403        rowR (Icc 1 J) (fun k => -(((-(2 * μ) : ℝ) : ℂ) * βk k)) (fun k => k) (fun _ => 1)
404          (fun k => (((-(k : ℤ)) * H : ℤ) : ℝ) / r) (fun k σ => ΓmR r H (-(k : ℤ)) σ) (M * r) s +
405        rowR (Icc 1 J) (fun k => βk k) (fun k => k) (fun _ => M)
406          (fun k => (((k : ℤ) * H : ℤ) : ℝ) / ((M * r : ℕ) : ℝ)) (fun k σ => ΓcR M H Γ r k σ)
407          (M * r) s +
408        rowR (Icc 1 J) (fun k => -βk k) (fun k => k) (fun _ => M)
409          (fun k => (((-(k : ℤ)) * H : ℤ) : ℝ) / ((M * r : ℕ) : ℝ))
410          (fun k σ => ΓcR M H Γ r (-(k : ℤ)) σ) (M * r) s := by
411  classical
412  have hM1 : 1 ≤ M := by omega
413  have hMs : M.Coprime s := hsM.symm
414  have hsMs : s ≤ M * s := Nat.le_mul_of_pos_left s (by omega)
415  have hJs : s ≤ 2 * J := le_trans hsMs hJ
416  have hNs : ¬ s ∣ H * invM r s := fun h =>
417    t_mod_ne_zero H r s hs hrs hHs (Nat.mod_eq_zero_of_dvd h)
418  have hcls : ∀ b ∈ range M, (Γ b ((r * s) % M) : ℂ) *
419      (((sawC (GoldC.QuillonCopyC.classPhase H r s M b - 1 / M) : ℝ) : ℂ) -
420        ((sawC (GoldC.QuillonCopyC.classPhase H r s M b) : ℝ) : ℂ)) =
421      ∑ k ∈ Icc 1 J, (βk k * (((ampC k (M * s) : ℝ) : ℂ) *
422          (eR ((((k : ℤ) * H : ℤ) : ℝ) / ((M * r : ℕ) : ℝ) / s) * cbR M H Γ r b k s)) +
423        -βk k * (((ampC k (M * s) : ℝ) : ℂ) *
424          (eR ((((-(k : ℤ)) * H : ℤ) : ℝ) / ((M * r : ℕ) : ℝ) / s) *
425            cbR M H Γ r b (-(k : ℤ)) s))) := by
426    intro b _
427    rw [class_saw_eR H r s M b J hM hs hrs hHs hMs hJ, mul_sum]
428    refine sum_congr rfl fun k _ => ?_
429    rw [shift_phase _ M s hM1 (by omega), shift_phase _ M s hM1 (by omega),
430      class_phase_R H r s M b hM (by omega) hs hrs hMs, class_phase_R H r s M b hM (by omega) hs hrs hMs]
431    unfold cbR
432    rw [αm_eq]
433    ring
434  unfold rowR
435  simp only [ΓmR_mod, ΓcR_mod, one_mul]
436  unfold rowAbs
437  simp only [Complex.ofReal_add, Complex.ofReal_mul, Complex.ofReal_sum, Complex.ofReal_sub]
438  have e0 : (H : ℝ) * invM r s / s = ((H * invM r s : ℕ) : ℝ) / s := by push_cast; ring
439  rw [e0, saw_eR _ s J hs hNs hJs, sum_congr rfl hcls, sum_comm, mul_sum, ← sum_add_distrib,
440    ← sum_add_distrib, ← sum_add_distrib, ← sum_add_distrib]
441  refine sum_congr rfl fun k _ => ?_
442  rw [main_phase_R H r s hr hs hrs, main_phase_R H r s hr hs hrs, sum_add_distrib]
443  unfold ΓcR ΓmR
