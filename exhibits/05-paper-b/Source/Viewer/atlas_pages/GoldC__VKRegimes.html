348    rwa [mul_div_assoc, div_self hLpos.ne', mul_one] at h

350    (by simpa using hT2) hlen isOpen_Ioi hsub (recipPhase_contDiffOn _ _) hder

353/-- **All Type I regimes, conditional only on `UniformVinogradovEstimate`.**
354For every `A > 0` and all large `L`, every `T > 0` (with `log L ≥ (log T)^σ` required only when
355`T ≥ L`), every shift and every run in the window `x+θ ∈ [L, 2L]`:
356`|Σ e(TL/(x+θ))| ≤ L (log L)^{-A} + 8L/T`. -/
357theorem recipSum_all_regimes {σ : ℝ} (hσ : 2 / 3 < σ) (hV : UniformVinogradovEstimate)
358    (A : ℝ) (hA : 0 < A) :
359    ∃ L₁ : ℝ, ∀ (L T θ : ℝ) (a : ℤ) (n : ℕ),
360      L₁ ≤ L → 0 < T → (L ≤ T → Real.log T ^ σ ≤ Real.log L) →
361      (∀ x ∈ runIcc a n, L ≤ x + θ ∧ x + θ ≤ 2 * L) →
362      ‖expSum (recipPhase (T * L) θ) a n‖ ≤ L * Real.log L ^ (-A) + 8 * L / T := by
363  obtain ⟨T₀, hT₀⟩ := reciprocalPhaseVK_of_uniform hσ hV A hA
364  obtain ⟨c, C, hc, hC, hmid⟩ := recipSum_middle_T hV
365  obtain ⟨y₀, hy₀1, hy₀⟩ := eventually_const_mul_exp_neg_le C c 1 A hc one_pos
366  refine ⟨max (max T₀ 2) (Real.exp y₀), fun L T θ a n hL hT hσL hwin => ?_⟩
367  have hL2 : 2 ≤ L := ((le_max_right _ _).trans (le_max_left _ _)).trans hL
368  have hLT₀ : T₀ ≤ L := ((le_max_left _ _).trans (le_max_left _ _)).trans hL
369  have hLpos : 0 < L := by linarith
370  have hlogL : y₀ ≤ Real.log L := by
371    rw [Real.le_log_iff_exp_le hLpos]; exact (le_max_right _ _).trans hL
372  have hlogLpos : 0 < Real.log L := lt_of_lt_of_le (by linarith) hlogL
373  have hpow_nn : 0 ≤ L * Real.log L ^ (-A) := mul_nonneg hLpos.le (Real.rpow_nonneg hlogLpos.le _)
374  have h8 : 0 ≤ 8 * L / T := div_nonneg (by linarith) hT.le
375  rcases le_or_gt T (L / 2) with hsmall | hbig
376  · have := recipSum_small_T a n hLpos hT hsmall hwin
377    linarith
378  rcases le_or_gt T L with hmidT | hlarge
379  · have hm := hmid L T θ a n hL2 hbig hmidT hwin
380    have hy := hy₀ (Real.log L) hlogL
381    have hrp : L ^ (-c) = Real.exp (-(c * Real.log L ^ (1 : ℝ))) := by
382      rw [Real.rpow_one, Real.rpow_def_of_pos hLpos]; ring_nf
383    calc ‖expSum (recipPhase (T * L) θ) a n‖ ≤ C * L * L ^ (-c) := hm
384      _ = L * (C * Real.exp (-(c * Real.log L ^ (1 : ℝ)))) := by rw [hrp]; ring
385      _ ≤ L * Real.log L ^ (-A) := mul_le_mul_of_nonneg_left hy hLpos.le
386      _ ≤ L * Real.log L ^ (-A) + 8 * L / T := by linarith
387  · have hvk := hT₀ L T θ a n (hLT₀.trans hlarge.le) hL2 hlarge.le (hσL hlarge.le) hwin
388    have hmono : Real.log T ^ (-A) ≤ Real.log L ^ (-A) :=
389      Real.rpow_le_rpow_of_nonpos hlogLpos (Real.log_le_log hLpos hlarge.le) (by linarith)
390    calc ‖expSum (recipPhase (T * L) θ) a n‖ ≤ L * Real.log T ^ (-A) := hvk
391      _ ≤ L * Real.log L ^ (-A) := mul_le_mul_of_nonneg_left hmono hLpos.le
392      _ ≤ L * Real.log L ^ (-A) + 8 * L / T := by linarith

394/-- **Harmonic-weighted Type I class sum, all regimes, conditional only on the uniform
395Vinogradov estimate:** `|Σ_j e(TL/(j+θ)) / (Q(j+θ))| ≤ ((log L)^{-A} + 8/T)/Q`. -/
396theorem typeI_class_sum_all_regimes {σ : ℝ} (hσ : 2 / 3 < σ) (hV : UniformVinogradovEstimate)
397    (A : ℝ) (hA : 0 < A) :
398    ∃ L₁ : ℝ, ∀ (L T θ Q : ℝ) (a : ℤ) (n : ℕ),
399      L₁ ≤ L → 0 < T → (L ≤ T → Real.log T ^ σ ≤ Real.log L) → 0 < Q →
400      (∀ x ∈ runIcc a n, L ≤ x + θ ∧ x + θ ≤ 2 * L) →
401      ‖∑ i ∈ range n, ((1 / (Q * ((a : ℝ) + i + θ)) : ℝ) : ℂ) *
402          eAdd (recipPhase (T * L) θ ((a : ℝ) + i))‖ ≤
403        (Real.log L ^ (-A) + 8 / T) / Q := by
404  obtain ⟨L₁, hL₁⟩ := recipSum_all_regimes hσ hV A hA
405  refine ⟨max L₁ 1, fun L T θ Q a n hL hT hσL hQ hwin => ?_⟩
406  have hLpos : 0 < L := lt_of_lt_of_le one_pos ((le_max_right _ _).trans hL)
407  have hprefix : ∀ m ≤ n, ∀ x ∈ runIcc a m, L ≤ x + θ ∧ x + θ ≤ 2 * L := by
408    intro m hm x hx
409    apply hwin
410    refine ⟨hx.1, hx.2.trans ?_⟩
411    have : (m : ℝ) ≤ n := by exact_mod_cast hm
412    linarith
413  by_cases hn : n = 0
414  · subst hn
415    simp only [range_zero, sum_empty, norm_zero]
416    have : 0 ≤ Real.log L ^ (-A) := Real.rpow_nonneg (Real.log_nonneg
417      ((le_max_right _ _).trans hL)) _
418    exact div_nonneg (add_nonneg this (div_nonneg (by norm_num) hT.le)) hQ.le
419  have haθ : L ≤ (a : ℝ) + θ := by
420    have ha : (a : ℝ) ∈ runIcc a n := by
421      refine ⟨le_refl _, ?_⟩
422      have : (1 : ℝ) ≤ n := by exact_mod_cast Nat.one_le_iff_ne_zero.2 hn
423      linarith
424    exact (hwin _ ha).1
425  have hpos : ∀ i : ℕ, 0 < (a : ℝ) + i + θ := by
426    intro i; have : (0 : ℝ) ≤ i := Nat.cast_nonneg i; linarith
427  set w : ℕ → ℝ := fun i => 1 / (Q * ((a : ℝ) + i + θ)) with hw
428  have hw0 : ∀ i, 0 ≤ w i := fun i => by
429    simp only [hw]; exact div_nonneg zero_le_one (mul_pos hQ (hpos i)).le
430  have hwm : ∀ i, w (i + 1) ≤ w i := fun i => by
431    simp only [hw]
432    apply one_div_le_one_div_of_le (mul_pos hQ (hpos i))
433    apply mul_le_mul_of_nonneg_left _ hQ.le
434    push_cast; linarith
435  have hB : ∀ m ≤ n, ‖∑ i ∈ range m, eAdd (recipPhase (T * L) θ ((a : ℝ) + i))‖ ≤
436      L * Real.log L ^ (-A) + 8 * L / T := fun m hm =>
437    hL₁ L T θ a m ((le_max_left _ _).trans hL) hT hσL (hprefix m hm)
438  have habel := abel_antitone (z := fun i => eAdd (recipPhase (T * L) θ ((a : ℝ) + i)))
439    (w := w) n hw0 hwm hB
440  have hw0le : w 0 ≤ 1 / (Q * L) := by
441    simp only [hw]
442    apply one_div_le_one_div_of_le (mul_pos hQ hLpos)
443    apply mul_le_mul_of_nonneg_left _ hQ.le
444    simpa using haθ
445  have hBnn : 0 ≤ L * Real.log L ^ (-A) + 8 * L / T := by
446    have : 0 ≤ Real.log L ^ (-A) := Real.rpow_nonneg (Real.log_nonneg
447      ((le_max_right _ _).trans hL)) _
448    exact add_nonneg (mul_nonneg hLpos.le this) (div_nonneg (by linarith) hT.le)
449  calc ‖∑ i ∈ range n, ((1 / (Q * ((a : ℝ) + i + θ)) : ℝ) : ℂ) *
450          eAdd (recipPhase (T * L) θ ((a : ℝ) + i))‖
451      = ‖∑ i ∈ range n, (w i : ℂ) * eAdd (recipPhase (T * L) θ ((a : ℝ) + i))‖ := rfl
452    _ ≤ w 0 * (L * Real.log L ^ (-A) + 8 * L / T) := habel
453    _ ≤ 1 / (Q * L) * (L * Real.log L ^ (-A) + 8 * L / T) :=
454        mul_le_mul_of_nonneg_right hw0le hBnn
455    _ = (Real.log L ^ (-A) + 8 / T) / Q := by field_simp
