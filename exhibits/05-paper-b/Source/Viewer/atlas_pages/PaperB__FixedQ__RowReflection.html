48/-- Reflection for row discrepancy when the reflected weight agrees pointwise. -/
49theorem rowDiscrepancy_reflect
50    (lambda lambdaR : ℕ → ℝ) (H e1 e2 : ℕ)
51    (hreflect : ∀ z ∈ Finset.Ioo 0 H, lambda (H - z) = lambdaR z) :
52    rowDiscrepancy lambda H e1 e2 =
53      rowDiscrepancy lambdaR H e2 e1 := by
54  unfold rowDiscrepancy

56      rowConstrainedMass lambda H e1 e2 =
