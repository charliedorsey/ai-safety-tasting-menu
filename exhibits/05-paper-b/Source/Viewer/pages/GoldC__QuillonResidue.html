    6# Quillon CSC3 F7.8a: rough-number masses for the residues (crude sieve only)
    7

  308/-- The mass of a condition splits into the smooth part times the rough condition. -/
  309theorem split_mass (T : Finset ℕ) (hT : ∀ p ∈ T, 5 ≤ p) (X : ℕ) (cond cond₂ : Finset ℕ → Prop)
  310    [DecidablePred cond] [DecidablePred cond₂]
  311    (h : ∀ B₁ ⊆ T.filter (· < X), ∀ B₂ ⊆ T.filter (X ≤ ·), cond (B₁ ∪ B₂) → cond₂ B₂) :
  312    ∑ B ∈ T.powerset, (if cond B then ωR B else 0) ≤
  313      (∑ B₁ ∈ (T.filter (· < X)).powerset, ωR B₁) *
  314        ∑ B₂ ∈ (T.filter (X ≤ ·)).powerset, (if cond₂ B₂ then ωR B₂ else 0) := by
  315  classical
  316  rw [powerset_split_sum T X, sum_mul_sum]
  317  refine sum_le_sum fun B₁ hB₁ => sum_le_sum fun B₂ hB₂ => ?_
  318  have hB₁T := mem_powerset.1 hB₁
  319  have hB₂T := mem_powerset.1 hB₂
  320  have hdisj : Disjoint B₁ B₂ := by
  321    rw [disjoint_left]; intro p h1 h2
  322    have := (mem_filter.1 (hB₁T h1)).2; have := (mem_filter.1 (hB₂T h2)).2; omega
  323  have hω1 := ωR_nonneg B₁ fun p hp => hT p (filter_subset _ _ (hB₁T hp))
  324  have hω2 := ωR_nonneg B₂ fun p hp => hT p (filter_subset _ _ (hB₂T hp))
  325  by_cases hc : cond (B₁ ∪ B₂)
  326  · rw [if_pos hc, if_pos (h B₁ hB₁T B₂ hB₂T hc), ωR_union B₁ B₂ hdisj]
  327  · rw [if_neg hc]; split_ifs <;> positivity

  366/-- `Σ_{n∈(a,b]} 1/n ≤ 1 + log(b/a)` for `1 ≤ a ≤ b`. -/
  367lemma harmonic_add_Ioc (a : ℕ) : ∀ d, ((harmonic (a + d) : ℚ) : ℝ) =
  368    ((harmonic a : ℚ) : ℝ) + ∑ n ∈ Ioc a (a + d), (1 / (n : ℝ))
  369  | 0 => by simp
  370  | d + 1 => by
  371    have hI : Ioc a (a + (d + 1)) = insert (a + d + 1) (Ioc a (a + d)) := by
  372      ext n; simp only [mem_Ioc, mem_insert]; omega
  373    have hnot : a + d + 1 ∉ Ioc a (a + d) := by simp
  374    rw [hI, sum_insert hnot, show a + (d + 1) = (a + d) + 1 by ring, harmonic_succ, Rat.cast_add,
  375      harmonic_add_Ioc a d]
  376    push_cast; ring

  378/-- `Σ_{n∈(a,b]} 1/n ≤ 1 + log(b/a)` for `1 ≤ a ≤ b`. -/
  379lemma sum_Ioc_inv_le (a b : ℕ) (ha : 1 ≤ a) (hab : a ≤ b) :
  380    ∑ n ∈ Ioc a b, (1 / (n : ℝ)) ≤ 1 + Real.log ((b : ℝ) / a) := by

  388    Real.log_le_log haR (by push_cast; linarith)
  389  rw [Real.log_div hbR.ne' haR.ne']
  390  linarith

  392/-- **Interval mass**: `Σ_{C, ∏C ∈ (a,b]} ω(C) ≤ Aω(b)·(1 + log(b/a))`. -/
  393theorem interval_mass (T : Finset ℕ) (hT : ∀ p ∈ T, p.Prime ∧ 5 ≤ p) (a b : ℕ) (ha : 1 ≤ a)
  394    (hab : a ≤ b) :
  395    ∑ C ∈ T.powerset, (if (∏ p ∈ C, p) ∈ Ioc a b then ωR C else 0) ≤
  396      Real.exp 37.5 * (1 + Nat.log 2 (Nat.log 2 b) : ℝ) ^ 30 * (1 + Real.log ((b : ℝ) / a)) := by
  397  classical
  398  set Aω : ℝ := Real.exp 37.5 * (1 + Nat.log 2 (Nat.log 2 b) : ℝ) ^ 30 with hAω
  399  have hTp : ∀ p ∈ T, p.Prime := fun p hp => (hT p hp).1
  400  have hpt : ∀ C ∈ T.powerset, (if (∏ p ∈ C, p) ∈ Ioc a b then ωR C else 0) ≤
  401      (if (∏ p ∈ C, p) ∈ Ioc a b then Aω * (1 / ((∏ p ∈ C, p : ℕ) : ℝ)) else 0) := by
  402    intro C hC
  403    split_ifs with h
  404    · have hCT := mem_powerset.1 hC
  405      have := GoldC.QuillonOmega.omega_pointwise C (fun p hp => hT p (hCT hp)) b (mem_Ioc.1 h).2
  406      unfold ωR; rw [Nat.cast_prod, mul_one_div]; exact this
  407    · exact le_rfl
  408  refine (sum_le_sum hpt).trans ?_
  409  rw [← sum_filter]
  410  have hinj := prod_injOn T hTp
  411  rw [show ∑ C ∈ T.powerset.filter (fun C => (∏ p ∈ C, p) ∈ Ioc a b),
  412      Aω * (1 / ((∏ p ∈ C, p : ℕ) : ℝ)) =
  413      ∑ n ∈ (T.powerset.filter (fun C => (∏ p ∈ C, p) ∈ Ioc a b)).image (fun C => ∏ p ∈ C, p),
  414        Aω * (1 / (n : ℝ)) by
  415    rw [sum_image (fun C₁ h₁ C₂ h₂ heq => hinj (mem_filter.1 h₁).1 (mem_filter.1 h₂).1 heq)]]
  416  rw [← mul_sum]
  417  apply mul_le_mul_of_nonneg_left _ (by positivity)
  418  refine (sum_le_sum_of_subset_of_nonneg ?_ fun _ _ _ => by positivity).trans (sum_Ioc_inv_le a b ha hab)
  419  intro n hn
  420  obtain ⟨C, hC, rfl⟩ := mem_image.1 hn
  421  exact (mem_filter.1 hC).2

  423/-- **Total rough mass**: subsets `B ⊆ T` with `R₀ ≤ ∏B ≤ N₁` and no prime in `[X, (∏B)^u]`. -/
  424theorem rough_total (T : Finset ℕ) (hT : ∀ p ∈ T, p.Prime ∧ 5 ≤ p) (X : ℕ) (u : ℝ) (hu0 : 0 < u)
  425    (hu1 : u ≤ 1) (R₀ N₁ k₀ K I : ℕ) (hk₀ : 3 ≤ k₀) (hR₀ : (2 : ℝ) ^ k₀ ≤ (R₀ : ℝ) ^ u)
  426    (hXK : ∀ p ∈ T, p < X → p < 2 ^ (K + 1)) (hNI : N₁ < 2 ^ (I + 1)) :
  427    ∑ B ∈ T.powerset, (if R₀ ≤ ∏ p ∈ B, p ∧ ∏ p ∈ B, p ≤ N₁ ∧
  428        ∀ p ∈ B, p < X ∨ ((∏ q ∈ B, q : ℕ) : ℝ) ^ u < p then ωR B else 0) ≤
  429      Real.exp 10 * ((K : ℝ) + 1) ^ 10 *
  430        (1 + Real.exp (3 * (Nat.log 2 N₁ : ℝ) / ((2 : ℝ) ^ k₀ - 3)) * Real.exp 4 * 2916 *
  431          (1 + Real.log ((I : ℝ) + 1)) / u ^ 5) := by
  432  classical
  433  set κ := Real.exp (3 * (Nat.log 2 N₁ : ℝ) / ((2 : ℝ) ^ k₀ - 3)) with hκ
  434  have hT5 : ∀ p ∈ T, 5 ≤ p := fun p hp => (hT p hp).2
  435  set T₂' := (T.filter (X ≤ ·)).filter (fun p => 2 ^ k₀ ≤ p) with hT₂'
  436  set cond₂ : Finset ℕ → Prop := fun B₂ => B₂ = ∅ ∨ (B₂ ⊆ T₂' ∧ B₂.Nonempty ∧ ∏ p ∈ B₂, p ≤ N₁ ∧
  437      ∀ p ∈ B₂, ((∏ q ∈ B₂, q : ℕ) : ℝ) ^ u < p) with hcond₂
  438  have hsplit := split_mass T hT5 X (fun B => R₀ ≤ ∏ p ∈ B, p ∧ ∏ p ∈ B, p ≤ N₁ ∧
  439      ∀ p ∈ B, p < X ∨ ((∏ q ∈ B, q : ℕ) : ℝ) ^ u < p) cond₂ (by
  440    intro B₁ hB₁ B₂ hB₂ hc
  441    rcases B₂.eq_empty_or_nonempty with h0 | hne
  442    · exact Or.inl h0
  443    · right
  444      have hdisj : Disjoint B₁ B₂ := by
  445        rw [disjoint_left]; intro p h1 h2
  446        have := (mem_filter.1 (hB₁ h1)).2; have := (mem_filter.1 (hB₂ h2)).2; omega
  447      have hprod : ∏ p ∈ B₁ ∪ B₂, p = (∏ p ∈ B₁, p) * ∏ p ∈ B₂, p := prod_union hdisj
  448      have hB₁pos : 0 < ∏ p ∈ B₁, p := prod_pos fun p hp => (hT p (filter_subset _ _ (hB₁ hp))).1.pos
  449      have hle : ∏ p ∈ B₂, p ≤ ∏ p ∈ B₁ ∪ B₂, p := by
  450        rw [hprod]; exact Nat.le_mul_of_pos_left _ hB₁pos
  451      have hlarge : ∀ p ∈ B₂, ((∏ q ∈ B₁ ∪ B₂, q : ℕ) : ℝ) ^ u < p := by
  452        intro p hp
  453        rcases hc.2.2 p (mem_union_right _ hp) with h | h
  454        · have := (mem_filter.1 (hB₂ hp)).2; omega
  455        · exact h
  456      refine ⟨fun p hp => ?_, hne, hle.trans hc.2.1, fun p hp => ?_⟩
  457      · rw [hT₂', mem_filter]
  458        refine ⟨hB₂ hp, ?_⟩
  459        have h1 := hlarge p hp
  460        have h2 : (R₀ : ℝ) ^ u ≤ ((∏ q ∈ B₁ ∪ B₂, q : ℕ) : ℝ) ^ u :=
  461          Real.rpow_le_rpow (Nat.cast_nonneg _) (by exact_mod_cast hc.1) hu0.le
  462        have : ((2 ^ k₀ : ℕ) : ℝ) < p := by push_cast; linarith
  463        exact_mod_cast this.le
  464      · have h1 := hlarge p hp
  465        have h2 : ((∏ q ∈ B₂, q : ℕ) : ℝ) ^ u ≤ ((∏ q ∈ B₁ ∪ B₂, q : ℕ) : ℝ) ^ u :=
  466          Real.rpow_le_rpow (Nat.cast_nonneg _) (by exact_mod_cast hle) hu0.le
  467        linarith)
  468  refine hsplit.trans ?_
  469  have hsm := smooth_mass (T.filter (· < X)) (fun p hp => hT p (filter_subset _ _ hp)) K
  470    (fun p hp => hXK p (filter_subset _ _ hp) (mem_filter.1 hp).2)
  471  have hsm0 : 0 ≤ ∑ B₁ ∈ (T.filter (· < X)).powerset, ωR B₁ :=
  472    sum_nonneg fun B hB => ωR_nonneg B fun p hp => hT5 p (filter_subset _ _ ((mem_powerset.1 hB) hp))
  473  have hk8 : (8 : ℝ) ≤ (2 : ℝ) ^ k₀ := by
  474    have : (2 : ℝ) ^ 3 ≤ (2 : ℝ) ^ k₀ := pow_le_pow_right₀ (by norm_num) hk₀
  475    norm_num at this; linarith
  476  have hκ0 : 0 ≤ κ := (Real.exp_pos _).le
  477  -- the rough factor
  478  have hrough : ∑ B₂ ∈ (T.filter (X ≤ ·)).powerset, (if cond₂ B₂ then ωR B₂ else 0) ≤
  479      1 + κ * Real.exp 4 * 2916 * (1 + Real.log ((I : ℝ) + 1)) / u ^ 5 := by
  480    have hemp : ∅ ∈ (T.filter (X ≤ ·)).powerset := empty_mem_powerset _
  481    rw [← add_sum_erase _ _ hemp]
  482    have h0 : (if cond₂ ∅ then ωR ∅ else 0) = 1 := by
  483      rw [if_pos (Or.inl rfl)]; simp [ωR]
  484    rw [h0]
  485    refine add_le_add le_rfl ?_
  486    have hT₂'p : ∀ p ∈ T₂', p.Prime := fun p hp => (hT p (filter_subset _ _ (filter_subset _ _ hp))).1
  487    have hsr := selfrough_mass T₂' hT₂'p u hu0 hu1 k₀ hk₀ (fun p hp => (mem_filter.1 hp).2) N₁ I hNI κ hκ0
  488      (by
  489        intro B hB hBN
  490        have hz : 4 ≤ 2 ^ k₀ := le_trans (by norm_num) (Nat.pow_le_pow_right (by norm_num) hk₀)
  491        refine (omega_mul_prod_le B (2 ^ k₀) hz fun p hp => (mem_filter.1 (hB hp)).2).trans ?_
  492        apply Real.exp_le_exp.2
  493        have hcard : B.card ≤ Nat.log 2 N₁ := by
  494          rcases Nat.eq_zero_or_pos B.card with h | h
  495          · omega
  496          · apply Nat.le_log_of_pow_le (by norm_num)
  497            calc 2 ^ B.card = ∏ _p ∈ B, 2 := by rw [prod_const]
  498              _ ≤ ∏ p ∈ B, p := prod_le_prod' fun p hp => (hT₂'p p (hB hp)).two_le
  499              _ ≤ N₁ := hBN
  500        have hcR : (B.card : ℝ) ≤ Nat.log 2 N₁ := by exact_mod_cast hcard
  501        push_cast
  502        apply div_le_div_of_nonneg_right _ (by linarith)
  503        linarith)
  504    refine le_trans ?_ hsr
  505    -- terms on the erased powerset are supported on `T₂'`
  506    calc ∑ B₂ ∈ ((T.filter (X ≤ ·)).powerset).erase ∅, (if cond₂ B₂ then ωR B₂ else 0)
  507        ≤ ∑ B₂ ∈ (T.filter (X ≤ ·)).powerset, (if B₂ ⊆ T₂' ∧ (B₂.Nonempty ∧ ∏ p ∈ B₂, p ≤ N₁ ∧
  508            ∀ p ∈ B₂, ((∏ q ∈ B₂, q : ℕ) : ℝ) ^ u < p) then ωR B₂ else 0) := by
  509          refine (sum_le_sum fun B₂ hB₂ => ?_).trans (sum_le_sum_of_subset_of_nonneg (erase_subset _ _)
  510            fun B₂ hB₂ _ => ?_)
  511          · have hne : B₂ ≠ ∅ := ne_of_mem_erase hB₂
  512            by_cases hc : cond₂ B₂
  513            · rcases hc with h | h
  514              · exact absurd h hne
  515              · rw [if_pos (Or.inr h), if_pos h]
  516            · rw [if_neg hc]; split_ifs
  517              · exact ωR_nonneg B₂ fun p hp => hT5 p (filter_subset _ _
  518                  ((mem_powerset.1 (mem_of_mem_erase hB₂)) hp))
  519              · exact le_rfl
  520          · split_ifs
  521            · exact ωR_nonneg B₂ fun p hp => hT5 p (filter_subset _ _ ((mem_powerset.1 hB₂) hp))
  522            · exact le_rfl
  523      _ = ∑ B₂ ∈ T₂'.powerset, (if B₂.Nonempty ∧ ∏ p ∈ B₂, p ≤ N₁ ∧
  524            ∀ p ∈ B₂, ((∏ q ∈ B₂, q : ℕ) : ℝ) ^ u < p then ωR B₂ else 0) := by
  525          rw [← sum_filter, ← sum_filter]
  526          congr 1
  527          ext B₂
  528          simp only [mem_filter, mem_powerset]
  529          constructor
  530          · rintro ⟨_, h1, h2⟩; exact ⟨h1, h2⟩
  531          · rintro ⟨h1, h2⟩; exact ⟨h1.trans (filter_subset _ _), h1, h2⟩
  532  have hr0 : 0 ≤ ∑ B₂ ∈ (T.filter (X ≤ ·)).powerset, (if cond₂ B₂ then ωR B₂ else 0) :=
  533    sum_nonneg fun B₂ hB₂ => by
  534      split_ifs
  535      · exact ωR_nonneg B₂ fun p hp => hT5 p (filter_subset _ _ ((mem_powerset.1 hB₂) hp))
  536      · exact le_rfl
  537  exact mul_le_mul hsm hrough hr0 (by positivity)
