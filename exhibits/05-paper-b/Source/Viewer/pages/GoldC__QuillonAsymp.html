    5/-!
    6# Quillon CSC3 F7.9a: asymptotic toolkit (number of prime factors, gcd mode sums)
    7
    8* `card_le_y`: for a set `C` of primes and any `y > 1`, `|C| ≤ y + 1 + log(∏C)/log y`.
    9* `four_pow_card_le`: for every `η > 0` there is `L₀` such that `log ∏C ≥ L₀` implies
   10  `4^{|C|} ≤ (∏C)^η`. This is the bound `ω(s) = o(log s)` in the form the routes use.
   11* `gcd_mode_sum`: `Σ_{j≤J} (1/j)·(∏_{p∈C, p|j} p)^a ≤ (1 + log J)·∏_{p∈C}(1 + p^{a−1})` for `0 ≤ a ≤ 1`.
   12
   13No axiom, no `sorry`.
   14-/

   21lemma card_le_y (C : Finset ℕ) (hC : ∀ p ∈ C, p.Prime) (y : ℝ) (hy : 1 < y) :
   22    (C.card : ℝ) ≤ y + 1 + Real.log ((∏ p ∈ C, p : ℕ) : ℝ) / Real.log y := by
   23  classical
   24  set Cs := C.filter (fun p : ℕ => (p : ℝ) < y) with hCs
   25  set Cb := C.filter (fun p : ℕ => ¬ (p : ℝ) < y) with hCb
   26  have hsplit : C.card = Cs.card + Cb.card := (card_filter_add_card_filter_not _).symm
   27  -- small primes
   28  have hs : (Cs.card : ℝ) ≤ y + 1 := by
   29    have hsub : Cs ⊆ range ⌈y⌉₊ := by
   30      intro p hp
   31      rw [hCs, mem_filter] at hp
   32      rw [mem_range]
   33      exact Nat.cast_lt.1 (lt_of_lt_of_le hp.2 (Nat.le_ceil y))
   34    have := card_le_card hsub
   35    rw [card_range] at this
   36    have h2 : (⌈y⌉₊ : ℝ) ≤ y + 1 := (Nat.ceil_lt_add_one (by linarith)).le
   37    exact (Nat.cast_le.2 this).trans h2
   38  -- large primes
   39  have hlogy : 0 < Real.log y := Real.log_pos hy
   40  have hb : (Cb.card : ℝ) ≤ Real.log ((∏ p ∈ C, p : ℕ) : ℝ) / Real.log y := by
   41    rw [le_div_iff₀ hlogy]
   42    have hpos : ∀ p ∈ C, (0 : ℝ) < p := fun p hp => by exact_mod_cast (hC p hp).pos
   43    have h1 : y ^ Cb.card ≤ ∏ p ∈ Cb, (p : ℝ) := by
   44      rw [← prod_const]
   45      refine prod_le_prod₀ (fun _ _ => by linarith) fun p hp => ?_
   46      rw [hCb, mem_filter] at hp; push Not at hp; exact hp.2
   47    have h2 : ∏ p ∈ Cb, (p : ℝ) ≤ ∏ p ∈ C, (p : ℝ) := by
   48      rw [← prod_filter_mul_prod_filter_not C (fun p : ℕ => ¬ (p : ℝ) < y)]
   49      have h1' : (1 : ℝ) ≤ ∏ p ∈ C.filter (fun p : ℕ => ¬ ¬ (p : ℝ) < y), (p : ℝ) :=
   50        one_le_prod₀ fun p hp => by exact_mod_cast (hC p (filter_subset _ _ hp)).one_le
   51      have h0 : 0 ≤ ∏ p ∈ Cb, (p : ℝ) := prod_nonneg fun p _ => Nat.cast_nonneg _
   52      rw [hCb] at h0 ⊢
   53      nlinarith
   54    have h3 : Real.log (y ^ Cb.card) ≤ Real.log ((∏ p ∈ C, p : ℕ) : ℝ) := by
   55      apply Real.log_le_log (by positivity)
   56      push_cast; linarith
   57    rw [Real.log_pow] at h3
   58    linarith
   59  rw [hsplit, Nat.cast_add]; linarith

   61/-- `ω(s) = o(log s)`, in the form `4^{ω(s)} ≤ s^η`. -/
   62theorem four_pow_card_le (η : ℝ) (hη : 0 < η) :
   63    ∃ L₀ : ℝ, 0 < L₀ ∧ ∀ C : Finset ℕ, (∀ p ∈ C, p.Prime) →
   64      L₀ ≤ Real.log ((∏ p ∈ C, p : ℕ) : ℝ) →
   65        (4 : ℝ) ^ C.card ≤ ((∏ p ∈ C, p : ℕ) : ℝ) ^ η := by
   66  set c := Real.log 4 with hc
   67  have hc0 : 0 < c := Real.log_pos (by norm_num)
   68  set L₀ := max 16 (max ((8 * c / η) ^ 2) (Real.exp (8 * c / η))) with hL₀
   69  refine ⟨L₀, by positivity, fun C hC hL => ?_⟩
   70  set ℓ := Real.log ((∏ p ∈ C, p : ℕ) : ℝ) with hℓ
   71  have hℓ16 : 16 ≤ ℓ := le_trans (le_max_left _ _) hL
   72  have hℓsq : (8 * c / η) ^ 2 ≤ ℓ := le_trans (le_trans (le_max_left _ _) (le_max_right _ _)) hL
   73  have hℓexp : Real.exp (8 * c / η) ≤ ℓ := le_trans (le_trans (le_max_right _ _) (le_max_right _ _)) hL
   74  have hsqrt : 4 ≤ Real.sqrt ℓ := by
   75    rw [show (4 : ℝ) = Real.sqrt 16 by rw [show (16 : ℝ) = 4 ^ 2 by norm_num, Real.sqrt_sq (by norm_num)]]
   76    exact Real.sqrt_le_sqrt hℓ16
   77  have hcard := card_le_y C hC (Real.sqrt ℓ) (by linarith)
   78  have hlogsqrt : Real.log (Real.sqrt ℓ) = Real.log ℓ / 2 := by
   79    rw [Real.log_sqrt (by linarith)]
   80  have hlogℓ : 8 * c / η ≤ Real.log ℓ := by
   81    rw [← Real.log_exp (8 * c / η)]; exact Real.log_le_log (Real.exp_pos _) hℓexp
   82  have hlogℓpos : 0 < Real.log ℓ := lt_of_lt_of_le (by positivity) hlogℓ
   83  have hsq2 : 8 * c / η ≤ Real.sqrt ℓ := by
   84    rw [← Real.sqrt_sq (by positivity : 0 ≤ 8 * c / η)]; exact Real.sqrt_le_sqrt hℓsq
   85  have hsqℓ : Real.sqrt ℓ * Real.sqrt ℓ = ℓ := Real.mul_self_sqrt (by linarith)
   86  -- `|C|·c ≤ η ℓ`
   87  have hkey : (C.card : ℝ) * c ≤ η * ℓ := by
   88    rw [hlogsqrt] at hcard
   89    have e1 : ℓ / (Real.log ℓ / 2) = 2 * ℓ / Real.log ℓ := by field_simp
   90    rw [e1] at hcard
   91    have t1 : (Real.sqrt ℓ + 1) * c ≤ η * ℓ / 2 := by
   92      have : (Real.sqrt ℓ + 1) * c ≤ 2 * Real.sqrt ℓ * c := by nlinarith
   93      have h2 : 2 * Real.sqrt ℓ * c ≤ η * ℓ / 2 := by
   94        have h8 : 8 * c ≤ η * Real.sqrt ℓ := by
   95          have := hsq2; rw [div_le_iff₀ hη] at this; linarith
   96        have hs0 : 0 ≤ Real.sqrt ℓ := Real.sqrt_nonneg _
   97        nlinarith
   98      linarith
   99    have t2 : 2 * ℓ / Real.log ℓ * c ≤ η * ℓ / 2 := by
  100      rw [div_mul_eq_mul_div, div_le_iff₀ hlogℓpos]
  101      have := mul_le_mul_of_nonneg_left hlogℓ (show 0 ≤ η * ℓ / 2 by positivity)
  102      have h4 : η * ℓ / 2 * (8 * c / η) = 4 * c * ℓ := by field_simp; ring
  103      nlinarith
  104    nlinarith
  105  have hprod_pos : (0 : ℝ) < ((∏ p ∈ C, p : ℕ) : ℝ) := by
  106    have : 0 < ∏ p ∈ C, p := prod_pos fun p hp => (hC p hp).pos
  107    exact_mod_cast this
  108  rw [← Real.exp_log hprod_pos, ← Real.exp_mul, ← hℓ]
  109  rw [show (4 : ℝ) ^ C.card = Real.exp (C.card * c) by
  110    rw [hc, mul_comm, Real.exp_mul, Real.exp_log (by norm_num)]; norm_cast]
  111  exact Real.exp_le_exp.2 (by linarith)

  113/-- A set of distinct primes divides `j` iff each of them does. -/
  114lemma prod_dvd_iff (D : Finset ℕ) (hD : ∀ p ∈ D, p.Prime) (j : ℕ) :
  115    (∏ p ∈ D, p) ∣ j ↔ ∀ p ∈ D, p ∣ j := by
  116  constructor
  117  · intro h p hp; exact (dvd_prod_of_mem _ hp).trans h
  118  · intro h
  119    exact Finset.prod_primes_dvd j (fun p hp => (hD p hp).prime) h

  121/-- **The gcd mode sum.** -/
  122theorem gcd_mode_sum (C : Finset ℕ) (hC : ∀ p ∈ C, p.Prime) (J : ℕ) (a : ℝ) (ha0 : 0 ≤ a)
  123    (ha1 : a ≤ 1) :
  124    ∑ j ∈ Icc 1 J, (1 / (j : ℝ)) * (((∏ p ∈ C.filter (fun p => p ∣ j), p : ℕ) : ℝ) ^ a) ≤
  125      (1 + Real.log J) * ∏ p ∈ C, (1 + (p : ℝ) ^ (a - 1)) := by
  126  classical
  127  set w : Finset ℕ → ℝ := fun D => ∏ p ∈ D, ((p : ℝ) ^ a - 1) with hw
  128  have hw0 : ∀ D ⊆ C, 0 ≤ w D := fun D hD => prod_nonneg fun p hp => by
  129    have : (1 : ℝ) ≤ (p : ℝ) ^ a := Real.one_le_rpow (by exact_mod_cast (hC p (hD hp)).one_le) ha0
  130    linarith
  131  have hexp : ∀ j, (((∏ p ∈ C.filter (fun p => p ∣ j), p : ℕ) : ℝ) ^ a) =
  132      ∑ D ∈ C.powerset, (if (∏ p ∈ D, p) ∣ j then w D else 0) := by
  133    intro j
  134    rw [Nat.cast_prod, ← Real.finset_prod_rpow _ _ (fun p _ => Nat.cast_nonneg p)]
  135    have h1 : ∏ p ∈ C.filter (fun p => p ∣ j), (p : ℝ) ^ a =
  136        ∏ p ∈ C.filter (fun p => p ∣ j), (1 + ((p : ℝ) ^ a - 1)) :=
  137      prod_congr rfl fun p _ => by ring
  138    rw [h1, prod_one_add]
  139    rw [← sum_filter]
  140    apply sum_congr
  141    · ext D
  142      simp only [mem_powerset, mem_filter]
  143      constructor
  144      · intro hD
  145        exact ⟨hD.trans (filter_subset _ _), (prod_dvd_iff D (fun p hp => hC p ((hD.trans
  146          (filter_subset _ _)) hp)) j).2 fun p hp => (mem_filter.1 (hD hp)).2⟩
  147      · rintro ⟨hD, hdvd⟩ p hp
  148        exact mem_filter.2 ⟨hD hp, (prod_dvd_iff D (fun p hp => hC p (hD hp)) j).1 hdvd p hp⟩
  149    · intro D _; rfl
  150  have hJ : Icc 1 J = Ioc 0 J := by ext j; simp only [mem_Icc, mem_Ioc]; omega
  151  calc ∑ j ∈ Icc 1 J, (1 / (j : ℝ)) * (((∏ p ∈ C.filter (fun p => p ∣ j), p : ℕ) : ℝ) ^ a)
  152      = ∑ D ∈ C.powerset, w D * ∑ j ∈ (Ioc 0 J).filter (fun j => (∏ p ∈ D, p) ∣ j), (1 / (j : ℝ)) := by
  153        simp only [hexp, mul_sum]
  154        rw [hJ, sum_comm]
  155        refine sum_congr rfl fun D _ => ?_
  156        rw [sum_filter]
  157        refine sum_congr rfl fun j _ => ?_
  158        split_ifs <;> ring
  159    _ ≤ ∑ D ∈ C.powerset, w D * ((1 / ((∏ p ∈ D, p : ℕ) : ℝ)) * (1 + Real.log J)) := by
  160        refine sum_le_sum fun D hD => mul_le_mul_of_nonneg_left ?_ (hw0 D (mem_powerset.1 hD))
  161        exact GoldC.QuillonDivisorMoment.sum_inv_multiples _ J (Nat.one_le_iff_ne_zero.2 (prod_ne_zero_iff.2 fun p hp =>
  162          (hC p ((mem_powerset.1 hD) hp)).ne_zero))
  163    _ ≤ ∑ D ∈ C.powerset, (∏ p ∈ D, (p : ℝ) ^ (a - 1)) * (1 + Real.log J) := by
  164        refine sum_le_sum fun D hD => ?_
  165        have hDC := mem_powerset.1 hD
  166        have hlog : 0 ≤ 1 + Real.log J := GoldC.QuillonDivisorMoment.one_add_log_nonneg J
  167        rw [← mul_assoc]
  168        apply mul_le_mul_of_nonneg_right _ hlog
  169        rw [Nat.cast_prod, one_div, ← prod_inv_distrib]
  170        simp only [hw]
  171        rw [← prod_mul_distrib]
  172        refine prod_le_prod₀ (fun p hp => ?_) fun p hp => ?_
  173        · have : (1 : ℝ) ≤ (p : ℝ) ^ a := Real.one_le_rpow (by exact_mod_cast (hC p (hDC hp)).one_le) ha0
  174          have : (0 : ℝ) < p := by exact_mod_cast (hC p (hDC hp)).pos
  175          have : 0 ≤ (p : ℝ) ^ a - 1 := by linarith
  176          positivity
  177        · have hp0 : (0 : ℝ) < p := by exact_mod_cast (hC p (hDC hp)).pos
  178          rw [Real.rpow_sub hp0, Real.rpow_one, div_eq_mul_inv]
  179          apply mul_le_mul_of_nonneg_right _ (by positivity)
  180          linarith
  181    _ = (1 + Real.log J) * ∏ p ∈ C, (1 + (p : ℝ) ^ (a - 1)) := by
  182        rw [← sum_mul, mul_comm, prod_one_add]

  184/-- `b^{ω(s)} ≤ s^η` for large `log s` (any base `b ≥ 2`). -/
  185theorem base_pow_card_le (b : ℝ) (hb : 2 ≤ b) (η : ℝ) (hη : 0 < η) :
  186    ∃ L₀ : ℝ, 0 < L₀ ∧ ∀ C : Finset ℕ, (∀ p ∈ C, p.Prime) →
  187      L₀ ≤ Real.log ((∏ p ∈ C, p : ℕ) : ℝ) →
  188        b ^ C.card ≤ ((∏ p ∈ C, p : ℕ) : ℝ) ^ η := by
  189  -- reduce to base 4: `b ≤ 4^{t}` with `t = log b / log 4`
  190  set t := Real.log b / Real.log 4 with ht
  191  have hl4 : 0 < Real.log 4 := Real.log_pos (by norm_num)
  192  have hlb : 0 < Real.log b := Real.log_pos (by linarith)
  193  have ht0 : 0 < t := div_pos hlb hl4
  194  obtain ⟨L₀, hL₀, hL⟩ := four_pow_card_le (η / t) (div_pos hη ht0)
  195  refine ⟨L₀, hL₀, fun C hC hLC => ?_⟩
  196  have h1 := hL C hC hLC
  197  have hb4 : b = (4 : ℝ) ^ t := by
  198    rw [ht, Real.rpow_def_of_pos (by norm_num),
  199      show Real.log 4 * (Real.log b / Real.log 4) = Real.log b by field_simp,
  200      Real.exp_log (by linarith)]
  201  have hprod_pos : (0 : ℝ) < ((∏ p ∈ C, p : ℕ) : ℝ) := by
  202    have : 0 < ∏ p ∈ C, p := prod_pos fun p hp => (hC p hp).pos
  203    exact_mod_cast this
  204  calc b ^ C.card = ((4 : ℝ) ^ C.card) ^ t := by
  205        rw [hb4, ← Real.rpow_natCast, ← Real.rpow_natCast, ← Real.rpow_mul (by norm_num),
  206          ← Real.rpow_mul (by norm_num), mul_comm]
  207    _ ≤ (((∏ p ∈ C, p : ℕ) : ℝ) ^ (η / t)) ^ t :=
  208        Real.rpow_le_rpow (by positivity) h1 ht0.le
  209    _ = ((∏ p ∈ C, p : ℕ) : ℝ) ^ η := by
  210        rw [← Real.rpow_mul hprod_pos.le, div_mul_cancel₀ _ ht0.ne']

  212/-- `(1 + log x)^m ≤ x^c` for large `x`. -/
  213theorem one_add_log_pow_le (m : ℕ) (c : ℝ) (hc : 0 < c) :
  214    ∃ x₀ : ℝ, 1 ≤ x₀ ∧ ∀ x : ℝ, x₀ ≤ x → (1 + Real.log x) ^ m ≤ x ^ c := by
  215  rcases Nat.eq_zero_or_pos m with hm | hm
  216  · refine ⟨1, le_rfl, fun x hx => ?_⟩
  217    rw [hm, pow_zero]; exact Real.one_le_rpow hx hc.le
  218  set η := c / (2 * m) with hη
  219  have hmR : (0 : ℝ) < m := by exact_mod_cast hm
  220  have hη0 : 0 < η := by positivity
  221  set K := (1 + 1 / η) ^ m with hK
  222  have hK1 : 1 ≤ K := one_le_pow₀ (by have : 0 < 1 / η := by positivity
  223                                      linarith)
  224  refine ⟨max 1 (K ^ (2 / c)), le_max_left _ _, fun x hx => ?_⟩
  225  have hx1 : 1 ≤ x := le_trans (le_max_left _ _) hx
  226  have hx0 : 0 < x := by linarith
  227  have hxη : 1 ≤ x ^ η := Real.one_le_rpow hx1 hη0.le
  228  have hlog : 1 + Real.log x ≤ (1 + 1 / η) * x ^ η := by
  229    have := Real.log_le_rpow_div hx0.le hη0
  230    have : x ^ η / η = (1 / η) * x ^ η := by ring
  231    nlinarith
  232  have hlog0 : 0 ≤ 1 + Real.log x := by have := Real.log_nonneg hx1; linarith
  233  have h1 : (1 + Real.log x) ^ m ≤ K * x ^ (c / 2) := by
  234    calc (1 + Real.log x) ^ m ≤ ((1 + 1 / η) * x ^ η) ^ m := pow_le_pow_left₀ hlog0 hlog m
  235      _ = K * (x ^ η) ^ m := by rw [mul_pow]
  236      _ = K * x ^ (c / 2) := by
  237          rw [← Real.rpow_natCast, ← Real.rpow_mul hx0.le]; congr 2; rw [hη]; field_simp
  238  have h2 : K ≤ x ^ (c / 2) := by
  239    have hx2 : K ^ (2 / c) ≤ x := le_trans (le_max_right _ _) hx
  240    calc K = (K ^ (2 / c)) ^ (c / 2) := by
  241          rw [← Real.rpow_mul (by linarith)]; field_simp; simp
  242      _ ≤ x ^ (c / 2) := Real.rpow_le_rpow (by positivity) hx2 (by positivity)
  243  calc (1 + Real.log x) ^ m ≤ K * x ^ (c / 2) := h1
  244    _ ≤ x ^ (c / 2) * x ^ (c / 2) := mul_le_mul_of_nonneg_right h2 (by positivity)
  245    _ = x ^ c := by rw [← Real.rpow_add hx0]; ring_nf

  247/-- `Λ^k ≤ exp(c Λ^σ)` for large `Λ`. -/
  248theorem rpow_le_exp_rpow (k c σ : ℝ) (hk : 0 ≤ k) (hc : 0 < c) (hσ : 0 < σ) :
  249    ∃ Λ₀ : ℝ, 1 ≤ Λ₀ ∧ ∀ Λ : ℝ, Λ₀ ≤ Λ → Λ ^ k ≤ Real.exp (c * Λ ^ σ) := by
  250  set τ := σ / 2 with hτ
  251  have hτ0 : 0 < τ := by positivity
  252  -- `k log Λ ≤ (k/τ) Λ^τ ≤ c Λ^σ` once `Λ^τ ≥ k/(τ c)`
  253  set A := k / (τ * c) with hA
  254  refine ⟨max 1 ((A + 1) ^ (1 / τ)), le_max_left _ _, fun Λ hΛ => ?_⟩
  255  have hΛ1 : 1 ≤ Λ := le_trans (le_max_left _ _) hΛ
  256  have hΛ0 : 0 < Λ := by linarith
  257  rw [← Real.exp_log (Real.rpow_pos_of_pos hΛ0 k), Real.log_rpow hΛ0]
  258  apply Real.exp_le_exp.2
  259  have hlog := Real.log_le_rpow_div hΛ0.le hτ0
  260  have hΛτ : A + 1 ≤ Λ ^ τ := by
  261    have h1 : (A + 1) ^ (1 / τ) ≤ Λ := le_trans (le_max_right _ _) hΛ
  262    have hA0 : 0 ≤ A := by positivity
  263    calc A + 1 = ((A + 1) ^ (1 / τ)) ^ τ := by
  264          rw [← Real.rpow_mul (by linarith)]; field_simp; simp
  265      _ ≤ Λ ^ τ := Real.rpow_le_rpow (by positivity) h1 hτ0.le
  266  have hσ2 : Λ ^ σ = Λ ^ τ * Λ ^ τ := by
  267    rw [← Real.rpow_add hΛ0]; congr 1; rw [hτ]; ring
  268  have hΛτ0 : 0 < Λ ^ τ := Real.rpow_pos_of_pos hΛ0 τ
  269  calc k * Real.log Λ ≤ k * (Λ ^ τ / τ) := mul_le_mul_of_nonneg_left hlog hk
  270    _ = (k / τ) * Λ ^ τ := by ring
  271    _ ≤ (c * Λ ^ τ) * Λ ^ τ := by
  272        apply mul_le_mul_of_nonneg_right _ hΛτ0.le
  273        have : k / τ = c * A := by rw [hA]; field_simp
  274        rw [this]; nlinarith
  275    _ = c * Λ ^ σ := by rw [hσ2]; ring
