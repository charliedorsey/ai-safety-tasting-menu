    4# Quillon LONG L5: the Weil-strength W′ supplier (squarefree modulus, coupled range)
    5
    6**The single external input.** `WeilHyp` is, character for character, the statement of the
    7already-GREEN `Orienne.CSC3.norm_quillon_K_weil` (Weil–Estermann for the complete Quillon
    8Kloosterman sum `GoldC.QuillonKl.K`, general modulus, full gcd conductor). It is a *named
    9hypothesis*, never an axiom: every theorem that needs it takes `(hW : WeilHyp)` explicitly.
   10
   11From it, for squarefree `s = ∏P`:
   12* `norm_K_le_KW`: `‖K_s(b,c)‖ ≤ KW(c) := 4^{|P|/2}·s^{1/2}·(∏_{p∈P, p∣c} p)^{1/2}`
   13  (`τ(s) = 2^{|P|}` and `gcd(gcd(b,c),s) ≤ gcd(c,s) = ∏_{p∈P,p∣c}p`);
   14* `KW_offdiag`: the off-diagonal frequency `c₀(ȳ−ȳ')` (via `prod_freq_le`);

   16* `W_cell_coupled_W`: the coupled W′ Cauchy bound with the Weil majorant. Equal-index pairs and
   17  distinct-but-congruent pairs are kept (`L·Σ‖α‖²` and `L·2(1+log Y)²/s`); `G₀ = ∏_{p∣c₀}p`
   18  stays explicit (no zero-mode deletion).
   19

   34    ‖GoldC.QuillonKl.K s a b‖ ≤

   37lemma card_divisors_prod (P : Finset ℕ) (hP : ∀ p ∈ P, p.Prime) :
   38    (∏ p ∈ P, p).divisors.card = 2 ^ P.card := by
   39  induction P using Finset.induction_on with
   40  | empty => simp
   41  | insert a P ha ih =>
   42    rw [prod_insert ha, card_insert_of_notMem ha, pow_succ]
   43    have hap : a.Prime := hP a (mem_insert_self _ _)
   44    have hP' : ∀ p ∈ P, p.Prime := fun p hp => hP p (mem_insert_of_mem hp)
   45    have hcop : a.Coprime (∏ p ∈ P, p) := by
   46      apply Nat.Coprime.prod_right
   47      intro p hp
   48      exact (Nat.coprime_primes hap (hP' p hp)).2 (fun h => ha (h ▸ hp))
   49    rw [Nat.Coprime.card_divisors_mul hcop, ih hP', Nat.Prime.divisors hap,
   50      card_pair (Nat.Prime.one_lt hap).ne]
   51    ring

   57lemma gcd_le_filter_prod (P : Finset ℕ) (hP : ∀ p ∈ P, p.Prime) (s : ℕ) (hs : s = ∏ p ∈ P, p) (x : ℕ) :
   58    Nat.gcd x s ≤ ∏ p ∈ P.filter (fun p => p ∣ x), p := by
   59  classical
   60  have hpos : 0 < ∏ p ∈ P.filter (fun p => p ∣ x), p := prod_pos fun p hp => (hP p (mem_filter.1 hp).1).pos
   61  apply Nat.le_of_dvd hpos
   62  have hsplit : s = (∏ p ∈ P.filter (fun p => p ∣ x), p) * ∏ p ∈ P.filter (fun p => ¬ p ∣ x), p := by
   63    rw [hs, prod_filter_mul_prod_filter_not]
   64  have hcop : Nat.Coprime (Nat.gcd x s) (∏ p ∈ P.filter (fun p => ¬ p ∣ x), p) := by
   65    apply Nat.Coprime.prod_right
   66    intro p hp
   67    obtain ⟨hpP, hpx⟩ := mem_filter.1 hp
   68    apply Nat.Coprime.symm
   69    rw [Nat.Prime.coprime_iff_not_dvd (hP p hpP)]
   70    exact fun h => hpx (h.trans (Nat.gcd_dvd_left _ _))
   71  have hd : Nat.gcd x s ∣ (∏ p ∈ P.filter (fun p => p ∣ x), p) * ∏ p ∈ P.filter (fun p => ¬ p ∣ x), p := by
   72    rw [← hsplit]; exact Nat.gcd_dvd_right _ _
   73  exact hcop.dvd_of_dvd_mul_right hd

   98theorem norm_K_le_KW (hW : WeilHyp) (P : Finset ℕ) (hP : ∀ p ∈ P, p.Prime) (s : ℕ) (hs : s = ∏ p ∈ P, p)
   99    [NeZero s] (b c : ZMod s) : ‖K s b c‖ ≤ KW P c := by
  100  have h1 := hW s b c
  101  have hs0 : 0 < s := Nat.pos_of_ne_zero (NeZero.ne s)
  102  have hτ : (s.divisors.card : ℝ) = (4 : ℝ) ^ ((P.card : ℝ) / 2) := by
  103    rw [hs, card_divisors_prod P hP]
  104    push_cast
  105    rw [show (4 : ℝ) = 2 ^ (2 : ℝ) by norm_num, ← Real.rpow_mul (by norm_num),
  106      show (2 : ℝ) * ((P.card : ℝ) / 2) = ((P.card : ℕ) : ℝ) by ring, Real.rpow_natCast]
  107  have hg : (Nat.gcd (Nat.gcd b.val c.val) s : ℝ) ≤
  108      ∏ p ∈ P.filter (fun p => (ZMod.cast c : ZMod p) = 0), (p : ℝ) := by
  109    have a1 := gcd_gcd_le b.val c.val s hs0
  110    have a2 := gcd_le_filter_prod P hP s hs c.val
  111    rw [filter_cast_eq P s hs c] at a2
  112    have : Nat.gcd (Nat.gcd b.val c.val) s ≤ ∏ p ∈ P.filter (fun p => (ZMod.cast c : ZMod p) = 0), p :=
  113      a1.trans a2
  114    rw [← Nat.cast_prod]; exact_mod_cast this
  115  refine h1.trans ?_
  116  unfold KW
  117  rw [hτ, mul_assoc]
  118  apply mul_le_mul_of_nonneg_left _ (by positivity)
  119  rw [Real.sqrt_eq_rpow, Real.mul_rpow (by positivity) (by positivity), mul_comm]
  120  apply mul_le_mul_of_nonneg_left _ (by positivity)
  121  exact Real.rpow_le_rpow (by positivity) hg (by norm_num)

  146/-- Seravyn R0002 half-power gcd pair moment (ported verbatim). -/
  147theorem gcd_moment_half (P : Finset ℕ) (hP : ∀ p ∈ P, p.Prime) (Y : ℕ) :
  148    ∑ y ∈ Icc 1 Y, ∑ y' ∈ (Icc 1 Y).filter (fun y' => y ≠ y'),
  149      (1 / ((y : ℝ) * y')) *
  150        (∏ p ∈ P.filter (fun p : ℕ => (p : ℤ) ∣ (y : ℤ) - y'), (p : ℝ)) ^ ((1 : ℝ) / 2)
  151    ≤ 2 * (1 + Real.log Y) ^ 2 * ∏ p ∈ P, (1 + (p : ℝ) ^ (-(1 : ℝ) / 2)) := by
  152  classical
  153  set a : Finset ℕ → ℝ := fun D => ∏ p ∈ D, ((p : ℝ) ^ ((1 : ℝ) / 2) - 1) with ha_def
  154  have hq : ∀ p ∈ P, (1 : ℝ) ≤ (p : ℝ) ^ ((1 : ℝ) / 2) := fun p hp =>
  155    Real.one_le_rpow (by exact_mod_cast (hP p hp).one_lt.le) (by norm_num)
  156  have ha : ∀ D ∈ P.powerset, 0 ≤ a D := fun D hD => prod_nonneg fun p hp => by
  157    have := hq p (mem_powerset.1 hD hp); linarith
  158  set ind : Finset ℕ → ℕ → ℕ → ℝ := fun D y y' =>
  159    if ((∏ p ∈ D, p : ℕ) : ℤ) ∣ (y : ℤ) - y' then 1 else 0 with hind
  160  have step1 : ∀ y y' : ℕ,
  161      (∏ p ∈ P.filter (fun p : ℕ => (p : ℤ) ∣ (y : ℤ) - y'), (p : ℝ)) ^ ((1 : ℝ) / 2) ≤
  162        ∑ D ∈ P.powerset, a D * ind D y y' := by
  163    intro y y'
  164    rw [prod_rpow_half_eq]
  165    simp only [hind, mul_ite, mul_one, mul_zero]
  166    rw [← Finset.sum_filter]
  167    apply Finset.sum_le_sum_of_subset_of_nonneg
  168    · intro D hD
  169      rw [mem_powerset] at hD
  170      rw [mem_filter, mem_powerset]
  171      refine ⟨hD.trans (filter_subset _ _), ?_⟩
  172      apply prime_prod_dvd D (fun p hp => hP p (filter_subset _ _ (hD hp)))
  173      intro p hp; exact (mem_filter.1 (hD hp)).2
  174    · intro D hD _
  175      exact ha D (mem_filter.1 hD).1
  176  have hY := fun (D : Finset ℕ) (hD : D ∈ P.powerset) =>
  177    gcd_pair_sum_sym (∏ p ∈ D, p) Y
  178      (Nat.one_le_iff_ne_zero.2 (Finset.prod_ne_zero_iff.2 fun p hp =>
  179        (hP p (mem_powerset.1 hD hp)).ne_zero))
  180  calc _ ≤ ∑ y ∈ Icc 1 Y, ∑ y' ∈ (Icc 1 Y).filter (fun y' => y ≠ y'),
  181        ∑ D ∈ P.powerset, a D * ((1 / ((y : ℝ) * y')) * ind D y y') := by
  182        refine sum_le_sum fun y hy => sum_le_sum fun y' hy' => ?_
  183        have hpos : (0 : ℝ) ≤ 1 / ((y : ℝ) * y') := by positivity
  184        calc _ ≤ (1 / ((y : ℝ) * y')) * ∑ D ∈ P.powerset, a D * ind D y y' :=
  185              mul_le_mul_of_nonneg_left (step1 y y') hpos
  186          _ = _ := by rw [mul_sum]; exact sum_congr rfl fun D _ => by ring
  187    _ = ∑ D ∈ P.powerset, a D * ∑ y ∈ Icc 1 Y,
  188          ∑ y' ∈ (Icc 1 Y).filter (fun y' => y ≠ y' ∧
  189            ((∏ p ∈ D, p : ℕ) : ℤ) ∣ (y : ℤ) - y'), (1 / ((y : ℝ) * y')) := by
  190        rw [sum_congr rfl fun y _ => sum_comm, sum_comm]
  191        refine sum_congr rfl fun D _ => ?_
  192        rw [mul_sum]
  193        refine sum_congr rfl fun y _ => ?_
  194        rw [mul_sum, ← filter_filter,
  195          Finset.sum_filter (p := fun y' : ℕ => ((∏ p ∈ D, p : ℕ) : ℤ) ∣ (y : ℤ) - y')]
  196        refine sum_congr rfl fun y' _ => ?_
  197        simp only [hind]; split_ifs <;> simp
  198    _ ≤ ∑ D ∈ P.powerset, a D * (2 * (1 + Real.log Y) ^ 2 / ((∏ p ∈ D, p : ℕ) : ℝ)) :=
  199        sum_le_sum fun D hD => mul_le_mul_of_nonneg_left (hY D hD) (ha D hD)
  200    _ ≤ ∑ D ∈ P.powerset, 2 * (1 + Real.log Y) ^ 2 * ∏ p ∈ D, (p : ℝ) ^ (-(1 : ℝ) / 2) := by
  201        refine sum_le_sum fun D hD => ?_
  202        have hDP := mem_powerset.1 hD
  203        have hC : (0 : ℝ) ≤ 2 * (1 + Real.log Y) ^ 2 := by positivity
  204        have key : a D / ((∏ p ∈ D, p : ℕ) : ℝ) ≤ ∏ p ∈ D, (p : ℝ) ^ (-(1 : ℝ) / 2) := by
  205          rw [ha_def, Nat.cast_prod, ← prod_div_distrib]
  206          apply Finset.prod_le_prod₀
  207          · intro p hp
  208            have := hq p (hDP hp)
  209            have : (0 : ℝ) < p := by exact_mod_cast (hP p (hDP hp)).pos
  210            apply div_nonneg <;> linarith
  211          · intro p hp
  212            have hp0 : (0 : ℝ) < p := by exact_mod_cast (hP p (hDP hp)).pos
  213            calc ((p : ℝ) ^ ((1 : ℝ) / 2) - 1) / p ≤ (p : ℝ) ^ ((1 : ℝ) / 2) / p :=
  214                  div_le_div_of_nonneg_right (by linarith) hp0.le
  215              _ = (p : ℝ) ^ (-(1 : ℝ) / 2) := by
  216                  rw [← Real.rpow_sub_one hp0.ne']; norm_num
  217        calc a D * (2 * (1 + Real.log Y) ^ 2 / ((∏ p ∈ D, p : ℕ) : ℝ))
  218            = 2 * (1 + Real.log Y) ^ 2 * (a D / ((∏ p ∈ D, p : ℕ) : ℝ)) := by ring
  219          _ ≤ _ := mul_le_mul_of_nonneg_left key hC
  220    _ = _ := by rw [← mul_sum, ← Finset.prod_one_add]

  225/-- **W5, coupled.** Same bound as `W_cell`, with the coupled range `n ∈ (L_y, R_y]`. -/
  226theorem W_cell_coupled_W {s : ℕ} [NeZero s] (hs2 : 2 ≤ s) (P : Finset ℕ) (hP : ∀ p ∈ P, p.Prime)
  227    (hsP : s = ∏ p ∈ P, p) (hW : WeilHyp) (c₀ : ZMod s) (𝒴 𝒩 : Finset ℕ) (Y N M : ℕ) (hNM : N ≤ M)
  228    (h𝒩 : 𝒩 ⊆ Ioc N M) (h𝒴 : 𝒴 ⊆ Icc 1 Y) (hunit : ∀ y ∈ 𝒴, IsUnit (y : ZMod s))
  229    (α β : ℕ → ℂ) (A : ℝ) (hα : ∀ y ∈ 𝒴, ‖α y‖ ≤ A / y) (L R : ℕ → ℕ) :

  231      (∑ n ∈ 𝒩, ‖β n‖ ^ 2) * (((M - N : ℕ) : ℝ) * ∑ y ∈ 𝒴, ‖α y‖ ^ 2 +
  232        A ^ 2 * (((M - N : ℕ) : ℝ) * (2 * (1 + Real.log Y) ^ 2 / s) +
  233          (4 : ℝ) ^ ((P.card : ℝ) / 2) * (s : ℝ) ^ ((1 : ℝ) / 2) *
  234            (∏ p ∈ P.filter (fun p => (ZMod.cast c₀ : ZMod p) = 0), (p : ℝ)) ^ ((1 : ℝ) / 2) *
  235            (((M - N : ℕ) : ℝ) / s + 1 + Real.log s) *
  236            (2 * (1 + Real.log Y) ^ 2 * ∏ p ∈ P, (1 + (p : ℝ) ^ (-(1 : ℝ) / 2))))) := by
  237  classical
  238  refine (W_cauchy_coupled hs2 c₀ 𝒴 𝒩 hNM h𝒩 hunit α β L R (KW P)
  239    (fun c b => norm_K_le_KW hW P hP s hsP b c)).trans ?_
  240  apply mul_le_mul_of_nonneg_left _ (sum_nonneg fun n _ => by positivity)
  241  set L : ℝ := ((M - N : ℕ) : ℝ) with hL
  242  set G : ℝ := (∏ p ∈ P.filter (fun p => (ZMod.cast c₀ : ZMod p) = 0), (p : ℝ)) ^ ((1 : ℝ) / 2)
  243    with hG
  244  set Kc : ℝ := (4 : ℝ) ^ ((P.card : ℝ) / 2) * (s : ℝ) ^ ((1 : ℝ) / 2) * G *
  245    (L / s + 1 + Real.log s) with hKc
  246  have hL0 : 0 ≤ L := by positivity
  247  have hsR : (1 : ℝ) ≤ s := by exact_mod_cast (show 1 ≤ s by omega)
  248  have hlogs : 0 ≤ Real.log s := Real.log_nonneg hsR
  249  have hG0 : 0 ≤ G := by positivity
  250  have hKc0 : 0 ≤ Kc := by positivity
  251  have hA : ∀ y ∈ 𝒴, 0 ≤ A / y := fun y hy => (norm_nonneg _).trans (hα y hy)
  252  set T1 : ℕ → ℕ → ℝ := fun y y' => if y = y' then L * ‖α y‖ ^ 2 else 0 with hT1
  253  set T2 : ℕ → ℕ → ℝ := fun y y' =>
  254    if y ≠ y' ∧ ((s : ℕ) : ℤ) ∣ (y : ℤ) - y' then 1 / ((y : ℝ) * y') else 0 with hT2
  255  set T3 : ℕ → ℕ → ℝ := fun y y' => if y ≠ y' then (1 / ((y : ℝ) * y')) *
  256    (∏ p ∈ P.filter (fun p : ℕ => (p : ℤ) ∣ (y : ℤ) - y'), (p : ℝ)) ^ ((1 : ℝ) / 2) else 0
  257    with hT3
  258  have hT2n : ∀ y y', 0 ≤ T2 y y' := fun y y' => by simp only [hT2]; split_ifs <;> positivity
  259  have hT3n : ∀ y y', 0 ≤ T3 y y' := fun y y' => by simp only [hT3]; split_ifs <;> positivity
  260  have hT1n : ∀ y y', 0 ≤ T1 y y' := fun y y' => by simp only [hT1]; split_ifs <;> positivity
  261  -- pointwise
  262  have hpt : ∀ y ∈ 𝒴, ∀ y' ∈ 𝒴, ‖α y‖ * ‖α y'‖ *
  263      (if (y : ZMod s) = y' then L else
  264        KW P (c₀ * ((y : ZMod s)⁻¹ - (y' : ZMod s)⁻¹)) * (L / s + 1 + Real.log s)) ≤
  265      T1 y y' + A ^ 2 * L * T2 y y' + A ^ 2 * Kc * T3 y y' := by
  266    intro y hy y' hy'
  267    have hy1 : 1 ≤ y := (mem_Icc.1 (h𝒴 hy)).1
  268    have hy1' : 1 ≤ y' := (mem_Icc.1 (h𝒴 hy')).1

  271            mul_le_mul (hα y hy) (hα y' hy') (norm_nonneg _) (hA y hy)
  272        _ = _ := by ring
  273    have h2 := hT2n y y'; have h3 := hT3n y y'; have h1 := hT1n y y'
  274    have hA2 : 0 ≤ A ^ 2 := sq_nonneg A
  275    by_cases hyy : y = y'
  276    · subst hyy
