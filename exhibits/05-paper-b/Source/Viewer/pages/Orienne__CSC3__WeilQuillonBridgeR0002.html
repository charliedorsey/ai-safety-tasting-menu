    7/-- Exact normalization identity: Quillon's complete unit-character sum
    8is the RML/GenKloosterman complete unit-character sum, with both phase
    9coefficients in the same order. -/
   10theorem quillon_K_eq_rml (s : ℕ) [NeZero s] (a b : ZMod s) :
   11    GoldC.QuillonKl.K s a b = GenKloosterman.kSum s b a := by
   12  rfl
   13
   14/-- General-modulus Weil–Estermann bound on the actual complete
   15Quillon Kloosterman sum, with full gcd conductor retained. -/
   16theorem norm_quillon_K_weil (s : ℕ) [NeZero s] (a b : ZMod s) :
   17    ‖GoldC.QuillonKl.K s a b‖ ≤
   18      (s.divisors.card : ℝ) *
   19        Real.sqrt (Nat.gcd (Nat.gcd a.val b.val) s * s) := by
   20  rw [quillon_K_eq_rml]
   21  simpa only [GenKloosterman.kgcd, Nat.gcd_comm b.val a.val] using (GenKloosterman.norm_kSum_le s b a)
   22
   23#print axioms quillon_K_eq_rml
   24#print axioms norm_quillon_K_weil
