26/-- Boolean Möbius transform on the interval below `s`.
27
28The exponent is written as `card (s \ t)` because this is convenient for the
29finite-set recursion. `mobius_eq_paper_formula` below proves that on the
30powerset summation range this is exactly the frozen paper exponent
31`|s| - |t|`.
32-/
33def mobius [DecidableEq ι] [AddCommGroup A]
34    (f : Finset ι → A) (s : Finset ι) : A :=
35  ∑ t ∈ s.powerset, ((-1 : ℤ) ^ (s \ t).card) • f t
36
37/-- The implementation of `mobius` is literally the frozen paper formula
38`∑_{t ⊆ s} (-1)^(|s|-|t|) f(t)` after the finite-cardinality identity
39`|s \ t| = |s| - |t|` on `t ⊆ s`.
40-/
41theorem mobius_eq_paper_formula [DecidableEq ι] [AddCommGroup A]
42    (f : Finset ι → A) (s : Finset ι) :
43    mobius f s =
44      ∑ t ∈ s.powerset, ((-1 : ℤ) ^ (s.card - t.card)) • f t := by
45  apply Finset.sum_congr rfl
46  intro t ht
47  have hts : t ⊆ s := Finset.mem_powerset.mp ht
48  rw [Finset.card_sdiff_of_subset hts]
49
50/-- Möbius transform as an operator on Boolean-fiber functions. -/
51def mobiusOp [DecidableEq ι] [AddCommGroup A]
52    (f : Finset ι → A) : Finset ι → A :=
53  fun s => mobius f s

55/-- Evaluation of a function-valued Boolean Möbius transform is pointwise. -/
56theorem mobius_apply [DecidableEq ι] [AddCommGroup A]
57    {X : Type*} (f : Finset ι → X → A) (s : Finset ι) (x : X) :
58    mobius f s x = mobius (fun t => f t x) s := by
59  simp [mobius, Finset.sum_apply]

61/-- Zeta contraction to the top of a finite Boolean fiber. -/
62def zetaTop [Fintype ι] [DecidableEq ι] [AddCommMonoid A]
63    (f : Finset ι → A) : A :=
64  ∑ s ∈ (Finset.univ : Finset ι).powerset, f s

66private theorem mobius_insert [DecidableEq ι] [AddCommGroup A]
67    (f : Finset ι → A) (a : ι) (s : Finset ι) (ha : a ∉ s) :
68    mobius f (insert a s) =
69      mobius (fun t => f (insert a t) - f t) s := by
70  rw [mobius, Finset.sum_powerset_insert ha, mobius]
71  rw [← Finset.sum_add_distrib]
72  apply Finset.sum_congr rfl
73  intro t ht
74  have ht_sub : t ⊆ s := Finset.mem_powerset.mp ht
75  have hat : a ∉ t := fun h => ha (ht_sub h)
76  have ha_diff : a ∉ s \ t := fun h => ha (Finset.sdiff_subset h)
77  have hleft : insert a s \ t = insert a (s \ t) :=
78    Finset.insert_sdiff_of_notMem s hat
79  have hright : insert a s \ insert a t = s \ t := by
80    rw [Finset.insert_sdiff_insert, Finset.sdiff_insert_of_notMem ha]
81  rw [hleft, Finset.card_insert_of_notMem ha_diff, hright, pow_succ]
82  simp only [mul_neg, mul_one, neg_zsmul, zsmul_sub]
83  abel

85/-- Finite Boolean zeta followed by Möbius, evaluated at an arbitrary top `s`. -/
86theorem zeta_mobius_on [DecidableEq ι] [AddCommGroup A]
87    (f : Finset ι → A) (s : Finset ι) :
88    ∑ t ∈ s.powerset, mobius f t = f s := by
89  induction s using Finset.induction_on generalizing f with
90  | empty => simp [mobius]
91  | @insert a s ha ih =>
92      rw [Finset.sum_powerset_insert ha]
93      rw [ih f]
94      have hsecond :
95          (∑ t ∈ s.powerset, mobius f (insert a t)) = f (insert a s) - f s := by
96        calc
97          (∑ t ∈ s.powerset, mobius f (insert a t)) =
98              ∑ t ∈ s.powerset, mobius (fun u => f (insert a u) - f u) t := by
99                apply Finset.sum_congr rfl
100                intro t ht
101                have ht_sub : t ⊆ s := Finset.mem_powerset.mp ht
102                exact mobius_insert f a t (fun hat => ha (ht_sub hat))
103          _ = f (insert a s) - f s := ih (fun u => f (insert a u) - f u)
104      rw [hsecond]
105      abel

107/-- Global Boolean zeta–Möbius identity at the full finite endpoint fiber. -/
108theorem zeta_mobius [Fintype ι] [DecidableEq ι] [AddCommGroup A]
109    (f : Finset ι → A) :
110    zetaTop (mobiusOp f) = f Finset.univ := by
111  simpa [zetaTop, mobiusOp] using zeta_mobius_on f (Finset.univ : Finset ι)

122/-- Free-fiber Möbius transform `M_T`.
123
124It is polymorphic in its codomain. In the left side of the paper identity the
125codomain is the endpoint-fiber function space `Finset E → A`; after `ev_E` it
126is simply `A`. This makes the same Lean operator represent both typed uses of
127paper notation `M_T`.
128-/
129def mobiusT {X : Type*} [AddCommGroup X]
130    (f : Finset T → X) : Finset T → X :=
131  Boolean.mobiusOp f
132
133/-- Endpoint-fiber Möbius transform `M_E`, acting pointwise in the free fiber. -/
134def mobiusE (f : Finset T → Finset E → A) : Finset T → Finset E → A :=
135  fun t => Boolean.mobiusOp (f t)
136
137/-- Endpoint zeta/contraction `Z_E`, acting pointwise in the free fiber. -/
138def zetaE (g : Finset T → Finset E → A) : Finset T → A :=
139  fun t => Boolean.zetaTop (g t)
140
141/-- Evaluation at the full endpoint set, `ev_E`. -/
142def evE (f : Finset T → Finset E → A) : Finset T → A :=
143  fun t => f t Finset.univ
144
145/-- Pointwise form of the frozen endpoint-contraction identity.
146
147For every free Boolean choice `t`, this states
148`(Z_E M_E M_T f)(t) = (M_T ev_E f)(t)`.
149-/
150theorem endpointContractionPointwise
151    (f : Finset T → Finset E → A) (t : Finset T) :
152    zetaE (mobiusE (mobiusT f)) t = mobiusT (evE f) t := by
153  calc
154    zetaE (mobiusE (mobiusT f)) t = (mobiusT f t) Finset.univ := by
155      exact Boolean.zeta_mobius (mobiusT f t)
156    _ = mobiusT (evE f) t := by
157      change Boolean.mobius f t (Finset.univ : Finset E) =
158        Boolean.mobius (fun u => f u (Finset.univ : Finset E)) t
159      exact Boolean.mobius_apply f t (Finset.univ : Finset E)
160
161/-- Operator-form correspondence with the frozen paper display
162`Z_E M_E M_T = M_T ev_E`.
163
164This is an equality of the resulting functions on the free Boolean fiber.
165-/
166theorem endpointContraction
167    (f : Finset T → Finset E → A) :
168    zetaE (mobiusE (mobiusT f)) = mobiusT (evE f) := by
169  funext t
170  exact endpointContractionPointwise f t

172end EndpointContraction
