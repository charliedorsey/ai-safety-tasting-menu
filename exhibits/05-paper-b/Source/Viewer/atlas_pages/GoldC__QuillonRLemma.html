94theorem cauchy_class_split {Q : ℕ} (hQ : 1 ≤ Q) (𝒰 I 𝒱 : Finset ℕ) (h𝒰 : 𝒰 ⊆ I)
95    (a b : ℕ → ℂ) (φ : ℕ → ℕ → ℝ) (G : ℕ → ℕ → ℂ) (hG : ∀ v ρ, ‖G v ρ‖ ≤ 1) :
96    ‖∑ u ∈ 𝒰, b u * ∑ v ∈ 𝒱, a v * (eR (φ v u) * G v (u % Q))‖ ^ 2 ≤
97      (∑ u ∈ 𝒰, ‖b u‖ ^ 2) *
98        ∑ v ∈ 𝒱, ∑ v' ∈ 𝒱, ‖a v‖ * ‖a v'‖ *
99          ∑ ρ ∈ range Q, ‖∑ u ∈ I.filter (fun u => u % Q = ρ), eR (φ v u - φ v' u)‖ := by
100  set F : ℕ → ℂ := fun u => ∑ v ∈ 𝒱, a v * (eR (φ v u) * G v (u % Q)) with hF
101  -- Cauchy–Schwarz
102  have hCS : ‖∑ u ∈ 𝒰, b u * F u‖ ^ 2 ≤ (∑ u ∈ 𝒰, ‖b u‖ ^ 2) * ∑ u ∈ 𝒰, ‖F u‖ ^ 2 := by
103    have h1 : ‖∑ u ∈ 𝒰, b u * F u‖ ≤ ∑ u ∈ 𝒰, ‖b u‖ * ‖F u‖ :=
104      (norm_sum_le _ _).trans (le_of_eq (Finset.sum_congr rfl fun u _ => norm_mul _ _))
105    have h3 : (∑ u ∈ 𝒰, ‖b u‖ * ‖F u‖) ^ 2 ≤
106        (∑ u ∈ 𝒰, ‖b u‖ ^ 2) * ∑ u ∈ 𝒰, ‖F u‖ ^ 2 :=
107      Finset.sum_mul_sq_le_sq_mul_sq 𝒰 (fun u => ‖b u‖) (fun u => ‖F u‖)
108    calc ‖∑ u ∈ 𝒰, b u * F u‖ ^ 2 ≤ (∑ u ∈ 𝒰, ‖b u‖ * ‖F u‖) ^ 2 := by
109          gcongr
110      _ ≤ _ := h3
111  refine hCS.trans (mul_le_mul_of_nonneg_left ?_ (Finset.sum_nonneg fun u _ => by positivity))
112  -- drop the mask: Σ_{𝒰} ‖F‖² ≤ Σ_I ‖F‖²
113  have hdrop : ∑ u ∈ 𝒰, ‖F u‖ ^ 2 ≤ ∑ u ∈ I, ‖F u‖ ^ 2 :=
114    Finset.sum_le_sum_of_subset_of_nonneg h𝒰 fun u _ _ => by positivity
115  refine hdrop.trans ?_
116  -- expand ‖F u‖² = Σ_{v,v'} a_v conj(a_v') e(φ v u - φ v' u) G v ρ conj(G v' ρ)
117  have hexp : ∀ u, ((‖F u‖ ^ 2 : ℝ) : ℂ) = ∑ v ∈ 𝒱, ∑ v' ∈ 𝒱,
118      a v * (starRingEnd ℂ) (a v') * (eR (φ v u - φ v' u) *
119        (G v (u % Q) * (starRingEnd ℂ) (G v' (u % Q)))) := by
120    intro u
121    rw [Complex.ofReal_pow, ← Complex.mul_conj', hF]
122    simp only [map_sum, map_mul]
123    rw [Finset.sum_mul_sum]
124    refine Finset.sum_congr rfl fun v _ => Finset.sum_congr rfl fun v' _ => ?_
125    rw [eR_sub]; ring
126  -- regroup by classes
127  have hreal : ∑ u ∈ I, ‖F u‖ ^ 2 = ‖∑ u ∈ I, ((‖F u‖ ^ 2 : ℝ) : ℂ)‖ := by
128    rw [← Complex.ofReal_sum, Complex.norm_real, Real.norm_eq_abs,
129      abs_of_nonneg (Finset.sum_nonneg fun u _ => by positivity)]
130  rw [hreal]
131  simp_rw [hexp]
132  rw [Finset.sum_comm]
133  refine (norm_sum_le _ _).trans (Finset.sum_le_sum fun v _ => ?_)
134  rw [Finset.sum_comm]
135  refine (norm_sum_le _ _).trans (Finset.sum_le_sum fun v' _ => ?_)
136  -- split I by residue classes
137  have hsplit : ∑ u ∈ I, a v * (starRingEnd ℂ) (a v') * (eR (φ v u - φ v' u) *
138        (G v (u % Q) * (starRingEnd ℂ) (G v' (u % Q)))) =
139      a v * (starRingEnd ℂ) (a v') * ∑ ρ ∈ range Q, (G v ρ * (starRingEnd ℂ) (G v' ρ)) *
140        ∑ u ∈ I.filter (fun u => u % Q = ρ), eR (φ v u - φ v' u) := by
141    rw [← Finset.mul_sum]
142    congr 1
143    rw [← Finset.sum_fiberwise_of_maps_to (g := fun u => u % Q) (t := range Q)
144      (fun u _ => Finset.mem_range.2 (Nat.mod_lt _ (by omega)))]
145    refine Finset.sum_congr rfl fun ρ _ => ?_
146    rw [Finset.mul_sum]
147    refine Finset.sum_congr rfl fun u hu => ?_
148    rw [(Finset.mem_filter.1 hu).2]; ring
149  rw [hsplit, norm_mul, norm_mul, Complex.norm_conj]
150  gcongr
151  refine (norm_sum_le _ _).trans (Finset.sum_le_sum fun ρ _ => ?_)
152  rw [norm_mul, norm_mul, Complex.norm_conj]
153  have h1 := hG v ρ
154  have h2 := hG v' ρ
155  calc ‖G v ρ‖ * ‖G v' ρ‖ * ‖∑ u ∈ I.filter (fun u => u % Q = ρ), eR (φ v u - φ v' u)‖
156      ≤ 1 * 1 * ‖∑ u ∈ I.filter (fun u => u % Q = ρ), eR (φ v u - φ v' u)‖ := by gcongr
157    _ = _ := by ring
