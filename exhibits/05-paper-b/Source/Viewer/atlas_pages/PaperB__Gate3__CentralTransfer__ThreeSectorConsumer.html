107admissible anchor a, original parity condition, and eventually in h. -/
108def SourceUniformSublog (S : ℕ → ℕ → ℕ → ℝ) : Prop :=
109  ∀ q : ℕ, 0 < q → Squarefree q →
110    ∀ eps : ℝ, 0 < eps →
111      ∃ H0 : ℕ, ∀ a h : ℕ,
112        q.Coprime a → 0 < h → (¬ 2 ∣ q → 2 ∣ h) →
113        H0 ≤ h →
114        |S q a h| ≤ eps * Real.log (h : ℝ)

131    (Jlow Jhigh : ℕ → ℕ) (hord : ∀ h, Jlow h ≤ Jhigh h)
132    (hLow : SourceUniformSublog (lowSector Jlow))
133    (hMiddle : SourceUniformSublog (middleSector Jlow Jhigh))
