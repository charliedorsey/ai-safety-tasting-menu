63/-- The standard character factors through CRT. -/
64theorem stdAddChar_crt {m n : ℕ} [NeZero m] [NeZero n] (h : m.Coprime n) (z : ZMod (m * n)) :
65    haveI : NeZero (m * n) := ⟨Nat.mul_ne_zero (NeZero.ne m) (NeZero.ne n)⟩
66    ZMod.stdAddChar z =
67      ZMod.stdAddChar ((ZMod.castHom (dvd_mul_right m n) (ZMod m) z) * (n : ZMod m)⁻¹) *
68      ZMod.stdAddChar ((ZMod.castHom (dvd_mul_left n m) (ZMod n) z) * (m : ZMod n)⁻¹) := by
69  haveI : NeZero (m * n) := ⟨Nat.mul_ne_zero (NeZero.ne m) (NeZero.ne n)⟩
70  obtain ⟨k, hk⟩ := crt_unit_sum (m := m) (n := n) h
71  set n' := ((n : ZMod m)⁻¹).val
72  set m' := ((m : ZMod n)⁻¹).val
73  set v := z.val
74  have hz : z = ((v : ℤ) : ZMod (m * n)) := by simp [v]
75  have h1 : (ZMod.castHom (dvd_mul_right m n) (ZMod m) z) * (n : ZMod m)⁻¹ =
76      (((v * n' : ℕ) : ℤ) : ZMod m) := by
77    rw [hz]; simp [n']
78  have h2 : (ZMod.castHom (dvd_mul_left n m) (ZMod n) z) * (m : ZMod n)⁻¹ =
79      (((v * m' : ℕ) : ℤ) : ZMod n) := by
80    rw [hz]; simp [m']
81  rw [h1, h2, hz, ZMod.stdAddChar_coe, ZMod.stdAddChar_coe, ZMod.stdAddChar_coe,
82    ← Complex.exp_add]
83  -- exponent difference is 2πi·(v k)
84  have hmR : (m : ℂ) ≠ 0 := by exact_mod_cast NeZero.ne m
85  have hnR : (n : ℂ) ≠ 0 := by exact_mod_cast NeZero.ne n
86  have hkC : (n : ℂ) * n' + (m : ℂ) * m' = 1 + (m : ℂ) * n * k := by
87    have := congrArg (fun t : ℤ => (t : ℂ)) hk
88    push_cast at this; exact this
89  have key : 2 * (Real.pi : ℂ) * Complex.I * (((v * n' : ℕ) : ℤ) : ℂ) / (m : ℂ) +
90      2 * (Real.pi : ℂ) * Complex.I * (((v * m' : ℕ) : ℤ) : ℂ) / (n : ℂ) =
91      2 * (Real.pi : ℂ) * Complex.I * (((v : ℕ) : ℤ) : ℂ) / ((m * n : ℕ) : ℂ) +
92        ((v : ℤ) * k : ℤ) * (2 * Real.pi * Complex.I) := by
93    push_cast
94    calc 2 * (Real.pi : ℂ) * Complex.I * ((v : ℂ) * n') / m +
95          2 * (Real.pi : ℂ) * Complex.I * ((v : ℂ) * m') / n
96        = 2 * (Real.pi : ℂ) * Complex.I * v * ((n : ℂ) * n' + (m : ℂ) * m') / ((m : ℂ) * n) := by
97          field_simp; try ring
98      _ = 2 * (Real.pi : ℂ) * Complex.I * v * (1 + (m : ℂ) * n * k) / ((m : ℂ) * n) := by
99          rw [hkC]
100      _ = _ := by field_simp; try ring
101  rw [key, Complex.exp_add, Complex.exp_int_mul_two_pi_mul_I, mul_one]

104theorem K_mul {m n : ℕ} [NeZero m] [NeZero n] (h : m.Coprime n) (a b : ZMod (m * n)) :
105    haveI : NeZero (m * n) := ⟨Nat.mul_ne_zero (NeZero.ne m) (NeZero.ne n)⟩

108          (ZMod.castHom (dvd_mul_right m n) (ZMod m) b * (n : ZMod m)⁻¹) *
109        K n (ZMod.castHom (dvd_mul_left n m) (ZMod n) a * (m : ZMod n)⁻¹)
110          (ZMod.castHom (dvd_mul_left n m) (ZMod n) b * (m : ZMod n)⁻¹) := by
111  haveI : NeZero (m * n) := ⟨Nat.mul_ne_zero (NeZero.ne m) (NeZero.ne n)⟩
112  set f₁ := ZMod.castHom (dvd_mul_right m n) (ZMod m)
113  set f₂ := ZMod.castHom (dvd_mul_left n m) (ZMod n)
114  -- the unit equivalence
115  let E : (ZMod (m * n))ˣ ≃ (ZMod m)ˣ × (ZMod n)ˣ :=
116    ((Units.mapEquiv (ZMod.chineseRemainder h).toMulEquiv).trans MulEquiv.prodUnits).toEquiv
117  have hE1 : ∀ x : (ZMod (m * n))ˣ, ((E x).1 : ZMod m) = f₁ (x : ZMod (m * n)) := by
118    intro x; simp [E, f₁, ZMod.chineseRemainder, MulEquiv.prodUnits, ZMod.castHom_apply]
119  have hE2 : ∀ x : (ZMod (m * n))ˣ, ((E x).2 : ZMod n) = f₂ (x : ZMod (m * n)) := by
120    intro x; simp [E, f₂, ZMod.chineseRemainder, MulEquiv.prodUnits, ZMod.castHom_apply]
121  unfold K
122  rw [← Equiv.sum_comp E.symm, Fintype.sum_prod_type, Finset.sum_mul_sum]
123  refine Finset.sum_congr rfl fun x₁ _ => Finset.sum_congr rfl fun x₂ _ => ?_
124  set y := E.symm (x₁, x₂)
125  have hy : E y = (x₁, x₂) := by simp [y]
126  have hy1 : (x₁ : ZMod m) = f₁ (y : ZMod (m * n)) := by rw [← hE1, hy]
127  have hy2 : (x₂ : ZMod n) = f₂ (y : ZMod (m * n)) := by rw [← hE2, hy]
128  have hyi1 : ((x₁⁻¹ : (ZMod m)ˣ) : ZMod m) = f₁ ((y⁻¹ : (ZMod (m * n))ˣ) : ZMod (m * n)) := by
129    have : f₁ ((y⁻¹ : (ZMod (m * n))ˣ) : ZMod (m * n)) * f₁ (y : ZMod (m * n)) = 1 := by
130      rw [← map_mul, Units.inv_mul, map_one]

132        = ((x₁⁻¹ : (ZMod m)ˣ) : ZMod m) * (f₁ ((y⁻¹ : (ZMod (m * n))ˣ) : ZMod (m * n)) *
133            f₁ (y : ZMod (m * n))) := by rw [this, mul_one]
134      _ = f₁ ((y⁻¹ : (ZMod (m * n))ˣ) : ZMod (m * n)) *
135            (((x₁⁻¹ : (ZMod m)ˣ) : ZMod m) * (x₁ : ZMod m)) := by rw [hy1]; ring
136      _ = _ := by rw [Units.inv_mul, mul_one]
137  have hyi2 : ((x₂⁻¹ : (ZMod n)ˣ) : ZMod n) = f₂ ((y⁻¹ : (ZMod (m * n))ˣ) : ZMod (m * n)) := by
138    have : f₂ ((y⁻¹ : (ZMod (m * n))ˣ) : ZMod (m * n)) * f₂ (y : ZMod (m * n)) = 1 := by
139      rw [← map_mul, Units.inv_mul, map_one]
140    calc ((x₂⁻¹ : (ZMod n)ˣ) : ZMod n)
141        = ((x₂⁻¹ : (ZMod n)ˣ) : ZMod n) * (f₂ ((y⁻¹ : (ZMod (m * n))ˣ) : ZMod (m * n)) *
