22lemma norm_main_le (μ : ℝ) (k s : ℕ) (hk : 1 ≤ k) (hs : 1 ≤ s) :
23    ‖((-(2 * μ) : ℝ) : ℂ) * αm k s‖ ≤ |μ| / k := by
24  rw [norm_mul, Complex.norm_real, Real.norm_eq_abs, abs_neg, abs_mul, abs_two]
25  have h := norm_αm_le k s hk hs
26  have hkR : (1 : ℝ) ≤ k := by exact_mod_cast hk
27  have hpi : (2 : ℝ) ≤ Real.pi := Real.two_le_pi
28  calc 2 * |μ| * ‖αm k s‖ ≤ 2 * |μ| * (1 / (2 * Real.pi * k)) := by gcongr
29    _ ≤ |μ| / k := by
30        rw [div_eq_mul_one_div |μ|]
31        have : 2 * |μ| * (1 / (2 * Real.pi * k)) = |μ| * (1 / (Real.pi * k)) := by field_simp

33        apply mul_le_mul_of_nonneg_left _ (abs_nonneg _)

36lemma norm_bS_le (M : ℕ) (Γ : ℕ → ℕ → ℝ) (Gb : ℝ) (hGb : ∀ b ν, |Γ b ν| ≤ Gb) (s b : ℕ) (z : ℤ)
37    (ρ : ZMod M) : ‖bS M Γ s b z ρ‖ ≤ 2 * Gb := by
38  unfold bS
39  rw [norm_mul, norm_mul, norm_eR, one_mul, Complex.norm_real, Real.norm_eq_abs]
40  have h1 : ‖eR ((z : ℝ) * ((M - 1 : ℕ) : ℝ) / M) - 1‖ ≤ 2 := by

42      _ = 2 := by rw [norm_eR, norm_one]; norm_num

48lemma norm_BclS_le (M : ℕ) (Γ : ℕ → ℕ → ℝ) (Gb : ℝ) (hGb : ∀ b ν, |Γ b ν| ≤ Gb) (s : ℕ) (z : ℤ)
49    (ρ : ZMod M) : ‖BclS M Γ s z ρ‖ ≤ 2 * M * Gb := by
50  unfold BclS
51  refine (norm_sum_le _ _).trans ?_
52  calc ∑ b ∈ range M, ‖bS M Γ s b z ρ‖ ≤ ∑ _b ∈ range M, 2 * Gb :=
53        sum_le_sum fun b _ => norm_bS_le M Γ Gb hGb s b z ρ
54    _ = 2 * M * Gb := by rw [sum_const, card_range, nsmul_eq_mul]; ring

56lemma norm_class_le (M : ℕ) (Γ : ℕ → ℕ → ℝ) (Gb : ℝ) (hGb : ∀ b ν, |Γ b ν| ≤ Gb) (s k : ℕ) (z : ℤ)
57    (ρ : ZMod M) (hk : 1 ≤ k) (hMs : 1 ≤ M * s) :
58    ‖αm k (M * s) * BclS M Γ s z ρ‖ ≤ M * Gb / k := by
59  rw [norm_mul]
60  have h1 := norm_αm_le k (M * s) hk hMs
61  have h2 := norm_BclS_le M Γ Gb hGb s z ρ
62  have h0 : 0 ≤ Gb := (abs_nonneg _).trans (hGb 0 0)
63  have hkR : (1 : ℝ) ≤ k := by exact_mod_cast hk
64  have hpi : (2 : ℝ) ≤ Real.pi := Real.two_le_pi
65  calc ‖αm k (M * s)‖ * ‖BclS M Γ s z ρ‖ ≤ (1 / (2 * Real.pi * k)) * (2 * M * Gb) :=
66        mul_le_mul h1 h2 (norm_nonneg _) (by positivity)
67    _ ≤ M * Gb / k := by
68        rw [div_eq_mul_one_div (M * Gb : ℝ)]
69        have : 1 / (2 * Real.pi * k) * (2 * M * Gb) = (M * Gb : ℝ) * (1 / (Real.pi * k)) := by field_simp

71        apply mul_le_mul_of_nonneg_left _ (by positivity)
