52/-- The W′ part of the CORE pair sum at decoration `d`. -/
53noncomputable def WPart (Jlow Jhigh q a h : ℕ) (δ : ℝ) (d : Finset ℕ) : ℝ :=

55    if B ≠ ∅ ∧ C ≠ ∅ ∧ Disjoint B C ∧
56        (Mk0 Jlow Jhigh h (h / primeSetProduct d) δ B C ∧ WMask h δ (Tsrc q h d) B C) then
