  227/-- `C` misses `J_R(B)` (and `s ≤ r²`): every prime of `C` is `≤ X` or `> s^{ε/2}`. -/
  228lemma C_rough_of (h : ℕ) (hΛ1 : 1 ≤ Real.log h) (T B C : Finset ℕ) (hCB : C ⊆ T \ B)
  229    (hJR : ¬ (C ∩ GoldC.QuillonRPhysical.JR h T B).Nonempty) (hr1 : 1 ≤ ∏ p ∈ B, p)
  230    (hsr2 : ∏ p ∈ C, p ≤ (∏ p ∈ B, p) ^ 2) :
  231    ∀ p ∈ C, p < Xp h + 1 ∨ ((∏ q ∈ C, q : ℕ) : ℝ) ^ (εp h / 2) < p := by
  232  set r := ∏ p ∈ B, p
  233  set s := ∏ p ∈ C, p
  234  obtain ⟨hε0, -⟩ := GoldC.QuillonCPhysical.εp_le_one (h := h) hΛ1
  235  set ε := εp h
  236  have hrR : (1 : ℝ) ≤ r := by exact_mod_cast hr1
  237  have hr0 : (0 : ℝ) < r := by linarith
  238  have hX1 : 1 ≤ Xp h := by
  239    have : (1 : ℝ) ≤ Xp h := le_trans (Real.one_le_rpow hΛ1 (by norm_num)) (Xp_ge hΛ1)
  240    exact_mod_cast this
  241  intro p hp
  242  have hpJ : ¬ (Xp h < p ∧ p ≤ 2 ^ Kvp h r * Xp h) := by
  243    intro hJ
  244    exact hJR ⟨p, mem_inter.2 ⟨hp, mem_filter.2 ⟨hCB hp, hJ⟩⟩⟩
  245  by_cases hpX : p ≤ Xp h
  246  · exact Or.inl (by omega)
  247  · right
  248    push_neg at hpX
  249    have hp2 : 2 ^ Kvp h r * Xp h < p := by
  250      by_contra hc'; push_neg at hc'; exact hpJ ⟨hpX, hc'⟩
  251    have hρ1 : (1 : ℝ) ≤ (r : ℝ) ^ ε := Real.one_le_rpow hrR hε0
  252    obtain ⟨hKv1, -⟩ := Kv_bounds ((r : ℝ) ^ ε) hρ1 (Kvp h r) rfl
  253    have hX1R : (1 : ℝ) ≤ Xp h := by exact_mod_cast hX1
  254    have h3 : (r : ℝ) ^ ε < p := by
  255      have : ((2 ^ Kvp h r * Xp h : ℕ) : ℝ) < p := by exact_mod_cast hp2
  256      push_cast at this
  257      nlinarith
  258    have hsr2' : (s : ℝ) ≤ (r : ℝ) ^ (2 : ℝ) := by
